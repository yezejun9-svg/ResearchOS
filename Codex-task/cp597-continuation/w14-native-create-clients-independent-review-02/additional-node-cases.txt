function controlledNative(){
  const timers=[];const socket=new EventEmitter();socket.destroyed=false;
  socket.setTimeout=delay=>{socket.idleDelay=delay;};socket.write=()=>{};
  socket.destroy=()=>{if(!socket.destroyed){socket.destroyed=true;socket.emit('close');}};
  const context={require:name=>name==='node:net'?{connect:()=>socket}:name==='./http-wire.js'?require(path.join(lib,'http-wire.js')):require(name),
    module:{exports:{}},exports:{},Buffer,TextDecoder,console,setTimeout:(fn,delay)=>{const t={fn,delay,cleared:false};timers.push(t);return t;},clearTimeout:t=>{t.cleared=true;}};
  vm.runInNewContext(fs.readFileSync(path.join(lib,'native.js'),'utf8'),context);
  return {api:context.module.exports,socket,timers};
}
