"""Root observes an actual independent message and registers qualified W07."""
import argparse
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
from pathlib import Path
import subprocess
import sys
import uuid

ROOT = Path(__file__).resolve().parents[6]
HERE = Path(__file__).resolve().parent


def now():
    return datetime.now(timezone.utc).isoformat()


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def ref(path):
    return {"path": path.relative_to(ROOT).as_posix(), "sha256": sha(path)}


def save(path, value, crlf=False):
    data = json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False) + "\n"
    path.write_bytes((data.replace("\n", "\r\n") if crlf else data).encode("utf-8"))


def module(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    value = importlib.util.module_from_spec(spec)
    sys.modules[name] = value
    spec.loader.exec_module(value)
    return value


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--context", type=Path, required=True)
    parser.add_argument("--review", type=Path, required=True)
    parser.add_argument("--received-message", type=Path, required=True)
    args = parser.parse_args()
    context_path, review_path, message_path = [p.resolve(strict=True) for p in (args.context, args.review, args.received_message)]
    prefix = ROOT / "execution/v2.4/receipts/W07/W07-99"
    assert all(p.is_relative_to(prefix) for p in (context_path, review_path, message_path))
    api = module(ROOT / "execution/v2.4/formal_case_index.py", "researchos_root_w07_index")
    reader = api.Reader(ROOT)
    context, review, message = [reader.read(p) for p in (context_path, review_path, message_path)]
    assert context["schema"] == "researchos.w07.current-context.v1"
    assert review["schema"] == "researchos.w07.independent-review.v1"
    assert review["decision"] == "ACCEPTED" and review["missing"] == [] and review["failed"] == []
    assert review["reviewer_id"] == "codex:/root/w05_merge_review"
    assert context["executor_id"] == review["executor_id"] == "codex:/root/sources_parent_impl"
    assert review["context_ref"] == ref(context_path) and review["proof_ref"] == context["proof_ref"]
    assert review["version_vector"] == context["version_vector"]
    assert review["acceptance_baseline_ref"] == context["acceptance_baseline_ref"]
    assert review["evidence_index_ref"] == ref(context_path)
    assert not review.get("supersedes_review_refs")
    assert message["schema"] == "researchos.root-received-independent-message.v1"
    assert message["recorded_by"] == message["recipient"] == "codex:/root"
    assert message["sender"] == review["reviewer_id"] and message["event_type"] in {"MESSAGE", "FINAL_ANSWER"}
    assert message["review_ref"] == ref(review_path)
    assert sha(review_path) in message["content"] and context["proof_ref"]["sha256"] in message["content"]
    assert api.stamp(context["created_at"], "context") <= api.stamp(review["issued_at"], "review") <= api.stamp(message["received_at"], "received")
    profile = api._w07_profile(reader)
    pinned, tools = profile._vector(api._profile_api(), reader, context)
    proof, _ = profile.product(api._profile_api(), reader, context, tools)
    profile._dispatches(api._profile_api(), reader, context, pinned, proof)
    profile._global(api._profile_api(), reader, context, proof, review_path)
    for key, ids in (("assertion_verdicts", profile.IDS), ("gate_verdicts", profile.GATES)):
        rows = review[key]
        api.exact_ids([row["id"] for row in rows], ids, key)
        for row in rows:
            assert row["status"] == "PASS" and api.meaningful(row["actual"]) and row["raw_evidence_refs"]
            for raw_ref in row["raw_evidence_refs"]:
                profile.raw(api._profile_api(), reader, raw_ref)
    state_path = ROOT / "execution/v2.4/step_state.json"
    cp_path = ROOT / "execution/v2.4/checkpoint.json"
    own_path = ROOT / "execution/v2.4/path_ownership.json"
    state, cp, own = [json.loads(p.read_bytes()) for p in (state_path, cp_path, own_path)]
    previous = {p.name: sha(p) for p in (state_path, cp_path, own_path)}
    owner = next(row for row in state["steps"] if row["id"] == "W07-99")
    assert owner["status"] in {"IN_PROGRESS", "EXECUTION_REPORTED"}
    plan = ROOT / "execution/v2.4/receipts/W04/W04-99/executor_codex/frozen_inputs/package/contracts/execution.plan.json"
    plan_value = reader.read(plan)
    step = next(row for row in plan_value["steps"] if row["id"] == "W07-99")
    statuses = {row["id"]: row["status"] for row in state["steps"]}
    assert all(statuses.get(k) == "ACCEPTED" for k in step["dependencies"])
    for reference in owner["evidence"]:
        path = reader.ref({"path": "execution/v2.4/" + reference["path"], "sha256": reference["sha256"]})
        if path.suffix == ".json":
            assert reader.read(path).get("schema") not in profile.SCHEMAS.values()
    reader.finish()
    assert all(sha(p) == previous[p.name] for p in (state_path, cp_path, own_path))
    out = HERE / ("current-issuance-" + uuid.uuid4().hex)
    out.mkdir()
    save(out / "previous-owner-registration.json", {"schema": "researchos.previous-owner-registration.v1", "owner": owner,
        "state_sha256": previous["step_state.json"], "recorded_at": now()})
    issuance = {"schema": "researchos.w07.root-observed-issuance.v1", "task_id": "W07", "step_id": "W07-99",
        "run_id": context["run_id"], "actor": "codex:/root", "executor_id": context["executor_id"],
        "reviewer_id": review["reviewer_id"], "context_ref": ref(context_path), "review_ref": ref(review_path),
        "delivery_ref": ref(message_path), "observed_at": now()}
    save(out / "issuance.json", issuance)
    owner.update(status="ACCEPTED", executor_id=context["executor_id"], verifier_id=review["reviewer_id"], accepted_at=now(),
        note="Actual current source/product/native4/GLOBAL evidence independently accepted and Root-observed. Historical evidence remains immutable.")
    for path in (context_path, review_path, out / "issuance.json"):
        owner["evidence"].append({"path": ref(path)["path"].removeprefix("execution/v2.4/"), "sha256": sha(path)})
    save(state_path, state, crlf=True)
    derived = api.derive_formal_cases(ROOT, state_path, plan)
    save(out / "actual-formal-index.json", derived)
    checked = api.recheck_inputs(derived["dependency_snapshot"])
    save(out / "input-recheck.json", checked)
    w07 = next(row for row in derived["owners"] if row["task_id"] == "W07")
    assert w07["integrity"] == "VALID_ACCEPTED" and w07["accepted_case_ids"] == ["A05"], w07
    assert checked["status"] == "PASS", checked
    assert sha(cp_path) == previous["checkpoint.json"] and sha(own_path) == previous["path_ownership.json"]
    old_revision, old_ownership_revision = cp["revision"], own["revision"]
    cp.update(revision=old_revision + 1, expected_previous_revision=old_revision, updated_at=now(), updated_by="codex:/root",
        head=subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip())
    cp["accepted_steps"] = [row["id"] for row in state["steps"] if row["status"] == "ACCEPTED"]
    assert not any(str(p).startswith("W07") for p in cp["parent_tasks_accepted"])
    cp["parent_tasks_accepted"].append("W07(full owned current LIVE_SERVICE A05 and gates, actual independent acceptance)")
    cp["next_command"] = "Preserve W07 actual independent CURRENT acceptance; register scoped W04 revalidation and continue W05 open Q10/TUI/LIVE obligations without model401/UNKNOWN retries."
    cp["note"] += " | Actual W07 current A05 accepted after independent full review and Root-observed message; other formal owners remain open or stale. All previous failure counters retained."
    for worker in cp["workers"]:
        if worker.get("step_id") == "W07-99" and not worker.get("slice_id"):
            worker.update(state="ACCEPTED_CURRENT_FORMAL_A05", independent_review=ref(review_path), current_context_ref=ref(context_path))
    own.update(revision=old_ownership_revision + 1, updated_at=now(), updated_by="codex:/root")
    own["card_path_bindings"]["W07-99"].update(state="ACCEPTED_CURRENT_FORMAL_A05", current_context_ref=ref(context_path),
        independent_review_ref=ref(review_path), root_issuance_ref=ref(out / "issuance.json"))
    save(out / "registration.json", {"schema": "researchos.root-current-acceptance-registration.v1", "at": now(),
        "actor": "codex:/root", "previous_sha256": previous, "checkpoint_revision": cp["revision"], "ownership_revision": own["revision"],
        "accepted_cards": len(cp["accepted_steps"]), "accepted_parents": len(cp["parent_tasks_accepted"]),
        "formal_current": derived["formal_cases_accepted"], "failure_origins": len(cp["failure_counters"]), "counter_reset": False,
        "context_ref": ref(context_path), "independent_review_ref": ref(review_path), "issuance_ref": ref(out / "issuance.json"),
        "derived_formal_index_ref": ref(out / "actual-formal-index.json")})
    cp["evidence_index"].append(ref(out / "registration.json")["path"])
    save(out / "handoff.json", {"schema": "researchos.root-resume-handoff.v1", "created_at": now(),
        "goal_status": "ACTIVE", "project_id": cp["project_id"], "main_head": cp["head"],
        "checkpoint_revision": cp["revision"], "ownership_revision": own["revision"],
        "registration_ref": ref(out / "registration.json"), "accepted_cards": len(cp["accepted_steps"]),
        "accepted_parents": len(cp["parent_tasks_accepted"]), "formal_current": derived["formal_cases_accepted"],
        "required_parents": 22, "required_formal_cases": 42, "failure_origin_count": len(cp["failure_counters"]),
        "blocked_steps": cp["blocked_steps"], "next_action": cp["next_command"], "counter_reset": False})
    cp["evidence_index"].append(ref(out / "handoff.json")["path"])
    save(cp_path, cp)
    save(own_path, own, crlf=True)
    ledger = module(ROOT / "execution/v2.4/parent_gate_ledger.py", "researchos_root_parent_ledger")
    save(out / "actual-parent-ledger.json", ledger.build(ROOT, plan.parent.parent))
    print(json.dumps({"status": "W07_CURRENT_ACCEPTED", "registration_ref": ref(out / "registration.json"),
        "formal_current": derived["formal_cases_accepted"], "accepted_parents": len(cp["parent_tasks_accepted"])}))


if __name__ == "__main__":
    main()
