HERE = Path(__file__).resolve().parent

SELECTORS = ('W02-G01','W02-G02','W02-G03')


def read(path):
    return json.loads(Path(path).read_text(encoding='utf-8-sig'))


def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream,'sha256').hexdigest()


def save(path,data):
    with Path(path).open('x',encoding='utf-8',newline='\n') as stream:
        json.dump(data,stream,ensure_ascii=False,indent=2); stream.write('\n')


class Gate:
    sha=staticmethod(sha)
    save=staticmethod(save)
    def __init__(self,selector):
        self.selector=selector
        self.out=HERE/('run-'+dt.datetime.now(dt.timezone.utc).strftime('%Y%m%dT%H%M%SZ-')+uuid.uuid4().hex[:12])
        self.out.mkdir(exist_ok=False)
        self.pin=read(HERE/'parent_inputs_codex.json')
        repair=HERE/'g01_repair_inputs_codex.json'
        self.repair=read(repair) if repair.exists() else None
        if self.repair:
            self.pin['build_sources'].update(self.repair['additional_build_sources'])
            self.pin['files'].update(self.repair['additional_build_sources'])
        self.main=Path(self.pin['resource_root'])
        self.spec=Path(self.pin['spec_root'])
        self.rows=[];self.inputs={};self.commands=[];self.pending=False
        self.scratch=self.out/'scratch'
        self.scratch.mkdir()
        self.inputs[str(HERE/'parent_inputs_codex.json')]=sha(HERE/'parent_inputs_codex.json')
        if self.repair:self.inputs[str(repair)]=sha(repair)
        (self.out/'driver-sources').mkdir()
        for name in ('parent_driver.py','dispatcher_fixture.py','dispatch_parent.py'):
            self.inputs[str(HERE/name)]=sha(HERE/name)
            shutil.copyfile(HERE/name,self.out/'driver-sources'/name)

    def source(self,rel):
        path=ROOT/rel; actual=sha(path)
        expected=self.pin['files'].get(rel)
        if expected is not None and expected != actual: raise ValueError('SOURCE_DRIFT:'+rel)
        self.inputs[str(path)]=actual
        return path

    def tool(self,name):
        item=self.pin['tools'][name];path=Path(item['path'])
        if sha(path)!=item['sha256']:raise ValueError('TOOL_DRIFT:'+name)
        self.inputs[str(path)]=item['sha256'];return str(path)

    def check(self,key,ok,detail):
        self.rows.append({'id':self.selector+'-'+key,'ok':ok is True,'detail':detail})

    def env(self):
        allowed=('SystemRoot','SYSTEMROOT','TEMP','TMP','USERPROFILE','LOCALAPPDATA','APPDATA','PROGRAMDATA','ProgramFiles','ProgramFiles(x86)','PATH')
        env={k:os.environ[k] for k in allowed if k in os.environ}
        env.update(PYTHONUTF8='1',PYTHONDONTWRITEBYTECODE='1',GOTOOLCHAIN='local',GOENV='off',GOPROXY='off',GOSUMDB='off',
                   GOFLAGS='-mod=readonly',GOMODCACHE=str(self.main/'.local/gopath/pkg/mod'),
                   GOPATH=str(self.scratch/'gopath'),GOCACHE=str(self.scratch/'gocache'),GOTMPDIR=str(self.scratch/'gotmp'))
        (self.scratch/'gotmp').mkdir(exist_ok=True)
        return env

    def cmd(self,label,argv,cwd=ROOT,timeout=300):
        env=self.env();started=dt.datetime.now(dt.timezone.utc).isoformat()
        try:
            result=subprocess.run(list(map(str,argv)),cwd=cwd,env=env,capture_output=True,timeout=timeout,
                                  creationflags=subprocess.CREATE_NO_WINDOW if os.name=='nt' else 0)
            code,out,err=result.returncode,result.stdout,result.stderr
        except subprocess.TimeoutExpired as exc:
            code,out,err=124,exc.stdout or b'',exc.stderr or b''
        streams={}
        for name,data in [('stdout',out),('stderr',err)]:
            path=self.out/(label+'.'+name+'.txt');path.write_bytes(data)
            streams[name]={'path':path.name,'sha256':sha(path)}
        record=dict(label=label,argv=list(map(str,argv)),cwd=str(cwd),environment_names=sorted(env),
                    fixed_environment={k:env[k] for k in ('GOTOOLCHAIN','GOENV','GOPROXY','GOSUMDB','GOFLAGS','GOMODCACHE','GOCACHE')},
                    exit_code=code,started_at=started,ended_at=dt.datetime.now(dt.timezone.utc).isoformat(),**streams)
        save(self.out/(label+'.command.json'),record);self.commands.append(record)
        return code,out.decode('utf-8','replace'),err.decode('utf-8','replace')

    def finish(self):
        # Cleanup only this invocation's exact scratch. Windows native cmdlet;
        # reject junction/escape targets before a recursive operation.
        if self.scratch.exists():
            resolved=self.scratch.resolve()
            if resolved.parent != self.out.resolve() or resolved.name!='scratch':raise ValueError('CLEANUP_BOUNDARY')
            ps=Path(os.environ['SystemRoot'])/'System32/WindowsPowerShell/v1.0/powershell.exe'
            env={'SystemRoot':os.environ['SystemRoot'],'W0299_SCRATCH':str(resolved)}
            cleanup=subprocess.run([str(ps),'-NoProfile','-Command',"Remove-Item -LiteralPath $env:W0299_SCRATCH -Recurse -Force -ErrorAction Stop"],env=env,capture_output=True)
            save(self.out/'cleanup.json',{'target':str(resolved),'exit_code':cleanup.returncode,'absent':not resolved.exists()})
            self.check('OWNED_SCRATCH_CLEANUP',cleanup.returncode==0 and not resolved.exists(),str(resolved))
        ids=[r['id'] for r in self.rows]
        passed=bool(ids) and len(ids)==len(set(ids)) and all(r['ok'] is True for r in self.rows)
        only_binding=all(r['ok'] or r['id']==self.selector+'-EXACT_PARENT_BINDING' for r in self.rows)
        status='PASS' if passed else 'PENDING_INTEGRATION' if self.pending and only_binding else 'FAIL'
        report={'schema':'researchos.w02-parent-gate.v1','check_id':self.selector,
                'mode':'FIXTURE' if self.selector=='W02-G03' else 'DEVELOPMENT','status':status,
                'scope':'W02_CONSTRUCTION_ONLY','assertions':len(ids),'assertion_ids':ids,'checks':self.rows,
                'source_inputs_sha256':self.inputs,'commands':self.commands,'owned_acceptance_ids':[],
                'formal_cases_accepted':[],'whole_card_accepted':False,'independent_review':'PENDING',
                'new_provider_calls':0,'new_service_operations':0,'new_gui_operations':0}
        save(self.out/'receipt.json',report)
        save(self.out/'manifest.json',{p.relative_to(self.out).as_posix():sha(p) for p in sorted(self.out.rglob('*')) if p.is_file()})
        print(json.dumps({'status':status,'scope':report['scope'],'assertions':len(ids),'assertion_ids':ids,'receipt':str(self.out/'receipt.json')}))
        return 0 if passed else 1


def g01(g):
    stage=g.scratch/'source';stage.mkdir()
    # Materialize only committed, fixed source files into an empty directory.
    files=g.pin['build_sources']
    for rel,expected in files.items():
        source=g.source(rel)
        if sha(source)!=expected:raise ValueError('BUILD_INPUT_DRIFT:'+rel)
        target=stage/rel;target.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(source,target)
    save(g.out/'clean-source-manifest.json',{rel:sha(stage/rel) for rel in files})
    g.check('CLEAN_SOURCE',all(sha(stage/p)==h for p,h in files.items()),{'files':len(files),'source':'tracked clean source vector; no binaries/cache copied'})
    go=g.tool('go')
    if not reuse_g01(g,'FIXED_GO_DEPENDENCIES'):
        code,out,_=g.cmd('verify-go-modules',[go,'mod','verify'],stage)
        g.check('FIXED_GO_DEPENDENCIES',code==0 and 'all modules verified' in out,'offline pinned module cache verified by go.mod/go.sum')
    artifacts={}
    for name,pkg in [('control','./apps/control'),('researchos','./apps/cli')]:
        exe=g.scratch/(name+'.exe')
        code,_,_=g.cmd('build-'+name,[go,'build','-trimpath','-o',exe,pkg],stage)
        g.check('BUILD_'+name.upper(),code==0 and exe.is_file(),{'path':str(exe),'sha256':sha(exe) if exe.exists() else None})
        if not exe.exists():continue
        artifacts[name]={'build_path':str(exe),'sha256':sha(exe),'retention':'hash/raw retained; temporary executable cleaned'}
        code,out,_=g.cmd('buildinfo-'+name,[go,'version','-m',exe],stage)
        g.check('BUILDINFO_'+name.upper(),code==0 and 'go1.27.1' in out,'actual built binary metadata')
    exe=g.scratch/'researchos.exe'
    if exe.exists():
        code,out,_=g.cmd('cli-version',[exe,'version'],stage)
        value=json.loads(out)
        g.check('CLI_PROTOCOL',code==0 and value.get('protocol')=='researchos-client/2' and value.get('status')=='COMPLETED',value)
        code,out,err=g.cmd('cli-unimplemented',[exe,'w0299-unimplemented-command'],stage)
        g.check('CLI_UNIMPLEMENTED_NONZERO',code==3 and 'NOT_IMPLEMENTED' in err,{'exit_code':code,'stdout':out})
    if not reuse_g01(g,'MINIMUM_TESTS'):
        code,out,_=g.cmd('minimum-go-tests',[go,'test','-json','-count=1','./internal/canonical','./internal/store'],stage)
        events=[json.loads(line) for line in out.splitlines() if line.strip()]

        g.check('MINIMUM_TESTS',code==0 and {'TestJSON','TestUTF16AndFieldOrder','TestOpenSettings'}<=tests and not any(r.get('Action') in ('skip','fail') for r in events),sorted(tests))
    validator=g.spec/'tools/validate_package.py'
    if sha(validator)!=g.pin['spec_files']['tools/validate_package.py']:raise ValueError('SPEC_VALIDATOR_DRIFT')
    g.inputs[str(validator)]=sha(validator)
    if not reuse_g01(g,'V24_SCHEMA_API_STATIC'):
        code,out,_=g.cmd('v24-schema-api-static',[g.tool('python'),'-B',validator],g.spec)
        report=json.loads(out)
        g.check('V24_SCHEMA_API_STATIC',code==0 and report.get('status')=='PASS' and bool(report.get('checks')),
                {'scope':'SPECIFICATION_PACKAGE_ONLY','checks':report.get('checks'),'errors':report.get('errors')})
    proto='internal/api/contracts/client.protocol.schema.json'
    g.check('RUNTIME_PROTOCOL_FROZEN',sha(g.source(proto))==g.pin['spec_files']['contracts/client.protocol.schema.json'],'runtime embedded protocol equals frozen v2.4 schema')
    sample=ROOT/g.pin['sample']
    reused=[]
    for rel,expected in g.pin['renderer_inputs'].items():
        g.source(rel)
        if sha(ROOT/rel)!=expected:raise ValueError('RENDERER_REUSE_DRIFT:'+rel)
    for name,expected in g.pin['renderer_evidence'].items():
        path=sample/name
        if sha(path)!=expected:raise ValueError('RENDERER_RAW_DRIFT:'+name)
        g.inputs[str(path)]=expected;reused.append({'path':str(path),'sha256':expected})
    g.check('CURRENT_RENDERER_BUILD_REUSE',read(sample/'build-renderer.command.json')['exit_code']==0,
            {'unchanged_inputs':len(g.pin['renderer_inputs']),'actual_prior_build':reused,'new_gui_run':False})
    for name in ('main/main.js','preload/preload.js'):
        rel='apps/desktop/'+name
        if (ROOT/rel).exists():
            if not reuse_g01(g,'DESKTOP_SYNTAX_'+Path(name).stem.upper()):
                code,_,_=g.cmd('desktop-syntax-'+Path(name).stem,[g.tool('node'),'--check',g.source(rel)])
                g.check('DESKTOP_SYNTAX_'+Path(name).stem.upper(),code==0,rel)
    route=read(g.source('execution/v2.4/route.json'))
    component=read(g.source('execution/v2.4/component_lock.json'))
    manifest=read(g.source('third_party/mu.MANIFEST.json'))
    entries=manifest['entries'];bad=[rel for rel,item in entries.items() if not (ROOT/rel).is_file() or sha(ROOT/rel)!=item['sha256']]
    g.check('SOLE_ROUTE_FIXED_MU',route['selected_route']=='MU_FORK' and component['mu']['commit']==route['mu']['commit'] and len(entries)==component['mu']['file_count'] and not bad,
            {'route':'MU_FORK','business_authority':route['business_authority'],'vendor_files':len(entries),'changed':bad})
    g.check('INPUTS_UNCHANGED',all(sha(ROOT/p)==h for p,h in files.items()),'all clean build input hashes rechecked after commands')
    save(g.out/'build-identities.json',{'qualification_build_id':hashlib.sha256(json.dumps(files,sort_keys=True).encode()).hexdigest(),
          'protocol_version':'researchos-client/2','artifacts':artifacts,
          'desktop_entry':'apps/desktop/main/main.js','desktop_start_argv':[g.pin['tools']['electron']['path'],'apps/desktop/main/main.js'],

def reuse_g01(g,key):
    if not g.repair:return False
    prior=HERE/g.repair['prior_run']
    if sha(prior/'receipt.json')!=g.repair['prior_receipt_sha256']:raise ValueError('PRIOR_RECEIPT_DRIFT')
    manifest=read(prior/'manifest.json')
    if any(sha(prior/p)!=h for p,h in manifest.items()):raise ValueError('PRIOR_RAW_DRIFT')
    receipt=read(prior/'receipt.json')
    for name,h in receipt['source_inputs_sha256'].items():
        p=Path(name)
        if p.name in ('parent_driver.py','dispatcher_fixture.py','dispatch_parent.py'):continue
        if sha(p)!=h:raise ValueError('REUSED_INPUT_DRIFT:'+str(p))
        g.inputs[name]=h
    found=next((r for r in receipt['checks'] if r['id']=='W02-G01-'+key and r['ok'] is True),None)
    if not found:return False
    g.check(key,True,{'reused_passed_current_subcheck':str(prior/'receipt.json'),'receipt_sha256':sha(prior/'receipt.json'),
                    'raw_manifest_sha256':sha(prior/'manifest.json'),'original_detail':found['detail'],
                    'reason':'Only source-materialization inventory and Mu manifest reader changed; inputs to this passed check unchanged'})
    return True

    binding=entry.get('argv')==['-B',rel] and all(x in entry.get('selectors',{}).get('check_id',[]) for x in SELECTORS)

    g.check('EXACT_PARENT_BINDING',binding,{'expected_argv':['-B',rel],'actual_argv':entry.get('argv'),'state':'BOUND' if binding else 'PENDING_INTEGRATION'})
    owner_path=g.main/'execution/v2.4/path_ownership.json'
    owner=read(owner_path);g.inputs[str(owner_path)]=sha(owner_path)

        for annotated in item['paths']:

            if relpath in seen and seen[relpath]!=item['owner']:errors.append('duplicate_owner:'+relpath)
            seen[relpath]=item['owner']
            exists=(ROOT/relpath).exists() if not glob else None
            if not glob and not exists:errors.append('missing_alias:'+relpath)
            aliases.append({'alias':alias,'path':relpath,'owner':item['owner'],'exists':exists,'kind':'future_scoped_glob' if glob else 'actual'})
    save(g.out/'path-inventory.json',aliases)
    g.check('PATH_OWNER_UNIQUENESS',not errors,{'errors':errors,'rule':'exact registered paths unique and resolve as-is; parent scopes may contain child aliases; future scoped globs are not claimed as existing directories'})
    legacy=read(g.source('execution/path_ownership.json'))
    special=owner.get('special_paths',legacy['special_paths']);restricted=special['verifier_restricted_store']
    mapped=(ROOT/special['evaluation_adapter']['path']).is_dir() and (ROOT/special['design_tokens']['path']).is_file() and (ROOT/'evaluation/development').is_dir()

    mapped=mapped and bool(development.get('owner')) and (ROOT/development.get('path','__unmapped__')).is_dir()
    g.check('EVALUATION_UI_RESTRICTED_MAPPING',mapped and restricted.get('access_class')=='VERIFIER_ONLY' and restricted.get('status')=='UNBOUND' and restricted.get('path') is None,

    mapping=read(g.source('execution/v2.4/client_mapping.json'))

def main():
    selector=sys.argv[1] if len(sys.argv)==2 else ''
    if selector not in SELECTORS:
        print(json.dumps({'status':'NOT_IMPLEMENTED','assertions':0,'assertion_ids':[]}));return 3
    if selector in ('W02-G01','W02-G03'):
        from evidence_reuse import run
        return run(selector)
    g=Gate(selector)
    try:
        if selector=='W02-G01':g01(g)
        elif selector=='W02-G02':g02(g)
        else:
            from dispatcher_fixture import run
            run(g)
    except Exception as exc:
        # No arbitrary exception text: input paths and synthetic fixture errors
        # are sufficient for diagnosis without persisting ambient credentials.
        g.check('DRIVER_ERROR',False,{'type':type(exc).__name__,'reason':str(exc) if isinstance(exc,(ValueError,KeyError)) else 'See command raw; driver exception'})
    return g.finish()


if __name__=='__main__':
    sys.exit(main())
