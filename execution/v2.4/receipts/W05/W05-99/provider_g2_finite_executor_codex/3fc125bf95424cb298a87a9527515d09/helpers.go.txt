
// G2 controls are fixture evidence, never provider qualification.
func providerG2BeforeTaskBudget(t *testing.T) func(*fixture, Object) Object {
 return func(f *fixture, goal Object) Object {
  t.Helper()
  token := f.token
  defer func() { f.token = token }()
  id := Str(goal,"id")
  if goal["status"] != "ACTIVE" { t.Fatal("budget callback before Goal activation") }
  var tasks, runs int
  if e:=f.db.QueryRow(`SELECT count(*) FROM domain_objects WHERE project=? AND kind='Task' AND json_extract(body,'$.goal_id')=?`,f.project,id).Scan(&tasks); e!=nil {t.Fatal(e)}
  if e:=f.db.QueryRow(`SELECT count(*) FROM domain_objects WHERE project=? AND kind='Run'`,f.project).Scan(&runs); e!=nil {t.Fatal(e)}
  if tasks!=0 || runs!=0 {t.Fatal("budget callback after Task/Run creation",tasks,runs)}
  policyPath:=f.path("/goals/"+id+"/budget-policy")
  policy:=f.expect(200,"GET",policyPath,"","",nil)
  policy=f.expect(200,"PUT",policyPath,ID(),testTag(policy),capBody("10"))
  fresh:=f.read("goals",id)
  if fresh["id"]!=goal["id"] || fresh["status"]!="ACTIVE" {t.Fatal("budget changed original Goal identity")}
  providerEvidence(t,"before-task-budget.json",Object{"goal_before":goal,"goal_after":fresh,"policy":policy,"task_count":tasks,"run_count":runs,"token_restored_on_return":true})
  return fresh
 }
}

func providerG2TypedFaultMatches(err error, status int, code,message string) bool {
 var fault *Fault
 return errors.As(err,&fault) && fault.Status==status && fault.Code==code && fault.Message==message
}
func TestProviderBindingG2TypedFaultMatcher(t *testing.T) {
 for _,mode:=range []string{"exact","nil","wrong-status","wrong-code","wrong-message"} {t.Run(mode,func(t *testing.T){
  var err error=Fail(422,"GATE_FAILED","budget exhausted")
  switch mode {case "nil":err=nil;case "wrong-status":err=Fail(409,"GATE_FAILED","budget exhausted");case "wrong-code":err=Fail(422,"STATE_CONFLICT","budget exhausted");case "wrong-message":err=Fail(422,"GATE_FAILED","context stale")}
  got:=providerG2TypedFaultMatches(err,422,"GATE_FAILED","budget exhausted")
  providerEvidence(t,"component.json",Object{"mode":"SYNTHETIC_ASSERTION_CONTROL","case":mode,"actual":got,"expected":mode=="exact"})
  if got!=(mode=="exact") {t.Fatal("typed fault matcher",mode,got)}
 })}
}

type providerG2Trace struct {mu sync.Mutex; facts []Object}
func providerG2Observe(c *RuntimeCoordinator) *providerG2Trace {
 trace:=&providerG2Trace{facts:[]Object{}}
 previous:=c.boundaryHook
 c.boundaryHook=func(name string,fact Object){
  if previous!=nil {previous(name,fact)}
  copy,_:=Decode(Bytes(fact));trace.mu.Lock();defer trace.mu.Unlock()
  trace.facts=append(trace.facts,Object{"stage":name,"fact":copy})
 }
 return trace
}

func providerG2OriginalModelCauseMatches(actual,expected Object) bool {
 for _,key:=range []string{"project_id","task_id","run_id","epoch","dispatch_key","manifest_hash","profile_hash","transport_hash","model_key"} {
  if actual[key]==nil || expected[key]==nil || Hash(actual[key])!=Hash(expected[key]) {return false}
 }
 effects:=Arr(actual,"effects");ledger:=Arr(actual,"ledger")
 if len(effects)!=3 || len(ledger)!=1 {return false}
 var requestHash,reservation any
 for i,state:=range []string{"PREPARED","STARTED","UNKNOWN"} {
  e,ok:=effects[i].(map[string]any);if !ok || e["state"]!=state || e["project_id"]!=expected["project_id"] || e["task_id"]!=expected["task_id"] || e["run_id"]!=expected["run_id"] || Hash(e["epoch"])!=Hash(expected["epoch"]) || e["key"]!=expected["model_key"] {return false}
  if i==0 {requestHash,reservation=e["request_hash"],e["reservation_id"];if requestHash==nil || reservation==nil{return false}}
  if e["request_hash"]!=requestHash || e["reservation_id"]!=reservation{return false}
  if i==2 {r,ok:=e["receipt"].(map[string]any);if !ok || r["reason"]!=expected["unknown_reason"] || r["actual_cost"]!=nil{return false}}
 }
 l,ok:=ledger[0].(map[string]any)
 return ok && l["id"]==reservation && l["run_id"]==expected["run_id"] && l["provider_request_id"]==expected["model_key"] && l["provider_event_id"]=="prepare/"+Str(expected,"model_key") && l["operation"]=="RESERVE" && l["cost_basis"]=="UPPER_BOUND" && l["reservation_id"]==nil
}

func providerG2Rows(t *testing.T, db *sql.DB) Object {
 t.Helper()
 rows,e:=db.Query(`SELECT name,type,sql FROM sqlite_master WHERE name NOT LIKE 'sqlite_%' ORDER BY type,name`);if e!=nil{t.Fatal(e)}
 schema:=Empty();tables:=[]string{}
 for rows.Next(){var name,kind string;var ddl sql.NullString;if e=rows.Scan(&name,&kind,&ddl);e!=nil{t.Fatal(e)};schema=append(schema,Object{"name":name,"type":kind,"sql":ddl.String});if kind=="table"{tables=append(tables,name)}}
 if e=rows.Err();e!=nil{t.Fatal(e)};rows.Close()
 all:=Object{"schema":schema}
 for _,table:=range tables {
  rows,e=db.Query(`SELECT * FROM "`+strings.ReplaceAll(table,`"`,`""`)+`"`);if e!=nil{t.Fatal(e)}
  cols,e:=rows.Columns();if e!=nil{t.Fatal(e)};values:=[]string{}
  for rows.Next(){v:=make([]any,len(cols));ptr:=make([]any,len(cols));for i:=range v{ptr[i]=&v[i]};if e=rows.Scan(ptr...);e!=nil{t.Fatal(e)};values=append(values,string(Bytes(v)))}
  if e=rows.Err();e!=nil{t.Fatal(e)};rows.Close();sort.Strings(values);all[table]=Object{"columns":cols,"row_multiset":values}
 }
 return all
}
