func providerG2QuarantinePayload(mode string) string {
	payload := ""
	switch mode {
	case "selected-alias":
		payload = "data: {\"choices\":[{\"index\":0,\"delta\":{\"reasoning_content\":\"host-only-model-\"}}]}\n\ndata: {\"choices\":[{\"index\":0,\"delta\":{\"reasoning\":\"\\u0066ixture\",\"content\":\"safe\"},\"finish_reason\":\"stop\"}]}\n\ndata: [DONE]\n\n"
	case "escaped-tool":
		payload = "data: {\"choices\":[{\"index\":0,\"delta\":{\"tool_calls\":[{\"index\":0,\"id\":\"fixture-call\",\"type\":\"function\",\"function\":{\"name\":\"read\",\"arguments\":\"{\\\"path\\\":\\\"host-only-model-\\\\u0066ixture\\\"}\"}}]},\"finish_reason\":\"tool_calls\"}]}\n\ndata: [DONE]\n\n"
	case "unsupported-detail":
		payload = "data: {\"choices\":[{\"index\":0,\"delta\":{\"reasoning_details\":[{\"type\":\"reasoning.text\",\"text\":\"safe but unsupported\"}],\"content\":\"safe\"},\"finish_reason\":\"stop\"}]}\n\ndata: [DONE]\n\n"
	case "early-eof":
		payload = "data: {\"choices\":[{\"index\":0,\"delta\":{\"reasoning\":\"partial\"}}]}\n\n"
	}
	return payload
}
