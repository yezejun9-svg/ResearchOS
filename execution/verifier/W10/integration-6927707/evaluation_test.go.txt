package evaluation
import("testing";"fmt";"os";"encoding/json";"path/filepath")
func independentCheck(t *testing.T,id string,want,got any){t.Helper();status:="PASS";if string(Bytes(want))!=string(Bytes(got)){status="FAIL"};fmt.Println("INDEPENDENT_W10 "+string(Bytes(Object{"assertion_id":id,"expected":want,"actual":got,"status":status})));if status!="PASS"{t.Errorf("%s want %v got %v",id,want,got)}}
func TestIndependentProviderStatePrecedence(t *testing.T){for _,mode:=range []string{"provider_error","abstain"}{t.Run(mode,func(t *testing.T){out:=Object{"selected":"ADOPT","score":0.99};want:="ERROR";if mode=="provider_error"{out[mode]="timeout after partial output"}else{out[mode]=true;want="ABSTAIN"};data:=Bytes(Object{"cases":[]any{Object{"id":"one","input":Object{"evidence_present":true,"evidence_current":true,"hard_checks_passed":true,"measured_net_gain_positive":true},"simulated_provider_output":out,"expected_decision":"ADOPT"}}});rows,e:=Execute(data,[]byte(`{"mode":"PROTOCOL_FIXTURE"}`));if e!=nil{t.Fatal(e)};h:=Digest(data);m:=Manifest{ProjectID:"p",ExperimentID:"e",RunID:"r",Engine:Object{"name":"NATIVE","version":"1","config_hash":h},DatasetHash:h,CandidateHash:h,BaselineHash:h,RubricHash:h,Expected:[]string{"one"}};zero:=0;raw:=Bytes(rows);report,e:=Normalize(m,m,raw,Digest(raw),&zero,1);if e!=nil{t.Fatal(e)};independentCheck(t,mode+"_survives_partial_matching_selection",want,report["results"].([]Result)[0].Verdict);os.WriteFile(filepath.Join(os.Getenv("W10_REVIEW_OUT"),"partial-"+mode+".json"),Bytes(Object{"dataset":json.RawMessage(data),"raw_export":string(raw),"raw_hash":Digest(raw),"report":report}),0600)})}}
func TestIndependentProviderControls(t *testing.T){for _,mode:=range []string{"error_duplicate","abstain_duplicate","pass","fail"}{t.Run(mode,func(t *testing.T){out:=Object{"selected":"ADOPT"};want:="PASS";dup:=0;if mode=="error_duplicate"{out["provider_error"]="timeout";out["duplicate_case_id"]="one";want="ERROR";dup=1};if mode=="abstain_duplicate"{out["abstain"]=true;out["duplicate_case_id"]="one";want="ABSTAIN";dup=1};if mode=="fail"{out["selected"]="REJECT";want="FAIL"};data:=Bytes(Dataset{Cases:[]Case{{ID:"one",Input:Object{"evidence_present":true,"evidence_current":true,"hard_checks_passed":true},Output:out,Expected:"ADOPT"}}});rows,e:=Execute(data,Bytes(Program{Mode:"PROTOCOL_FIXTURE"}));if e!=nil{t.Fatal(e)};independentCheck(t,mode+"_raw_count",1+dup,len(rows));for _,row:=range rows{independentCheck(t,mode+"_raw_status",want,row.Verdict)};h:=Digest(data);m:=Manifest{ProjectID:"p",ExperimentID:"e",RunID:"r",Engine:Object{"name":"NATIVE","version":"1","config_hash":h},DatasetHash:h,CandidateHash:h,BaselineHash:h,RubricHash:h,Expected:[]string{"one"}};zero:=0;raw:=Bytes(rows);report,e:=Normalize(m,m,raw,Digest(raw),&zero,1);if e!=nil{t.Fatal(e)};key:=map[string]string{"PASS":"pass_count","FAIL":"fail_count","ERROR":"error_count","ABSTAIN":"abstain_count"}[want];independentCheck(t,mode+"_normalized_class_count",[]any{want,1,dup},[]any{report["results"].([]Result)[0].Verdict,report["counts"].(Object)[key],report["counts"].(Object)["duplicate_count"]})})}}
