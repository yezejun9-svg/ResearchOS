package main

import (
 "encoding/json"
 "io"
 "net/http"
 "net/http/httptest"
 "os"
 "time"
)

func main() {
 rows:=[]map[string]any{}
 for i:=0;i<30;i++ {
  s:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){w.WriteHeader(204)}))
  c:=s.Client();c.Timeout=3*time.Second
  for attempt:=0;attempt<2;attempt++ {
   start:=time.Now();r,e:=c.Get(s.URL)
   row:=map[string]any{"server":i,"attempt":attempt,"address":s.URL,"duration_ms":time.Since(start).Milliseconds(),"error":nil}
   if e!=nil {row["error"]=e.Error()} else {io.Copy(io.Discard,r.Body);r.Body.Close();row["status"]=r.StatusCode}
   rows=append(rows,row)
  }
  s.Close()
 }
 json.NewEncoder(os.Stdout).Encode(rows)
}
