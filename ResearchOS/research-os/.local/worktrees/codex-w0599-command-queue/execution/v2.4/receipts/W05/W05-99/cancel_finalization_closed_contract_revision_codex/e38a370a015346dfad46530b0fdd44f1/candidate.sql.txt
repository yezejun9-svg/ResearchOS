CREATE TABLE f08_legacy_runs(
 project TEXT NOT NULL,run_id TEXT NOT NULL,
 inventory TEXT NOT NULL CHECK(typeof(inventory)='text' AND json_valid(inventory)),
 inventory_hash TEXT NOT NULL CHECK(typeof(inventory_hash)='text'),
 PRIMARY KEY(project,run_id)
);
-- Populate ONLY inside initial exact migration, before forbidding new inserts.
INSERT INTO f08_legacy_runs(project,run_id,inventory,inventory_hash)
SELECT project,run_id,inventory,researchos_f08_proof_v1(-1,CAST(inventory AS BLOB))
FROM (
 SELECT r.project,r.run_id,json_object(
 'format','researchos.cancel-finalization/1','source_schema_version',23,
 'project',r.project,'run_id',r.run_id,
 'versions',json((SELECT json_group_array(json_array(revision,hex(CAST(body AS BLOB)),content_hash))
   FROM (SELECT revision,body,content_hash FROM domain_versions
    WHERE project=r.project AND kind='Run' AND id=r.run_id ORDER BY revision))),
 'dispatches',json((SELECT json_group_array(json_array(epoch,dispatch_key,hex(CAST(manifest AS BLOB)),manifest_hash,state,worker,journal_locator,hex(CAST(identity AS BLOB)),observation_sequence,observation_hash,reason,prepared_at,updated_at))
   FROM (SELECT * FROM runtime_dispatches WHERE project=r.project AND run=r.run_id ORDER BY epoch)))
 ) inventory FROM (
 SELECT project,id run_id FROM domain_versions WHERE kind='Run'
 UNION SELECT project,id FROM domain_objects WHERE kind='Run'
 UNION SELECT project,run FROM runtime_dispatches
 ) r
);
CREATE TRIGGER f08_legacy_no_insert BEFORE INSERT ON f08_legacy_runs BEGIN SELECT RAISE(ABORT,'F08_LEGACY_COHORT_FROZEN'); END;
CREATE TRIGGER f08_legacy_no_update BEFORE UPDATE ON f08_legacy_runs BEGIN SELECT RAISE(ABORT,'F08_LEGACY_COHORT_FROZEN'); END;
CREATE TRIGGER f08_legacy_no_delete BEFORE DELETE ON f08_legacy_runs BEGIN SELECT RAISE(ABORT,'F08_LEGACY_COHORT_FROZEN'); END;

CREATE TABLE f08_dispatch_origins(
 project TEXT NOT NULL,run_id TEXT NOT NULL,epoch INTEGER NOT NULL CHECK(typeof(epoch)='integer' AND epoch>0),
 dispatch_key TEXT NOT NULL UNIQUE,
 document TEXT NOT NULL CHECK(typeof(document)='text' AND json_valid(document)),
 body_hash TEXT NOT NULL CHECK(typeof(body_hash)='text'),
 PRIMARY KEY(project,run_id,epoch),
 UNIQUE(project,run_id,epoch,body_hash),
 FOREIGN KEY(project,run_id,epoch) REFERENCES runtime_dispatches(project,run,epoch) DEFERRABLE INITIALLY DEFERRED
);
CREATE TRIGGER f08_origin_no_replace BEFORE INSERT ON f08_dispatch_origins
WHEN EXISTS(SELECT 1 FROM f08_dispatch_origins o WHERE (o.project=NEW.project AND o.run_id=NEW.run_id AND o.epoch=NEW.epoch) OR o.dispatch_key=NEW.dispatch_key)
BEGIN SELECT RAISE(ABORT,'F08_ORIGIN_IMMUTABLE'); END;
CREATE TRIGGER f08_origin_legacy_reject BEFORE INSERT ON f08_dispatch_origins
WHEN EXISTS(SELECT 1 FROM f08_legacy_runs l WHERE l.project=NEW.project AND l.run_id=NEW.run_id)
BEGIN SELECT RAISE(ABORT,'F08_LEGACY_ORIGIN_FORBIDDEN'); END;
CREATE TRIGGER f08_origin_no_update BEFORE UPDATE ON f08_dispatch_origins BEGIN SELECT RAISE(ABORT,'F08_ORIGIN_IMMUTABLE'); END;
CREATE TRIGGER f08_origin_no_delete BEFORE DELETE ON f08_dispatch_origins BEGIN SELECT RAISE(ABORT,'F08_ORIGIN_IMMUTABLE'); END;

CREATE TABLE f08_admitted_groups(
 principal TEXT NOT NULL,project TEXT NOT NULL,command_id TEXT NOT NULL,request_hash TEXT NOT NULL,
 operation TEXT NOT NULL CHECK(operation='v2:session.control'),
 private_header TEXT NOT NULL CHECK(typeof(private_header)='text' AND length(CAST(private_header AS BLOB))<=1398200),
 root_hash TEXT NOT NULL,group_hash TEXT NOT NULL,
 document TEXT NOT NULL CHECK(typeof(document)='text' AND json_valid(document)),
 PRIMARY KEY(principal,project,command_id,request_hash),
 UNIQUE(principal,project,command_id),
 UNIQUE(principal,project,command_id,request_hash,root_hash),
 FOREIGN KEY(principal,project,operation,command_id,request_hash)
  REFERENCES idempotency(principal,project,operation,key,request_hash) DEFERRABLE INITIALLY DEFERRED,
 FOREIGN KEY(principal,command_id,request_hash)
  REFERENCES client_command_identities(principal,command_id,request_hash) DEFERRABLE INITIALLY DEFERRED
);
CREATE TABLE f08_admitted_members(
 principal TEXT NOT NULL,project TEXT NOT NULL,command_id TEXT NOT NULL,request_hash TEXT NOT NULL,
 task_id TEXT NOT NULL,member_index INTEGER NOT NULL CHECK(typeof(member_index)='integer' AND member_index>=0),
 root_hash TEXT NOT NULL,body_hash TEXT NOT NULL,
 document TEXT NOT NULL CHECK(typeof(document)='text' AND json_valid(document)),
 PRIMARY KEY(principal,project,command_id,request_hash,task_id),
 UNIQUE(principal,project,command_id,request_hash,member_index),
 UNIQUE(principal,project,command_id,request_hash,task_id,body_hash),
 FOREIGN KEY(principal,project,command_id,request_hash,root_hash)
  REFERENCES f08_admitted_groups(principal,project,command_id,request_hash,root_hash) DEFERRABLE INITIALLY DEFERRED
);
CREATE TABLE f08_final_groups(
 principal TEXT NOT NULL,project TEXT NOT NULL,command_id TEXT NOT NULL,request_hash TEXT NOT NULL,
 admitted_root_hash TEXT NOT NULL,root_hash TEXT NOT NULL,group_hash TEXT NOT NULL,
 document TEXT NOT NULL CHECK(typeof(document)='text' AND json_valid(document)),
 PRIMARY KEY(principal,project,command_id,request_hash),
 UNIQUE(principal,project,command_id,request_hash,root_hash),
 FOREIGN KEY(principal,project,command_id,request_hash,admitted_root_hash)
  REFERENCES f08_admitted_groups(principal,project,command_id,request_hash,root_hash) DEFERRABLE INITIALLY DEFERRED,
 FOREIGN KEY(principal,project,command_id,request_hash,root_hash)
  REFERENCES f08_final_commits(principal,project,command_id,request_hash,final_root_hash) DEFERRABLE INITIALLY DEFERRED
);
CREATE TABLE f08_final_members(
 principal TEXT NOT NULL,project TEXT NOT NULL,command_id TEXT NOT NULL,request_hash TEXT NOT NULL,
 task_id TEXT NOT NULL,member_index INTEGER NOT NULL CHECK(typeof(member_index)='integer' AND member_index>=0),
 admitted_member_hash TEXT NOT NULL,body_hash TEXT NOT NULL,
 document TEXT NOT NULL CHECK(typeof(document)='text' AND json_valid(document)),
 PRIMARY KEY(principal,project,command_id,request_hash,task_id),
 UNIQUE(principal,project,command_id,request_hash,member_index),
 FOREIGN KEY(principal,project,command_id,request_hash)
  REFERENCES f08_final_groups(principal,project,command_id,request_hash) DEFERRABLE INITIALLY DEFERRED,
 FOREIGN KEY(principal,project,command_id,request_hash,task_id,admitted_member_hash)
  REFERENCES f08_admitted_members(principal,project,command_id,request_hash,task_id,body_hash) DEFERRABLE INITIALLY DEFERRED
);
CREATE TABLE f08_final_commits(
 principal TEXT NOT NULL,project TEXT NOT NULL,command_id TEXT NOT NULL,request_hash TEXT NOT NULL,
 final_root_hash TEXT NOT NULL,original_response_digest TEXT NOT NULL,final_response_digest TEXT NOT NULL,
 PRIMARY KEY(principal,project,command_id,request_hash),
 UNIQUE(principal,project,command_id,request_hash,final_root_hash),
 FOREIGN KEY(principal,project,command_id,request_hash,final_root_hash)
  REFERENCES f08_final_groups(principal,project,command_id,request_hash,root_hash) DEFERRABLE INITIALLY DEFERRED,
 FOREIGN KEY(principal,command_id,request_hash)
  REFERENCES client_command_identities(principal,command_id,request_hash) DEFERRABLE INITIALLY DEFERRED
);
CREATE TABLE f08_version_pins(
 principal TEXT NOT NULL,project TEXT NOT NULL,command_id TEXT NOT NULL,request_hash TEXT NOT NULL,
 phase TEXT NOT NULL CHECK(phase IN ('ADMITTED','FINALIZED')),
 target_project TEXT NOT NULL,kind TEXT NOT NULL,id TEXT NOT NULL,revision INTEGER NOT NULL CHECK(typeof(revision)='integer' AND revision>0),
 content_hash TEXT,
 PRIMARY KEY(principal,project,command_id,request_hash,phase,target_project,kind,id,revision),
 FOREIGN KEY(target_project,kind,id,revision)
  REFERENCES domain_versions(project,kind,id,revision) DEFERRABLE INITIALLY DEFERRED
);
CREATE TABLE f08_observation_obligations(
 project TEXT NOT NULL,run_id TEXT NOT NULL,epoch INTEGER NOT NULL,sequence INTEGER NOT NULL,
 receipt_hash TEXT NOT NULL,
 PRIMARY KEY(project,run_id,epoch,sequence),
 FOREIGN KEY(project,run_id,epoch,receipt_hash)
  REFERENCES runtime_observations(project,run,epoch,receipt_hash) DEFERRABLE INITIALLY DEFERRED
);

-- Group materialization is within the root INSERT statement. There is no
-- application-visible partially populated group and no 'ready' column.
CREATE TRIGGER f08_admitted_materialize AFTER INSERT ON f08_admitted_groups BEGIN
 INSERT INTO f08_admitted_members(principal,project,command_id,request_hash,task_id,member_index,root_hash,body_hash,document)
 SELECT NEW.principal,NEW.project,NEW.command_id,NEW.request_hash,
  json_extract(value,'$.task_ref.id'),CAST(key AS INTEGER),NEW.root_hash,
  researchos_f08_proof_v1(-1,CAST(value AS BLOB)),value
 FROM json_each(NEW.document,'$.members');
 INSERT INTO f08_version_pins(principal,project,command_id,request_hash,phase,target_project,kind,id,revision,content_hash)
 SELECT NEW.principal,NEW.project,NEW.command_id,NEW.request_hash,'ADMITTED',
  json_extract(value,'$.project'),json_extract(value,'$.kind'),json_extract(value,'$.id'),json_extract(value,'$.revision'),json_extract(value,'$.content_hash')
 FROM json_each(NEW.document,'$.pins');
 SELECT CASE WHEN
 (SELECT count(*) FROM f08_admitted_members m WHERE m.principal=NEW.principal AND m.project=NEW.project AND m.command_id=NEW.command_id AND m.request_hash=NEW.request_hash)
 IS NOT json_array_length(NEW.document,'$.members') THEN RAISE(ABORT,'F08_PARTIAL_ADMISSION') END;
END;
CREATE TRIGGER f08_final_materialize AFTER INSERT ON f08_final_groups BEGIN
 INSERT INTO f08_final_members(principal,project,command_id,request_hash,task_id,member_index,admitted_member_hash,body_hash,document)
 SELECT NEW.principal,NEW.project,NEW.command_id,NEW.request_hash,
  json_extract(value,'$.task_admitted_ref.id'),CAST(key AS INTEGER),
  json_extract(value,'$.admitted_member_ref.content_hash'),
  researchos_f08_proof_v1(-1,CAST(value AS BLOB)),value
 FROM json_each(NEW.document,'$.members');
 INSERT INTO f08_version_pins(principal,project,command_id,request_hash,phase,target_project,kind,id,revision,content_hash)
 SELECT NEW.principal,NEW.project,NEW.command_id,NEW.request_hash,'FINALIZED',
  json_extract(value,'$.project'),json_extract(value,'$.kind'),json_extract(value,'$.id'),json_extract(value,'$.revision'),json_extract(value,'$.content_hash')
 FROM json_each(NEW.document,'$.pins');
 SELECT CASE WHEN EXISTS(
  SELECT task_id FROM f08_admitted_members WHERE principal=NEW.principal AND project=NEW.project AND command_id=NEW.command_id AND request_hash=NEW.request_hash
  EXCEPT SELECT task_id FROM f08_final_members WHERE principal=NEW.principal AND project=NEW.project AND command_id=NEW.command_id AND request_hash=NEW.request_hash
 ) OR EXISTS(
  SELECT task_id FROM f08_final_members WHERE principal=NEW.principal AND project=NEW.project AND command_id=NEW.command_id AND request_hash=NEW.request_hash
  EXCEPT SELECT task_id FROM f08_admitted_members WHERE principal=NEW.principal AND project=NEW.project AND command_id=NEW.command_id AND request_hash=NEW.request_hash
 ) THEN RAISE(ABORT,'F08_FINAL_MEMBER_SET') END;
END;

-- Complete-group raw byte/hash checks independent of any supplied matching hash.
CREATE TRIGGER f08_admission_hash BEFORE INSERT ON f08_admitted_groups BEGIN
 SELECT CASE WHEN researchos_f08_proof_v1(-1,CAST(json_extract(NEW.document,'$.root') AS BLOB)) IS NOT NEW.root_hash
 THEN RAISE(ABORT,'F08_ADMITTED_ROOT_HASH') END;
 SELECT CASE WHEN researchos_f08_proof_v1(-2,CAST(NEW.private_header AS BLOB)) IS NOT json_extract(NEW.document,'$.root.private_header_digest')
 THEN RAISE(ABORT,'F08_HEADER_HASH') END;
 SELECT CASE WHEN json_extract(NEW.document,'$.root.member_count') IS NOT json_array_length(NEW.document,'$.members')
 THEN RAISE(ABORT,'F08_ADMITTED_COUNT') END;
END;
CREATE TRIGGER f08_final_hash BEFORE INSERT ON f08_final_groups BEGIN
 SELECT CASE WHEN researchos_f08_proof_v1(-1,CAST(json_extract(NEW.document,'$.root') AS BLOB)) IS NOT NEW.root_hash
 THEN RAISE(ABORT,'F08_FINAL_ROOT_HASH') END;
 SELECT CASE WHEN json_extract(NEW.document,'$.root.member_count') IS NOT json_array_length(NEW.document,'$.members')
 THEN RAISE(ABORT,'F08_FINAL_COUNT') END;
END;

-- Native source cohort/monotonicity predicates not based on missing PID.
CREATE TRIGGER f08_dispatch_first BEFORE INSERT ON runtime_dispatches BEGIN
 SELECT CASE WHEN EXISTS(SELECT 1 FROM runtime_dispatches d WHERE (d.project=NEW.project AND d.run=NEW.run AND d.epoch=NEW.epoch) OR d.dispatch_key=NEW.dispatch_key)
 THEN RAISE(ABORT,'F08_DISPATCH_NO_REPLACE') END;
 SELECT CASE WHEN NEW.state IS NOT 'PREPARED' OR NEW.worker IS NOT '' OR NEW.journal_locator IS NOT '' OR NEW.identity IS NOT '{}' OR NEW.observation_sequence IS NOT 0 OR NEW.observation_hash IS NOT ''
 THEN RAISE(ABORT,'F08_DISPATCH_INITIAL_STATE') END;
 -- An old Run may still enter a genuinely authorized later epoch under the
 -- unchanged existing admission/lease/recovery gates. It never gains origin.
END;
CREATE TRIGGER f08_dispatch_no_rewind BEFORE UPDATE ON runtime_dispatches BEGIN
 SELECT CASE WHEN
 (OLD.state='PREPARED' AND NEW.state NOT IN ('STARTED','UNKNOWN')) OR
 (OLD.state='STARTED' AND NEW.state NOT IN ('STARTED','UNKNOWN','OBSERVED')) OR
 (OLD.state='UNKNOWN' AND NEW.state NOT IN ('UNKNOWN','OBSERVED')) OR
 (OLD.state='OBSERVED' AND NEW.state<>'OBSERVED')
 THEN RAISE(ABORT,'F08_DISPATCH_REWIND') END;
 SELECT CASE WHEN OLD.worker<>'' AND NEW.worker IS NOT OLD.worker
 OR OLD.journal_locator<>'' AND NEW.journal_locator IS NOT OLD.journal_locator
 OR OLD.identity<>'{}' AND CAST(NEW.identity AS BLOB) IS NOT CAST(OLD.identity AS BLOB)
 OR NEW.observation_sequence<OLD.observation_sequence
 OR (NEW.observation_sequence=OLD.observation_sequence AND NEW.observation_hash IS NOT OLD.observation_hash)
 THEN RAISE(ABORT,'F08_DISPATCH_IDENTITY_RESET') END;
 SELECT CASE WHEN NEW.state='STARTED' AND EXISTS(
 SELECT 1 FROM f08_final_members m WHERE m.project=NEW.project
 AND json_extract(m.document,'$.proof_kind')='NEVER_LAUNCHED_NATIVE'
 AND json_extract(m.document,'$.run_admitted_ref.id')=NEW.run
 AND json_extract(m.document,'$.dispatch_final_binding.epoch')=NEW.epoch)
 THEN RAISE(ABORT,'F08_CANCELLED_LAUNCH') END;
END;
-- This source-order obligation permits actual observeRuntime UPDATE before
-- INSERT observation; COMMIT still requires exact original observation tuple.
CREATE TRIGGER f08_dispatch_observation AFTER UPDATE ON runtime_dispatches
WHEN NEW.state='OBSERVED' AND (OLD.state<>'OBSERVED' OR NEW.observation_sequence>OLD.observation_sequence)
BEGIN
 INSERT INTO f08_observation_obligations(project,run_id,epoch,sequence,receipt_hash)
 VALUES(NEW.project,NEW.run,NEW.epoch,NEW.observation_sequence,NEW.observation_hash);
END;

-- Each source tuple is a fixed positional array. All stored JSON, response,
-- private-header and artifact bytes are hex, NEVER reparsed by SQLite first.
-- Full rows across projects are intentional: cross-project/global-principal
-- contradictions and reverse references must not disappear by selection.
CREATE VIEW f08_source_rows(category,row_json) AS
 SELECT 'domain_versions',json_array(project,kind,id,revision,hex(CAST(body AS BLOB)),content_hash) FROM domain_versions
 UNION ALL SELECT 'domain_objects',json_array(project,kind,id,revision,hex(CAST(body AS BLOB)),content_hash,immutable) FROM domain_objects
 UNION ALL SELECT 'domain_refs',json_array(project,kind,id,revision,field,target_project,target_kind,target_id,target_revision) FROM domain_refs
 UNION ALL SELECT 'outbox',json_array(project,sequence,id,hex(CAST(body AS BLOB)),acknowledged) FROM outbox
 UNION ALL SELECT 'idempotency',json_array(principal,project,operation,key,request_hash,status,hex(CAST(body AS BLOB)),hex(CAST(headers AS BLOB)),created_at,typeof(body),typeof(headers)) FROM idempotency
 UNION ALL SELECT 'identities',json_array(principal,command_id,protocol,project,method,operation,request_hash,created_at) FROM client_command_identities
 UNION ALL SELECT 'dispatches',json_array(project,run,epoch,dispatch_key,hex(CAST(manifest AS BLOB)),manifest_hash,state,worker,journal_locator,hex(CAST(identity AS BLOB)),observation_sequence,observation_hash,prepared_at,updated_at,reason) FROM runtime_dispatches
 UNION ALL SELECT 'observations',json_array(project,run,epoch,receipt_hash,hex(CAST(body AS BLOB)),observed_at) FROM runtime_observations
 UNION ALL SELECT 'effects',json_array(project,run,effect_key,sequence,hex(CAST(body AS BLOB))) FROM resource_effects
 UNION ALL SELECT 'routes',json_array(id,project,run,hex(CAST(body AS BLOB))) FROM resource_route_history
 UNION ALL SELECT 'uploads',json_array(storage_key,project,run,epoch) FROM runtime_uploads
 UNION ALL SELECT 'content',json_array(storage_key,project,principal,hex(bytes),content_hash,expires_at,sealed) FROM uploaded_content
 UNION ALL SELECT 'fences',json_array(project,task_id,session_id,generation,state,accepted_through,sealed_through,run_id,runtime_epoch,dispatch_key,manifest_hash,CASE WHEN process_identity IS NULL THEN NULL ELSE hex(CAST(process_identity AS BLOB)) END,restore_fenced,revision) FROM client_queue_fences
 UNION ALL SELECT 'queue',json_array(principal,project,command_id,session_id,task_id,goal_id,generation,queue_sequence,request_hash,message_id,accepted_writer_epoch,delivery,hex(CAST(content AS BLOB)),content_hash,state,revision,last_receipt_id,created_at) FROM client_command_queue
 UNION ALL SELECT 'controls',json_array(principal,project,command_id,session_id,control_request_id,request_hash,target_id,hex(CAST(task_ids AS BLOB)),state,CASE WHEN original_response IS NULL THEN NULL ELSE hex(CAST(original_response AS BLOB)) END,CASE WHEN final_response IS NULL THEN NULL ELSE hex(CAST(final_response AS BLOB)) END,created_at) FROM client_control_requests
 UNION ALL SELECT 'receipts',json_array(receipt_id,project,generation,command_id,kind,hex(CAST(body AS BLOB)),body_hash,previous_receipt_id,created_at) FROM client_control_receipts
 UNION ALL SELECT 'sessions',json_array(id,project_id,revision,created_at,updated_at,goal_id,task_id,run_id,writer_epoch,writer_client_id,lease_expires_at,event_watermark,upstream_session_id,upstream_log_id,upstream_log_hash,workspace_ref,title,model_config_ref,core_build_id,transcript_artifact_id,resume_requires_explicit_user,sealed_at,archived_at,protocol) FROM client_sessions
 UNION ALL SELECT 'legacy',json_array(project,run_id,hex(CAST(inventory AS BLOB)),inventory_hash) FROM f08_legacy_runs
 UNION ALL SELECT 'origins',json_array(project,run_id,epoch,dispatch_key,hex(CAST(document AS BLOB)),body_hash) FROM f08_dispatch_origins
 UNION ALL SELECT 'admitted',json_array(principal,project,command_id,request_hash,operation,hex(CAST(private_header AS BLOB)),root_hash,group_hash,hex(CAST(document AS BLOB))) FROM f08_admitted_groups
 UNION ALL SELECT 'admitted_members',json_array(principal,project,command_id,request_hash,task_id,member_index,root_hash,body_hash,hex(CAST(document AS BLOB))) FROM f08_admitted_members
 UNION ALL SELECT 'final',json_array(principal,project,command_id,request_hash,admitted_root_hash,root_hash,group_hash,hex(CAST(document AS BLOB))) FROM f08_final_groups
 UNION ALL SELECT 'final_members',json_array(principal,project,command_id,request_hash,task_id,member_index,admitted_member_hash,body_hash,hex(CAST(document AS BLOB))) FROM f08_final_members
 UNION ALL SELECT 'commits',json_array(principal,project,command_id,request_hash,final_root_hash,original_response_digest,final_response_digest) FROM f08_final_commits
