// All data paths must resolve inside a parent-created, fresh fixture directory.
// Nonexistent output leaves are allowed only below an existing contained parent.
func historical22FixturePath(root, path string) error {
    resolvedRoot, err := filepath.EvalSymlinks(root)
    if err != nil { return err }
    if !filepath.IsAbs(path) { return fmt.Errorf("fixture path must be absolute") }
    parent, err := filepath.EvalSymlinks(filepath.Dir(path))
    if err != nil { return err }
    rel, err := filepath.Rel(resolvedRoot, parent)
    if err != nil || rel == ".." || strings.HasPrefix(rel, ".."+string(os.PathSeparator)) || filepath.IsAbs(rel) {
        return fmt.Errorf("fixture path escaped parent temporary root")
    }
    if _, err = os.Lstat(path); err == nil {
        resolved, resolveErr := filepath.EvalSymlinks(path)
        if resolveErr != nil { return resolveErr }
        rel, resolveErr = filepath.Rel(resolvedRoot, resolved)
        if resolveErr != nil || rel == ".." || strings.HasPrefix(rel, ".."+string(os.PathSeparator)) || filepath.IsAbs(rel) {
            return fmt.Errorf("fixture member escaped parent temporary root")
        }
    } else if !os.IsNotExist(err) { return err }
    return nil
}

func historical22CheckSource(t *testing.T, root string) {
    t.Helper()
    if Digest(historical22BridgeInputs) != historical22BridgeInputSHA {
        t.Fatal("historical input manifest bytes changed")
    }
    var inputs struct { Commit string `json:"commit"`; Files map[string]string `json:"files"` }
    if err := json.Unmarshal(historical22BridgeInputs, &inputs); err != nil { t.Fatal(err) }
    if inputs.Commit != "396287cc5427eb125aaedb8a2b7fadf63e6b9322" || len(inputs.Files) != 502 {
        t.Fatal("historical input manifest profile mismatch")
    }
    for rel, expected := range inputs.Files {
        path := filepath.Join(root, filepath.FromSlash(rel))
        if err := historical22FixturePath(root, path); err != nil { t.Fatal(err) }
        raw, err := os.ReadFile(path)
        if err != nil || Digest(raw) != "sha256:"+expected {
            t.Fatalf("historical input changed: %s", rel)
        }
    }
}

type historical22FixturePort struct { workspace string; admitErr error }
func (p historical22FixturePort) ResolveGoalBinding(context.Context, *sql.Tx, string, string, string) (string, string, error) {
    return p.workspace, "fixture-unqualified-model", nil
}
func (p historical22FixturePort) ValidateBinding(_ context.Context, _ *sql.Tx, _, _, workspace, model string) error {
    if workspace != p.workspace || model != "fixture-unqualified-model" { return fmt.Errorf("fixture binding mismatch") }
    return nil
}
func (p historical22FixturePort) AdmitTurn(context.Context, *sql.Tx, string, string, Object) error { return p.admitErr }

        t.Fatal("explicit schema22 fixture bridge invocation required")
    }
    decoder := json.NewDecoder(io.LimitReader(os.Stdin, 64*1024))
    decoder.DisallowUnknownFields()
    var request historical22BridgeRequest
    if err := decoder.Decode(&request); err != nil { t.Fatal("private fixture request decode failed") }
