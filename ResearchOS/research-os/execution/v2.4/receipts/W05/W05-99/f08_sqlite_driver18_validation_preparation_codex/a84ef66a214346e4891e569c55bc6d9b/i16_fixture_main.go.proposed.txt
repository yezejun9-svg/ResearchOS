// Future separate fork-linked producer. Runs only under a Root-owned process.
package main

import (
	"context"
	"database/sql"
	"database/sql/driver"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"time"

	sqlite "modernc.org/sqlite"
)

func run() (resultErr error) {
	if len(os.Args)!=2||!filepath.IsAbs(os.Args[1]){return fmt.Errorf("one fresh absolute DB path required")}
	p:=os.Args[1]
	for _,name:=range []string{p,p+"-wal",p+"-shm",p+"-journal"}{
		if _,err:=os.Lstat(name);!os.IsNotExist(err){return fmt.Errorf("fresh fixture path is not absent: %s (%v)",name,err)}
	}
	// Exclusive reserve of the fresh DB file; never overwrite an existing DB.
	f,err:=os.OpenFile(p,os.O_CREATE|os.O_EXCL|os.O_RDWR,0600);if err!=nil{return err};if err=f.Close();err!=nil{return err}
	err=sqlite.RegisterFunction("ros_i16",&sqlite.FunctionImpl{NArgs:1,Deterministic:true,Innocuous:true,Scalar:func(_ *sqlite.FunctionContext,a []driver.Value)(driver.Value,error){
		if len(a)!=1{return nil,fmt.Errorf("arity")};x,ok:=a[0].(int64);if !ok{return nil,fmt.Errorf("integer required")};if x>0{return int64(1),nil};return int64(0),nil
	}});if err!=nil{return err}
	db,err:=sql.Open("sqlite",p);if err!=nil{return err};db.SetMaxOpenConns(1)
	closed:=false;defer func(){if !closed{if err:=db.Close();err!=nil&&resultErr==nil{resultErr=err}}}()
	ctx,cancel:=context.WithTimeout(context.Background(),15*time.Second);defer cancel()
	c,err:=db.Conn(ctx);if err!=nil{return err};connClosed:=false
	defer func(){if !connClosed{if err:=c.Close();err!=nil&&resultErr==nil{resultErr=err}}}()
	if _,err=c.ExecContext(ctx,"PRAGMA trusted_schema=OFF");err!=nil{return err}
	var off int;if err=c.QueryRowContext(ctx,"PRAGMA trusted_schema").Scan(&off);err!=nil||off!=0{return fmt.Errorf("OFF readback %d %v",off,err)}
	var mode string;if err=c.QueryRowContext(ctx,"PRAGMA journal_mode=DELETE").Scan(&mode);err!=nil||mode!="delete"{return fmt.Errorf("DELETE journal %s %v",mode,err)}
	if _,err=c.ExecContext(ctx,"CREATE TABLE checked(x INTEGER CHECK(ros_i16(x)=1))");err!=nil{return err}
	if _,err=c.ExecContext(ctx,"INSERT INTO checked VALUES(7)");err!=nil{return err}
	if err=c.Close();err!=nil{return err};connClosed=true
	if err=db.Close();err!=nil{return err};closed=true
	for _,side:=range []string{p+"-wal",p+"-shm",p+"-journal"}{if _,err:=os.Lstat(side);!os.IsNotExist(err){return fmt.Errorf("unexpected sidecar %s: %v",side,err)}}
	return json.NewEncoder(os.Stdout).Encode(map[string]any{"schema":"researchos.driver18.fixture-prepared.v1","database":p,"closed":true,"phase_only_not_case_pass":true})
}
func main(){if err:=run();err!=nil{fmt.Fprintln(os.Stderr,err);os.Exit(1)}}
