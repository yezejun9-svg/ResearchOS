52: func ownedStopFrame(h hash.Hash, kind byte, raw []byte) {
53: 	_, _ = h.Write([]byte{kind})
54: 	var n [8]byte
55: 	binary.BigEndian.PutUint64(n[:], uint64(len(raw)))
56: 	_, _ = h.Write(n[:])
57: 	_, _ = h.Write(raw)

59: func ownedStopCell(h hash.Hash, v any) error {
60: 	switch x := v.(type) {
61: 	case nil:
62: 		ownedStopFrame(h, 'N', nil)
63: 	case []byte:
64: 		ownedStopFrame(h, 'B', x)
65: 	case string:
66: 		ownedStopFrame(h, 'S', []byte(x))
67: 	case int64:
68: 		ownedStopFrame(h, 'I', []byte(fmt.Sprintf("%d", x)))
69: 	case float64:
70: 		ownedStopFrame(h, 'F', []byte(fmt.Sprintf("%016x", math.Float64bits(x))))
71: 	case bool:
72: 		if x {
73: 			ownedStopFrame(h, 'T', nil)
74: 		} else {
75: 			ownedStopFrame(h, 'f', nil)
76: 		}
77: 	case time.Time:
78: 		ownedStopFrame(h, 'D', []byte(x.UTC().Format(time.RFC3339Nano)))
79: 	default:
80: 		return fmt.Errorf("unsupported SQLite cell type %T", v)
81: 	}
82: 	return nil
83: }

213: // OnlineOwnedStopBackup creates a new destination outside the source root and
214: // uses the pinned native online-backup API. It returns only hashes and counts.
215: func OnlineOwnedStopBackup(ctx context.Context, source, destination string) (OwnedStopCensus, string, error) {
216: 	if !filepath.IsAbs(source) || !filepath.IsAbs(destination) || filepath.Clean(destination) != destination {
217: 		return OwnedStopCensus{}, "", errors.New("canonical backup paths required")
218: 	}
219: 	rel, e := filepath.Rel(filepath.Dir(source), destination)
220: 	if e != nil || (!strings.HasPrefix(rel, ".."+string(filepath.Separator)) && rel != "..") {
221: 		return OwnedStopCensus{}, "", errors.New("backup destination must be outside data root")
222: 	}
223: 	if _, err := os.Stat(source); err != nil {
224: 		return OwnedStopCensus{}, "", err
225: 	}
226: 	f, err := os.OpenFile(destination, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
227: 	if err != nil {
228: 		return OwnedStopCensus{}, "", err
229: 	}
230: 	if err = f.Close(); err != nil {
231: 		return OwnedStopCensus{}, "", err
232: 	}
233: 	src, err := sql.Open("sqlite", ownedStopURI(source, "rw"))
234: 	if err != nil {
235: 		return OwnedStopCensus{}, "", err
236: 	}
237: 	defer src.Close()
238: 	src.SetMaxOpenConns(1)
239: 	conn, err := src.Conn(ctx)
240: 	if err != nil {
241: 		return OwnedStopCensus{}, "", err
242: 	}
243: 	defer conn.Close()
244: 	destURI := ownedStopURI(destination, "rw")
245: 	err = conn.Raw(func(driverConn any) error {
246: 		maker, ok := driverConn.(interface {
247: 			NewBackup(string) (*sqlite.Backup, error)
248: 		})
249: 		if !ok {
250: 			return errors.New("pinned native backup unavailable")
251: 		}
252: 		backup, e := maker.NewBackup(destURI)
253: 		if e != nil {
254: 			return e
255: 		}
256: 		var stepErr error
257: 		complete := false
258: 		for steps := 0; steps < 100000; steps++ {
259: 			if e = ctx.Err(); e != nil {
260: 				stepErr = e
261: 				break
262: 			}
263: 			var more bool
264: 			more, e = backup.Step(128)
265: 			if e != nil {
266: 				stepErr = e
267: 				break
268: 			}
269: 			if !more {
270: 				complete = true
271: 				break
272: 			}
273: 		}
274: 		if !complete && stepErr == nil {
275: 			stepErr = errors.New("online backup step limit")
276: 		}
277: 		return errors.Join(stepErr, backup.Finish())
278: 	})
279: 	if err != nil {
280: 		return OwnedStopCensus{}, "", err
281: 	}
282: 	closed, err := os.OpenFile(destination, os.O_RDWR, 0)
283: 	if err != nil {
284: 		return OwnedStopCensus{}, "", err
285: 	}
286: 	if err = closed.Sync(); err != nil {
287: 		closed.Close()
288: 		return OwnedStopCensus{}, "", err
289: 	}
290: 	h := sha256.New()
291: 	_, err = io.Copy(h, closed)
292: 	closeErr := closed.Close()
293: 	if err != nil {
294: 		return OwnedStopCensus{}, "", err
295: 	}
296: 	if closeErr != nil {
297: 		return OwnedStopCensus{}, "", closeErr
298: 	}
299: 	ro, err := sql.Open("sqlite", ownedStopURI(destination, "ro"))
300: 	if err != nil {
301: 		return OwnedStopCensus{}, "", err
302: 	}
303: 	defer ro.Close()
304: 	ro.SetMaxOpenConns(1)
305: 	census, err := ownedStopCensus(ctx, ro)
306: 	return census, hex.EncodeToString(h.Sum(nil)), err
307: }
