51: func validateAuthorityDocument(d grantAuthorityDocument, g operationGrant, role, action string) error {
52: 	if d.Schema != grantAuthoritySchema || d.Role != role || d.OperationID != g.OperationID || d.Action != action || d.Subject != g.Subject || d.Native != g.Native || d.Host != g.Host || d.OperationDir != g.OperationDir || d.HistoricalManifestSHA256 != g.HistoricalManifest.SHA256 || d.Reviewer == "" || strings.Contains(d.Reviewer, "owned_core_stop_builder") || d.Verdict != "ACCEPTED" {
53: 		return errors.New("authority document semantics differ")

65: func validateHistoricalManifest(g operationGrant, d *heldDocuments, evidence []EvidenceRef) error {
66: 	raw, err := d.read(g.HistoricalManifest, 128<<10)
67: 	if err != nil {
68: 		return err
69: 	}
70: 	var m historicalManifest
71: 	if err = decodeCanonicalAuthority(raw, &m); err != nil {
72: 		return err
73: 	}
74: 	if m.Schema != historicalManifestSchema || m.OperationID != g.OperationID || len(m.Evidence) != len(proofKinds) || len(evidence) != len(proofKinds) {
75: 		return errors.New("historical manifest identity/count")
76: 	}
77: 	for i, k := range proofKinds {

79: 			return errors.New("historical manifest class/order/binding")
80: 		}
81: 	}
82: 	return nil

179: func verifierPath(repo, path string) error {
180: 	if !filepath.IsAbs(repo) || !filepath.IsAbs(path) || filepath.Clean(path) != path {
181: 		return errors.New("canonical independent evidence path required")
182: 	}
183: 	root := filepath.Join(repo, "execution", "verifier")
184: 	rel, err := filepath.Rel(root, path)
185: 	if err != nil || rel == "." || strings.HasPrefix(rel, ".."+string(filepath.Separator)) || rel == ".." || filepath.IsAbs(rel) {
186: 		return errors.New("evidence outside independent verifier namespace")
187: 	}
188: 	actual, err := filepath.EvalSymlinks(path)
189: 	if err != nil || !strings.EqualFold(actual, path) {
190: 		return errors.New("evidence reparse or absence")
191: 	}
192: 	return nil
193: }

195: func readEvidence(repo string, ref EvidenceRef) (historicalEvidence, error) {
196: 	var out historicalEvidence
197: 	if !validSHA(ref.SHA256) {
198: 		return out, errors.New("evidence hash required")
199: 	}
200: 	if err := verifierPath(repo, ref.Path); err != nil {
201: 		return out, err
202: 	}
203: 	raw, err := os.ReadFile(ref.Path)
204: 	if err != nil {
205: 		return out, err
206: 	}
207: 	if len(raw) == 0 || len(raw) > 128<<10 || digest(raw) != ref.SHA256 {
208: 		return out, errors.New("evidence content/hash mismatch")
209: 	}
210: 	if err = strictJSON(raw, &out); err != nil {
211: 		return out, err
212: 	}
213: 	if out.Schema != proofSchema || out.Kind != ref.Kind || out.Reviewer == "" || strings.Contains(out.Reviewer, "owned_core_stop_builder") {
214: 		return out, errors.New("independent evidence identity invalid")
215: 	}
216: 	return out, nil
217: }

220: 	switch kind {
221: 	case "f01_ready":

240: func nativeExitEvidenceComplete(receipt historicalEvidence, corePID uint32) bool {
241: 	return receipt.NativePID != 0 && receipt.NativePID != corePID && receipt.NativeCreatedNS != 0 &&
242: 		receipt.NativeWaitAPI == "WaitForSingleObject" && receipt.NativeWaitResult != nil && *receipt.NativeWaitResult == 0 &&
243: 		receipt.NativeExitAPI == "GetExitCodeProcess" && receipt.NativeExitCode != nil &&
244: 		validSHA(receipt.NativeHandleRecordSHA256)

247: func validateHistoricalEvidence(plan Plan, held ...*heldDocuments) (string, []string) {
248: 	missing := []string{}
249: 	refs := map[string]EvidenceRef{}
250: 	for _, r := range plan.Input.Evidence {
251: 		if _, exists := refs[r.Kind]; exists {
252: 			missing = append(missing, "duplicate "+r.Kind)
253: 		}
254: 		refs[r.Kind] = r
255: 	}
256: 	if len(refs) != len(proofKinds) {
257: 		missing = append(missing, "exact seven independent receipts required")
258: 	}
259: 	observed := map[string]string{}
260: 	for _, kind := range proofKinds {
261: 		ref, ok := refs[kind]
262: 		if !ok {
263: 			missing = append(missing, kind+": missing")
264: 			continue
265: 		}
266: 		var receipt historicalEvidence
267: 		var err error
268: 		if len(held) > 0 && held[0] != nil {
269: 			if err = verifierPath(plan.Input.SourceRepository, ref.Path); err == nil {
270: 				var raw []byte
271: 				raw, err = held[0].read(grantDocument{Path: ref.Path, SHA256: ref.SHA256}, 128<<10)
272: 				if err == nil {
273: 					err = decodeCanonicalAuthority(raw, &receipt)
274: 				}
275: 				if err == nil && (receipt.Schema != proofSchema || receipt.Kind != ref.Kind || receipt.Reviewer == "" || strings.Contains(receipt.Reviewer, "owned_core_stop_builder")) {
276: 					err = errors.New("independent evidence identity invalid")
277: 				}
278: 			}
279: 		} else {
280: 			receipt, err = readEvidence(plan.Input.SourceRepository, ref)
281: 		}
282: 		if err != nil {
283: 			missing = append(missing, kind+": "+err.Error())
284: 			continue
285: 		}
286: 		if receipt.OperationID != plan.OperationID || receipt.SourceCommit != SourceBaseCommit || receipt.CorePID != plan.Native.PID || receipt.CoreCreatedNS != plan.Native.CreatedNS || receipt.CoreImageSHA256 != plan.Native.ImageSHA256 || receipt.HostID != plan.Host.HostID || receipt.OwnerID != plan.Host.OwnerID || receipt.OwnerRoot != plan.Host.OwnerRoot || receipt.Subject != plan.Input.Subject {
287: 			missing = append(missing, kind+": subject/authority/core binding differs")
288: 			continue
289: 		}
290: 		if kind == "worker_native_exit" || kind == "verifier_native_exit" {
291: 			// An ignored cmd.Wait error, a PID-list absence, or an assumed exit 0
292: 			// cannot satisfy this typed original OS wait and numeric-exit receipt.
293: 			if !nativeExitEvidenceComplete(receipt, plan.Native.PID) {
294: 				missing = append(missing, kind+": captured exact OS wait/exit/handle identity absent")
295: 			}
296: 		}
297: 		if kind == "effect_census" && receipt.BaselineSHA256 != plan.Baseline.SHA256 {
298: 			missing = append(missing, kind+": full baseline differs")
299: 		}
300: 		for _, name := range proofNeeds(kind) {
301: 			hash := receipt.RawSHA256[name]
302: 			if !validSHA(hash) {
303: 				missing = append(missing, fmt.Sprintf("%s: %s raw receipt missing", kind, name))
304: 			}
305: 			if kind == "independent_evidence_review" && hash != observed[name] {
306: 				missing = append(missing, kind+": evidence hash differs for "+name)
307: 			}
308: 		}
309: 		observed[kind] = ref.SHA256
310: 	}
311: 	// Bind the seven references and their order into a single reviewable digest.
312: 	raw, _ := canonicalJSON(plan.Input.Evidence)
313: 	return digest(raw), missing
314: }

316: func validatePlanReview(plan Plan, planSHA string) error {
317: 	if plan.Input.PlanReviewPath == "" {
318: 		return errors.New("independent operation plan review missing")
319: 	}
320: 	if err := verifierPath(plan.Input.SourceRepository, plan.Input.PlanReviewPath); err != nil {
321: 		return err
322: 	}
323: 	raw, err := os.ReadFile(plan.Input.PlanReviewPath)
324: 	if err != nil {
325: 		return err
326: 	}
327: 	if len(raw) == 0 || len(raw) > 64<<10 {
328: 		return errors.New("bounded plan review required")
329: 	}
330: 	var r operationPlanReview
331: 	if err = strictJSON(raw, &r); err != nil {
332: 		return err
333: 	}
334: 	if r.Schema != planReviewSchema || r.OperationID != plan.OperationID || r.PlanSHA256 != planSHA || r.EvidenceSHA256 != plan.EvidenceSHA256 || r.SourceCommit != SourceBaseCommit || r.CoreCreatedNS != plan.Native.CreatedNS || r.CoreImageSHA256 != plan.Native.ImageSHA256 || r.Reviewer == "" || strings.Contains(r.Reviewer, "owned_core_stop_builder") || r.Verdict != "ACCEPTED" || !validSHA(r.RootOperationScopeSHA256) {
335: 		return errors.New("independent exact plan/root-scope review unavailable")
336: 	}
337: 	return nil
338: }
