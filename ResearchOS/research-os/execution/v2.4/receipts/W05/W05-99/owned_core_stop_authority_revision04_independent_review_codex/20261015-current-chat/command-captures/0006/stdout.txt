504: func (h *OwnedCoreRetainedHandle) TerminateOnceAndWaitUntilContext(ctx context.Context, timeout time.Duration, invocationDeadline, wallExpiry time.Time) (OwnedCoreNativeExit, error) {
505: 	var out OwnedCoreNativeExit
506: 	if ctx == nil {
507: 		return out, errors.New("caller context required")
508: 	}
509: 	if timeout <= 0 || timeout > 30*time.Second {
510: 		return out, errors.New("bounded native wait required")
511: 	}
512: 	if earlier, ok := ctx.Deadline(); ok && earlier.Before(invocationDeadline) {
513: 		invocationDeadline = earlier
514: 	}
515: 	if err := ctx.Err(); err != nil {
516: 		return out, fmt.Errorf("caller unavailable before native signal: %w", err)
517: 	}
518: 	if err := h.verifyRetained(); err != nil {
519: 		return out, err
520: 	}
521: 	if h.terminateCalls != 0 {
522: 		return out, errors.New("native termination already attempted")
523: 	}
524: 	status, err := h.nativeWait(0)
525: 	if err != nil {
526: 		return out, err
527: 	}
528: 	if status == windows.WAIT_OBJECT_0 {
529: 		out.BeforeSignal = true
530: 		out.ExitObserved = true
531: 		if err = windows.GetExitCodeProcess(h.core.handle, &out.ExitCode); err != nil {
532: 			return out, err
533: 		}
534: 		return out, nil
535: 	}
536: 	if status != uint32(windows.WAIT_TIMEOUT) {
537: 		return out, fmt.Errorf("unexpected native wait status %d", status)
538: 	}
539: 	if err := ctx.Err(); err != nil {
540: 		return out, fmt.Errorf("caller unavailable before native signal: %w", err)
541: 	}
542: 	if now := time.Now(); !now.Before(invocationDeadline) || !now.Before(wallExpiry) {
543: 		return out, errors.New("invocation or grant expired before native signal")
544: 	}
545: 	if err := ctx.Err(); err != nil {
546: 		return out, fmt.Errorf("caller unavailable before native signal: %w", err)
547: 	}
548: 	h.terminateCalls = 1
549: 	out.TerminateCalls = 1
550: 	if err = h.nativeTerminate(OwnedCoreMaintenanceExitCode); err != nil {
551: 		return out, fmt.Errorf("one TerminateProcess failed: %w", err)
552: 	}
553: 	status, err = h.nativeWait(uint32(timeout.Milliseconds()))
554: 	if err != nil || status != windows.WAIT_OBJECT_0 {
555: 		return out, fmt.Errorf("native wait unknown: status=%d err=%v", status, err)
556: 	}
557: 	out.ExitObserved = true
558: 	if err = windows.GetExitCodeProcess(h.core.handle, &out.ExitCode); err != nil {
559: 		return out, err
560: 	}
561: 	return out, nil
562: }
