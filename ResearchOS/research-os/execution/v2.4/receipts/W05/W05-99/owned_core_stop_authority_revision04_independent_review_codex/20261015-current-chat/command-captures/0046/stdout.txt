32: func fileID(path string, dir bool) (nativeFile, error) {
33: 	w, e := windows.UTF16PtrFromString(path)
34: 	if e != nil {
35: 		return nativeFile{}, e
36: 	}
37: 	h, e := windows.CreateFile(w, windows.FILE_READ_ATTRIBUTES, windows.FILE_SHARE_READ|windows.FILE_SHARE_WRITE|windows.FILE_SHARE_DELETE, nil, windows.OPEN_EXISTING, windows.FILE_FLAG_OPEN_REPARSE_POINT|windows.FILE_FLAG_BACKUP_SEMANTICS, 0)
38: 	if e != nil {
39: 		return nativeFile{}, e
40: 	}
41: 	defer windows.CloseHandle(h)
42: 	var i windows.ByHandleFileInformation
43: 	if e = windows.GetFileInformationByHandle(h, &i); e != nil {
44: 		return nativeFile{}, e
45: 	}
46: 	if i.FileAttributes&windows.FILE_ATTRIBUTE_REPARSE_POINT != 0 || (i.FileAttributes&windows.FILE_ATTRIBUTE_DIRECTORY != 0) != dir {
47: 		return nativeFile{}, fmt.Errorf("reparse or type mismatch: %s", path)
48: 	}
49: 	return nativeFile{path, i.VolumeSerialNumber, uint64(i.FileIndexHigh)<<32 | uint64(i.FileIndexLow)}, nil
50: }
51: func private(path string, dir bool) error {
52: 	token := windows.GetCurrentProcessToken()
53: 	u, e := token.GetTokenUser()
54: 	if e != nil {
55: 		return e
56: 	}
57: 	sid := u.User.Sid.String()
58: 	ace := "(A;;FA;;;" + sid + ")"
59: 	if dir {
60: 		ace = "(A;OICI;FA;;;" + sid + ")"
61: 	}
62: 	sd, e := windows.SecurityDescriptorFromString("O:" + sid + "D:P" + ace)
63: 	if e != nil {
64: 		return e
65: 	}
66: 	acl, _, e := sd.DACL()
67: 	if e != nil {
68: 		return e
69: 	}
70: 	return windows.SetNamedSecurityInfo(path, windows.SE_FILE_OBJECT, windows.DACL_SECURITY_INFORMATION|windows.PROTECTED_DACL_SECURITY_INFORMATION, nil, nil, acl, nil)
71: }
