99: func parseDecimal100NS(s string) (int64, error) {
100: 	if s == "" || (len(s) > 1 && s[0] == '0') {
101: 		return 0, errors.New("noncanonical creation time")
102: 	}
103: 	v, err := strconv.ParseInt(s, 10, 64)
104: 	if err != nil || v <= 0 {
105: 		return 0, errors.New("invalid creation time")
106: 	}
107: 	return v, nil
108: }

110: func (n grantNativeIdentity) observed() (supervisor.OwnedCoreNativeIdentity, error) {
111: 	c, err := parseDecimal100NS(n.Created100NS)
112: 	if err != nil {
113: 		return supervisor.OwnedCoreNativeIdentity{}, err
114: 	}
115: 	p, err := parseDecimal100NS(n.ParentCreated100NS)
116: 	if err != nil {
117: 		return supervisor.OwnedCoreNativeIdentity{}, err
118: 	}
119: 	// Windows FILETIME.Nanoseconds() is the observed representation in the
120: 	// retained native handle. The grant stores the exact 100ns unit as a string.
121: 	if c > int64(^uint64(0)>>1)/100 || p > int64(^uint64(0)>>1)/100 {
122: 		return supervisor.OwnedCoreNativeIdentity{}, errors.New("creation time overflow")
123: 	}
124: 	return supervisor.OwnedCoreNativeIdentity{PID: n.PID, CreatedNS: c * 100, SID: n.SID, ImagePath: n.ImagePath, ImageSHA256: n.ImageSHA256, ImageVolume: n.ImageVolume, ImageIndex: n.ImageIndex, ParentPID: n.ParentPID, ParentCreatedNS: p * 100, ParentSID: n.ParentSID, ParentImagePath: n.ParentImagePath, ParentImageSHA256: n.ParentImageSHA256, ParentImageVolume: n.ParentImageVolume, ParentImageIndex: n.ParentImageIndex}, nil

131: func parseCompiledOperationGrant(encoded string) (operationGrant, error) {
132: 	var g operationGrant
133: 	if encoded == "" {
134: 		return g, errors.New("STOP_GRANT_UNCONFIGURED")
135: 	}
136: 	if len(encoded) > base64.StdEncoding.EncodedLen(64<<10) {
137: 		return g, errors.New("STOP_GRANT_INVALID: oversized base64")
138: 	}
139: 	raw, err := base64.StdEncoding.Strict().DecodeString(encoded)
140: 	if err != nil || len(raw) == 0 || len(raw) > 64<<10 || base64.StdEncoding.EncodeToString(raw) != encoded {
141: 		return g, errors.New("STOP_GRANT_INVALID: noncanonical base64")
142: 	}
143: 	if err = strictJSON(raw, &g); err != nil {
144: 		return operationGrant{}, fmt.Errorf("STOP_GRANT_INVALID: %w", err)
145: 	}
146: 	canonical, err := canonicalMapJSON(raw)
147: 	if err != nil || !bytes.Equal(raw, canonical) {
148: 		return operationGrant{}, errors.New("STOP_GRANT_INVALID: noncanonical JSON")
149: 	}
150: 	g.digest = digest(raw)
151: 	if err = g.validateStructure(); err != nil {
152: 		return operationGrant{}, fmt.Errorf("STOP_GRANT_INVALID: %w", err)
153: 	}
154: 	return g, nil
155: }

157: func canonicalMapJSON(raw []byte) ([]byte, error) {
158: 	d := json.NewDecoder(bytes.NewReader(raw))
159: 	d.UseNumber()
160: 	var v map[string]any
161: 	if err := d.Decode(&v); err != nil {
162: 		return nil, err
163: 	}
164: 	if v == nil {
165: 		return nil, errors.New("JSON object required")
166: 	}
167: 	out, err := json.MarshalIndent(v, "", "  ")
168: 	if err != nil {
169: 		return nil, err
170: 	}
171: 	return append(out, '\n'), nil
172: }

179: func (g *operationGrant) validateStructure() error {
180: 	if g.Schema != grantSchema || !validID(g.OperationID) || g.Purpose != "EXACT_RETAINED_CORE_CRASH_STOP" || g.TrustedOperatorSID == "" || g.RulesetVersion != "strict-closure-v1" || !validSHA(g.SchemaSHA256) {
181: 		return errors.New("grant identity/ruleset")
182: 	}
183: 	issued, e1 := time.Parse(time.RFC3339Nano, g.IssuedAt)
184: 	before, e2 := time.Parse(time.RFC3339Nano, g.NotBefore)
185: 	expiry, e3 := time.Parse(time.RFC3339Nano, g.ExpiresAt)
186: 	if e1 != nil || e2 != nil || e3 != nil || !issued.Equal(before) && issued.After(before) || !expiry.After(before) || expiry.Sub(before) > 30*time.Minute {
187: 		return errors.New("grant interval")
188: 	}
189: 	g.notBefore, g.expires = before, expiry
190: 	for _, f := range []grantFileIdentity{g.DataRoot, g.Database, g.OperationDir} {
191: 		if cleanOwnedLocalPath(f.Path) != nil || f.Volume == 0 || f.Index == 0 {
192: 			return errors.New("grant native file identity")
193: 		}
194: 	}
195: 	if outsideDataRoot(g.OperationDir.Path, g.DataRoot.Path) != nil || g.Database.Path != filepath.Join(g.DataRoot.Path, "researchos.sqlite") {
196: 		return errors.New("grant root/database path")
197: 	}
198: 	n, err := g.Native.observed()
199: 	if err != nil || n.PID == 0 || n.ParentPID == 0 || n.SID != g.TrustedOperatorSID || n.ParentSID != g.TrustedOperatorSID || !validSHA(n.ImageSHA256) || !validSHA(n.ParentImageSHA256) || n.ImageIndex == 0 || n.ParentImageIndex == 0 || cleanOwnedLocalPath(n.ImagePath) != nil || cleanOwnedLocalPath(n.ParentImagePath) != nil {
200: 		return errors.New("grant native process identity")
201: 	}
202: 	if g.Host.DataRoot != g.DataRoot.Path || g.Host.Database != g.Database.Path || g.Host.DatabaseVolume != g.Database.Volume || g.Host.DatabaseIndex != g.Database.Index || g.Host.SID != g.TrustedOperatorSID || g.Host.SchemaVersion != 24 || g.Host.HostID == "" || g.Host.OwnerID == "" || g.Host.OwnerRoot == "" || g.Host.PrincipalRef == "" || !validSHA(g.Host.HostStateSHA256) {
203: 		return errors.New("grant Host/Owner identity")
204: 	}
205: 	if g.Subject.Run != OldRun || g.Subject.Epoch != 1 || g.Subject.DispatchKey != OldDispatch || g.Subject.RequestKey != OldRequest || g.Subject.Project == "" {
206: 		return errors.New("grant old subject")
207: 	}
208: 	if len(g.RequiredProofKinds) != len(proofKinds) {
209: 		return errors.New("proof classes")
210: 	}
211: 	for i, k := range proofKinds {
212: 		if g.RequiredProofKinds[i] != k {
213: 			return errors.New("proof class mismatch")
214: 		}
215: 	}
216: 	if !validGrantDocument(g.HistoricalManifest) {
217: 		return errors.New("historical manifest pin")
218: 	}
219: 	switch g.GrantKind {
220: 	case "PREFLIGHT_ONLY":
221: 		if g.TerminateCount != 0 || len(g.AllowedActions) != 2 || g.AllowedActions[0] != "preflight" || g.AllowedActions[1] != "inspect" || !validGrantDocument(g.Proposal) || !validGrantDocument(g.PreflightScope) || !validGrantDocument(g.PreflightReview) || g.PlanSHA256 != "" || g.PreflightManifestSHA256 != "" || g.StopReview.Path != "" || g.RootOperationScope.Path != "" || g.PreflightGrantSHA256 != "" || g.PreflightExecutableSHA256 != "" {
222: 			return errors.New("preflight grant actions/pins")
223: 		}
224: 	case "STOP_ONCE":
225: 		if g.TerminateCount != 1 || len(g.AllowedActions) != 2 || g.AllowedActions[0] != "commit" || g.AllowedActions[1] != "inspect" || !validSHA(g.PlanSHA256) || !validSHA(g.PreflightManifestSHA256) || !validGrantDocument(g.StopReview) || !validGrantDocument(g.RootOperationScope) || !validSHA(g.PreflightGrantSHA256) || !validSHA(g.PreflightExecutableSHA256) || g.Proposal.Path != "" || g.PreflightScope.Path != "" || g.PreflightReview.Path != "" {
226: 			return errors.New("stop grant actions/pins")
227: 		}
228: 	default:
229: 		return errors.New("grant kind")
230: 	}
231: 	return nil
232: }

234: func (g operationGrant) checkAction(action string, live bool) (time.Time, error) {
235: 	allowed := false
236: 	for _, a := range g.AllowedActions {
237: 		if a == action {
238: 			allowed = true
239: 		}
240: 	}
241: 	if !allowed {
242: 		return time.Time{}, errors.New("STOP_GRANT_ACTION_DENIED")
243: 	}
244: 	if !live {
245: 		return time.Time{}, nil
246: 	}
247: 	now := time.Now()
248: 	if now.Before(g.notBefore) || !now.Before(g.expires) {
249: 		return time.Time{}, errors.New("STOP_GRANT_EXPIRED")
250: 	}
251: 	return now.Add(g.expires.Sub(now)), nil
252: }

315: func validSHA(s string) bool {
316: 	b, e := hex.DecodeString(s)
317: 	if e != nil || len(b) != 32 || strings.ToLower(s) != s {
318: 		return false
319: 	}
320: 	for _, v := range b {
321: 		if v != 0 {
322: 			return true
323: 		}
324: 	}

332: func strictJSON(raw []byte, v any) error {
333: 	if err := rejectDuplicateJSON(raw); err != nil {
334: 		return err
335: 	}
336: 	d := json.NewDecoder(bytes.NewReader(raw))
337: 	d.DisallowUnknownFields()
338: 	if err := d.Decode(v); err != nil {
339: 		return err
340: 	}
341: 	var rest any
342: 	if err := d.Decode(&rest); err != io.EOF {
343: 		return errors.New("trailing JSON data")
344: 	}
345: 	return nil
346: }

348: func rejectDuplicateJSON(raw []byte) error {
349: 	d := json.NewDecoder(bytes.NewReader(raw))
350: 	var walk func(int) error
351: 	walk = func(depth int) error {
352: 		if depth > 64 {
353: 			return errors.New("JSON nesting too deep")
354: 		}
355: 		tok, err := d.Token()
356: 		if err != nil {
357: 			return err
358: 		}
359: 		opening, ok := tok.(json.Delim)
360: 		if !ok {
361: 			return nil
362: 		}
363: 		switch opening {
364: 		case '{':
365: 			seen := make(map[string]struct{})
366: 			for d.More() {
367: 				keyToken, err := d.Token()
368: 				if err != nil {
369: 					return err
370: 				}
371: 				key, ok := keyToken.(string)
372: 				if !ok {
373: 					return errors.New("JSON object key expected")
374: 				}
375: 				if _, exists := seen[key]; exists {
376: 					return errors.New("duplicate JSON key: " + key)
377: 				}
378: 				seen[key] = struct{}{}
379: 				if err := walk(depth + 1); err != nil {
380: 					return err
381: 				}
382: 			}
383: 		case '[':
384: 			for d.More() {
385: 				if err := walk(depth + 1); err != nil {
386: 					return err
387: 				}
388: 			}
389: 		default:
390: 			return errors.New("unexpected JSON delimiter")
391: 		}
392: 		_, err = d.Token()
393: 		return err
394: 	}
395: 	if err := walk(0); err != nil {
396: 		return err
397: 	}
398: 	if _, err := d.Token(); err != io.EOF {
399: 		return errors.New("trailing JSON data")
400: 	}
401: 	return nil
402: }

411: func readInput(path string) (Input, string, error) {
412: 	var in Input
413: 	raw, e := os.ReadFile(path)
414: 	if e != nil {
