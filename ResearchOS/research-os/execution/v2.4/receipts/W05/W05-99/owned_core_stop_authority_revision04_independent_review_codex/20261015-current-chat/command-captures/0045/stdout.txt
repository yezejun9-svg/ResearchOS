19: func parseCommand(args []string) (string, map[string]string, error) {
20: 	if len(args) == 0 {
21: 		return "", nil, errors.New("one action required")
22: 	}
23: 	action := args[0]
24: 	var required []string
25: 	switch action {
26: 	case "preflight":
27: 		required = []string{"--input", "--operation-id", "--out"}
28: 	case "commit":
29: 		required = []string{"--plan", "--expect-plan-sha256"}
30: 	case "inspect":
31: 		required = []string{"--plan"}
32: 	default:
33: 		return "", nil, errors.New("unknown action")
34: 	}
35: 	if len(args) != 1+2*len(required) {
36: 		return "", nil, errors.New("exact action flags required")
37: 	}
38: 	allowed := make(map[string]bool, len(required))
39: 	for _, k := range required {
40: 		allowed[k] = true
41: 	}
42: 	parsed := make(map[string]string, len(required))
43: 	for i := 1; i < len(args); i += 2 {
44: 		key, value := args[i], args[i+1]
45: 		_, seen := parsed[key]
46: 		if !allowed[key] || seen || value == "" || strings.HasPrefix(value, "--") || strings.ContainsRune(value, '\x00') {
47: 			return "", nil, errors.New("mixed, duplicate, or malformed flags")
48: 		}
49: 		parsed[key] = value
50: 	}
51: 	if len(parsed) != len(required) {
52: 		return "", nil, errors.New("missing action flag")
53: 	}
