39: func TestOwnedCoreStopPostflightRealLeaseBlocksLogicalDrift(t *testing.T) {
40: 	fixture := os.Getenv("RESEARCHOS_OWNED_STOP_FIXTURE_EXE")
41: 	if fixture == "" {
42: 		t.Fatal("new owned fixture exe required")
43: 	}
44: 	base := t.TempDir()
45: 	root := filepath.Join(base, "data")
46: 	if err := os.Mkdir(root, 0700); err != nil {
47: 		t.Fatal(err)
48: 	}
49: 	dbPath := filepath.Join(root, "researchos.sqlite")
50: 	db, err := sql.Open("sqlite", dbPath)
51: 	if err != nil {
52: 		t.Fatal(err)
53: 	}
54: 	if _, err = db.Exec(`PRAGMA journal_mode=WAL;CREATE TABLE stop_fixture_rows(id TEXT PRIMARY KEY,payload BLOB)`); err != nil {
55: 		db.Close()
56: 		t.Fatal(err)
57: 	}
58: 	if err = db.Close(); err != nil {
59: 		t.Fatal(err)
60: 	}
61: 	markers := filepath.Join(base, "markers")
62: 	if err = os.Mkdir(markers, 0700); err != nil {
63: 		t.Fatal(err)
64: 	}
65: 	ready := filepath.Join(markers, "ready")
66: 	cmd := exec.Command(fixture, "--root", root, "--ready", ready, "--attempt", filepath.Join(markers, "never-attempt"), "--attempted", filepath.Join(markers, "never-attempted"))
67: 	if err = cmd.Start(); err != nil {
68: 		t.Fatal(err)
69: 	}
70: 	defer func() { _ = cmd.Process.Kill(); _ = cmd.Wait() }()
71: 	found := false
72: 	for deadline := time.Now().Add(6 * time.Second); time.Now().Before(deadline); time.Sleep(20 * time.Millisecond) {
73: 		if _, e := os.Stat(ready); e == nil {
74: 			found = true
75: 			break
76: 		}
77: 	}
78: 	if !found {
79: 		t.Fatal("fixture ready missing")
80: 	}
81: 	baseline, err := store.ReadOwnedStopCensus(context.Background(), dbPath)
82: 	if err != nil || baseline.Rows != 1 {
83: 		t.Fatalf("committed baseline: %+v %v", baseline, err)
84: 	}
85: 	id, err := supervisor.ObserveOwnedCoreReadOnly(uint32(cmd.Process.Pid))
86: 	if err != nil {
87: 		t.Fatal(err)
88: 	}
89: 	h, err := supervisor.OpenOwnedCoreRetained(id)
90: 	if err != nil {
91: 		t.Fatal(err)
92: 	}
93: 	defer h.Close()
94: 	exit, err := h.TerminateOnceAndWait(5 * time.Second)
95: 	if err != nil || !exit.ExitObserved || exit.TerminateCalls != 1 {
96: 		t.Fatalf("native fixture exit: %+v %v", exit, err)
97: 	}
98: 	db, err = sql.Open("sqlite", dbPath)
99: 	if err != nil {
100: 		t.Fatal(err)
101: 	}
102: 	if _, err = db.Exec(`INSERT INTO stop_fixture_rows VALUES('poststop-drift',x'ff')`); err != nil {
103: 		db.Close()
104: 		t.Fatal(err)
105: 	}
106: 	if err = db.Close(); err != nil {
107: 		t.Fatal(err)
108: 	}
109: 	out := filepath.Join(base, "postflight")
110: 	if err = createPrivateOutput(out, root); err != nil {
111: 		t.Fatal(err)
112: 	}
113: 	plan := Plan{OperationID: strings.Repeat("c", 32), OutputDir: out, Input: Input{DataRoot: root}, Baseline: baseline}
114: 	plan.Host.Database = dbPath
115: 	result, err := finishAfterNativeExit(context.Background(), plan, strings.Repeat("d", 64), exit)
116: 	if err != nil {
117: 		t.Fatal(err)
118: 	}
119: 	if result.State != "STOP_OBSERVED_MAINTENANCE_BLOCKED" || result.Postflight == nil || result.Postflight.Rows != 2 || *result.Postflight == baseline {
120: 		t.Fatalf("poststop logical drift escaped maintenance block: %+v", result)
121: 	}
122: 	if _, err := os.Stat(filepath.Join(out, "POSTFLIGHT.json")); err != nil {
123: 		t.Fatalf("durable blocked receipt missing: %v", err)
124: 	}
125: }

148: func TestCommitSignalBoundaryHonorsCallerCancellationAndEarlierDeadline(t *testing.T) {
149: 	now := time.Now()
150: 	g := operationGrant{AllowedActions: []string{"commit"}, notBefore: now.Add(-time.Minute), expires: now.Add(time.Minute)}
151: 	ctx, cancel := context.WithCancel(context.Background())
152: 	cancel()
153: 	if err := checkCommitSignalBoundary(ctx, g); err != context.Canceled {
154: 		t.Fatalf("canceled caller reached intent: %v", err)
155: 	}
156: 	ctx, cancel = context.WithDeadline(context.Background(), now.Add(-time.Second))
157: 	defer cancel()
158: 	if err := checkCommitSignalBoundary(ctx, g); err != context.DeadlineExceeded {
159: 		t.Fatalf("earlier caller deadline reached intent: %v", err)
160: 	}
161: 	if err := checkCommitSignalBoundary(context.Background(), g); err != nil {
162: 		t.Fatalf("valid caller/grant denied: %v", err)
163: 	}
164: }
