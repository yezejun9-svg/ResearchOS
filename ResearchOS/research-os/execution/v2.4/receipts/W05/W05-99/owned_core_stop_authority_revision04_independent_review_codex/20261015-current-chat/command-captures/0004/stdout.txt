69: 	user, err := windows.GetCurrentProcessToken().GetTokenUser()

81: func imageSecurityAuthorized(sd *windows.SECURITY_DESCRIPTOR, directory bool) error {
82: 	owner, _, err := sd.Owner()
83: 	if err != nil || owner == nil || !owner.IsValid() {
84: 		return errors.New("image owner unavailable")
85: 	}
86: 	trusted, err := trustedImageSID(owner.String())
87: 	if err != nil || !trusted {
88: 		return errors.New("image owner is outside trusted operator/OS/install boundary")
89: 	}
90: 	acl, _, err := sd.DACL()
91: 	if err != nil || acl == nil || acl.AceCount == 0 {
92: 		return errors.New("image DACL unavailable")
93: 	}
94: 	unsafeRights := uint32(windows.FILE_WRITE_DATA | windows.FILE_APPEND_DATA | windows.FILE_WRITE_EA | windows.FILE_WRITE_ATTRIBUTES | windows.DELETE | windows.WRITE_DAC | windows.WRITE_OWNER | windows.GENERIC_WRITE | windows.GENERIC_ALL | windows.MAXIMUM_ALLOWED | windows.ACCESS_SYSTEM_SECURITY)
95: 	if directory {
96: 		unsafeRights |= 0x40 // FILE_DELETE_CHILD on a directory.
97: 	}
98: 	for i := uint32(0); i < uint32(acl.AceCount); i++ {
99: 		var ace *windows.ACCESS_ALLOWED_ACE
100: 		if err := windows.GetAce(acl, i, &ace); err != nil || ace == nil || ace.Header.AceSize < 16 {
101: 			return errors.New("image ACE malformed")
102: 		}
103: 		if ace.Header.AceType != windows.ACCESS_ALLOWED_ACE_TYPE && ace.Header.AceType != windows.ACCESS_DENIED_ACE_TYPE {
104: 			return errors.New("image DACL contains unsupported ACE type")
105: 		}
106: 		sid := (*windows.SID)(unsafe.Pointer(&ace.SidStart))
107: 		if !sid.IsValid() || 8+sid.Len() > int(ace.Header.AceSize) {
108: 			return errors.New("image ACE SID malformed")
109: 		}
110: 		if ace.Header.AceType == windows.ACCESS_DENIED_ACE_TYPE || ace.Header.AceFlags&windows.INHERIT_ONLY_ACE != 0 {
111: 			continue
112: 		}
113: 		trusted, err := trustedImageSID(sid.String())
114: 		if err != nil {
115: 			return err
116: 		}
117: 		unsafeGranted := uint32(ace.Mask) & unsafeRights
118: 		// FILE_ADD_SUBDIRECTORY alone can create a sibling at an ancestor.
119: 		// It cannot replace the selected child while every selected component
120: 		// is retained without FILE_SHARE_DELETE. C: grants this isolated right
121: 		// to Authenticated Users on common Windows installations.
122: 		if directory && unsafeGranted == windows.FILE_APPEND_DATA {
123: 			unsafeGranted = 0
124: 		}
125: 		if !trusted && unsafeGranted != 0 {
126: 			return errors.New("image DACL grants write or control to untrusted trustee")
127: 		}
128: 	}
129: 	return nil
130: }

158: 	buf := make([]uint16, 32768)

170: func observeOwnedImageAnchor(h windows.Handle, directory bool) (ownedImageAnchor, windows.ByHandleFileInformation, error) {
171: 	var info windows.ByHandleFileInformation
172: 	if err := windows.GetFileInformationByHandle(h, &info); err != nil {
173: 		return ownedImageAnchor{}, info, err
174: 	}
175: 	if info.FileAttributes&windows.FILE_ATTRIBUTE_REPARSE_POINT != 0 || (info.FileAttributes&windows.FILE_ATTRIBUTE_DIRECTORY != 0) != directory || info.VolumeSerialNumber == 0 || uint64(info.FileIndexHigh)<<32|uint64(info.FileIndexLow) == 0 || (!directory && info.NumberOfLinks != 1) {
176: 		return ownedImageAnchor{}, info, errors.New("image ancestor reparse/alias/native identity rejected")
177: 	}
178: 	sd, err := windows.GetSecurityInfo(h, windows.SE_FILE_OBJECT, windows.OWNER_SECURITY_INFORMATION|windows.DACL_SECURITY_INFORMATION)
179: 	if err != nil || sd == nil || !sd.IsValid() {
180: 		return ownedImageAnchor{}, info, errors.New("image native security descriptor unavailable")
181: 	}
182: 	if err := imageSecurityAuthorized(sd, directory); err != nil {
183: 		return ownedImageAnchor{}, info, err
184: 	}
185: 	security := sd.String()
186: 	if security == "" {
187: 		return ownedImageAnchor{}, info, errors.New("image ancestor owner/DACL serialization unavailable")
188: 	}
189: 	return ownedImageAnchor{handle: h, volume: info.VolumeSerialNumber, index: uint64(info.FileIndexHigh)<<32 | uint64(info.FileIndexLow), security: security}, info, nil
190: }

192: func ownedNativeImage(path string) (_ *ownedImageFile, err error) {
193: 	image := &ownedImageFile{}
194: 	defer func() {
195: 		if err != nil {
196: 			image.Close()
197: 		}
198: 	}()
199: 	volume := filepath.VolumeName(path)
200: 	current := volume + string(filepath.Separator)
201: 	parts := strings.Split(strings.TrimPrefix(path[len(volume):], string(filepath.Separator)), string(filepath.Separator))
202: 	paths := []string{current}
203: 	for _, part := range parts {
204: 		if part == "" || part == "." || part == ".." {
205: 			return nil, errors.New("noncanonical image path component")
206: 		}
207: 		current = filepath.Join(current, part)
208: 		paths = append(paths, current)
209: 	}
210: 	for i, name := range paths {
211: 		last := i == len(paths)-1
212: 		wide, e := windows.UTF16PtrFromString(name)
213: 		if e != nil {
214: 			return nil, e
215: 		}
216: 		access := uint32(windows.FILE_READ_ATTRIBUTES | windows.READ_CONTROL)
217: 		share := uint32(windows.FILE_SHARE_READ)
218: 		if last {
219: 			access = windows.GENERIC_READ | windows.READ_CONTROL
220: 			share = windows.FILE_SHARE_READ
221: 		}
222: 		h, e := windows.CreateFile(wide, access, share, nil, windows.OPEN_EXISTING, windows.FILE_FLAG_OPEN_REPARSE_POINT|windows.FILE_FLAG_BACKUP_SEMANTICS, 0)
223: 		if e != nil {
224: 			return nil, e
225: 		}
226: 		anchor, info, e := observeOwnedImageAnchor(h, !last)
227: 		if e != nil {
228: 			_ = windows.CloseHandle(h)
229: 			return nil, fmt.Errorf("image component %q: %w", name, e)
230: 		}
231: 		if last {
232: 			if info.FileSizeHigh != 0 || info.FileSizeLow == 0 || info.FileSizeLow > 256<<20 {
233: 				_ = windows.CloseHandle(h)
234: 				return nil, errors.New("bounded image size required")
235: 			}
236: 			image.final = anchor
237: 			image.file = os.NewFile(uintptr(h), name)
238: 			hash := sha256.New()
239: 			var n int64
240: 			n, err = io.Copy(hash, io.LimitReader(image.file, 256<<20+1))
241: 			if err != nil {
242: 				return nil, err
243: 			}
244: 			var after windows.ByHandleFileInformation
245: 			if err = windows.GetFileInformationByHandle(h, &after); err != nil {
246: 				return nil, err
247: 			}
248: 			if n != int64(info.FileSizeLow) || info.VolumeSerialNumber != after.VolumeSerialNumber || info.FileIndexHigh != after.FileIndexHigh || info.FileIndexLow != after.FileIndexLow || info.FileSizeLow != after.FileSizeLow || info.LastWriteTime != after.LastWriteTime {
249: 				return nil, errors.New("image changed during retained read")
250: 			}
251: 			image.sha = hex.EncodeToString(hash.Sum(nil))
252: 		} else {
253: 			image.ancestors = append(image.ancestors, anchor)
254: 		}
255: 	}
256: 	return image, nil
257: }

259: func ownedNativeProcessFromHandle(h windows.Handle) (p ownedNativeProcess, err error) {
260: 	p = ownedNativeProcess{handle: h}
261: 	defer func() {
262: 		if err != nil {
263: 			p.close()
264: 		}
265: 	}()
266: 	var created, exited, kernel, user windows.Filetime
267: 	if err = windows.GetProcessTimes(h, &created, &exited, &kernel, &user); err != nil {
268: 		return p, err
269: 	}
270: 	p.created = created.Nanoseconds()
271: 	var token windows.Token
272: 	if err = windows.OpenProcessToken(h, windows.TOKEN_QUERY, &token); err != nil {
273: 		return p, err
274: 	}
275: 	tu, e := token.GetTokenUser()
276: 	_ = token.Close()
277: 	if e != nil || tu.User.Sid == nil || !tu.User.Sid.IsValid() {
278: 		return p, errors.New("process SID unavailable")
279: 	}
280: 	p.sid = tu.User.Sid.String()
281: 	if p.path, err = ownedNativePath(h); err != nil {
282: 		return p, err
283: 	}
284: 	p.image, err = ownedNativeImage(p.path)
285: 	if err == nil {
286: 		p.sha, p.volume, p.index = p.image.sha, p.image.final.volume, p.image.final.index
287: 	}
288: 	return p, err
289: }

291: func ownedNativeParentPID(pid uint32) (uint32, error) {
292: 	snap, err := windows.CreateToolhelp32Snapshot(windows.TH32CS_SNAPPROCESS, 0)
293: 	if err != nil {
294: 		return 0, err
295: 	}
296: 	defer windows.CloseHandle(snap)
297: 	entry := windows.ProcessEntry32{Size: uint32(unsafe.Sizeof(windows.ProcessEntry32{}))}
298: 	if err = windows.Process32First(snap, &entry); err != nil {
299: 		return 0, err
300: 	}
301: 	for {
302: 		if entry.ProcessID == pid {
303: 			if entry.ParentProcessID == 0 || entry.ParentProcessID == pid {
304: 				return 0, errors.New("invalid parent identity")
305: 			}
306: 			return entry.ParentProcessID, nil
307: 		}
308: 		err = windows.Process32Next(snap, &entry)
309: 		if errors.Is(err, windows.ERROR_NO_MORE_FILES) {
310: 			break
311: 		}
312: 		if err != nil {
313: 			return 0, err
314: 		}
315: 	}
316: 	return 0, errors.New("process absent from snapshot")
317: }

348: func ownedCoreOpen(pid uint32, access uint32) (out *OwnedCoreRetainedHandle, err error) {
349: 	if pid == 0 {
350: 		return nil, errors.New("zero PID")
351: 	}
352: 	h, err := windows.OpenProcess(access|windows.PROCESS_QUERY_LIMITED_INFORMATION|windows.SYNCHRONIZE, false, pid)
353: 	if err != nil {
354: 		return nil, err
355: 	}
356: 	out = &OwnedCoreRetainedHandle{}
357: 	defer func() {
358: 		if err != nil {
359: 			out.Close()
360: 			out = nil
361: 		}
362: 	}()
363: 	out.core, err = ownedNativeProcessFromHandle(h)
364: 	if err != nil {
365: 		return nil, err
366: 	}
367: 	actualPID, e := windows.GetProcessId(out.core.handle)
368: 	if e != nil || actualPID != pid {
369: 		return nil, errors.New("process handle PID mismatch")
370: 	}
371: 	parentPID, e := ownedNativeParentPID(pid)
372: 	if e != nil {
373: 		return nil, e
374: 	}
375: 	parentH, e := windows.OpenProcess(windows.PROCESS_QUERY_LIMITED_INFORMATION|windows.SYNCHRONIZE, false, parentPID)
376: 	if e != nil {
377: 		return nil, e
378: 	}
379: 	out.parent, err = ownedNativeProcessFromHandle(parentH)
380: 	if err != nil {
381: 		return nil, err
382: 	}
383: 	actualParent, e := windows.GetProcessId(out.parent.handle)
384: 	if e != nil || actualParent != parentPID {
385: 		return nil, errors.New("parent handle PID mismatch")
386: 	}
387: 	out.identity = OwnedCoreNativeIdentity{PID: pid, CreatedNS: out.core.created, SID: out.core.sid, ImagePath: out.core.path, ImageSHA256: out.core.sha, ImageVolume: out.core.volume, ImageIndex: out.core.index, ParentPID: parentPID, ParentCreatedNS: out.parent.created, ParentSID: out.parent.sid, ParentImagePath: out.parent.path, ParentImageSHA256: out.parent.sha, ParentImageVolume: out.parent.volume, ParentImageIndex: out.parent.index}
388: 	return out, nil
389: }

426: func verifyOwnedImageFile(image *ownedImageFile, volume uint32, index uint64) error {
427: 	if image == nil || image.file == nil || image.final.volume != volume || image.final.index != index {
428: 		return errors.New("retained image identity absent")
429: 	}
430: 	for _, original := range image.ancestors {
431: 		current, _, err := observeOwnedImageAnchor(original.handle, true)
432: 		if err != nil || current.volume != original.volume || current.index != original.index || current.security != original.security {
433: 			return errors.New("retained image ancestor identity/ACL changed")
434: 		}
435: 	}
436: 	current, _, err := observeOwnedImageAnchor(windows.Handle(image.file.Fd()), false)
437: 	if err != nil || current.volume != image.final.volume || current.index != image.final.index || current.security != image.final.security {
438: 		return errors.New("retained image file identity/ACL changed")
439: 	}
440: 	return nil
441: }

443: func (h *OwnedCoreRetainedHandle) verifyRetained() error {
444: 	if h == nil || h.closed || h.core.handle == 0 {
445: 		return errors.New("native handle closed")
446: 	}
447: 	pid, err := windows.GetProcessId(h.core.handle)
448: 	if err != nil || pid != h.identity.PID {
449: 		return errors.New("retained PID changed")
450: 	}
451: 	var created, exited, kernel, user windows.Filetime
452: 	if err = windows.GetProcessTimes(h.core.handle, &created, &exited, &kernel, &user); err != nil || created.Nanoseconds() != h.identity.CreatedNS {
453: 		return errors.New("retained process creation changed")
454: 	}
455: 	// Every path component remains open without delete sharing. The final image
456: 	// also denies write sharing; identity and security are rechecked by handle.
457: 	if err = verifyOwnedImageFile(h.core.image, h.identity.ImageVolume, h.identity.ImageIndex); err != nil {
458: 		return err
459: 	}
460: 	parentPID, err := windows.GetProcessId(h.parent.handle)
461: 	if err != nil || parentPID != h.identity.ParentPID {
462: 		return errors.New("retained parent PID changed")
463: 	}
464: 	if err = windows.GetProcessTimes(h.parent.handle, &created, &exited, &kernel, &user); err != nil || created.Nanoseconds() != h.identity.ParentCreatedNS {
465: 		return errors.New("retained parent creation changed")
466: 	}
467: 	parentStatus, err := h.nativeParentWait()
468: 	if err != nil || parentStatus != uint32(windows.WAIT_TIMEOUT) {
469: 		return errors.New("retained parent no longer live")
470: 	}
471: 	if err = verifyOwnedImageFile(h.parent.image, h.identity.ParentImageVolume, h.identity.ParentImageIndex); err != nil {
472: 		return err
473: 	}
474: 	return nil
475: }

492: func (h *OwnedCoreRetainedHandle) TerminateOnceAndWait(timeout time.Duration) (OwnedCoreNativeExit, error) {
493: 	return h.TerminateOnceAndWaitUntil(timeout, time.Now().Add(time.Hour), time.Now().Add(time.Hour))

495: 
496: // TerminateOnceAndWaitUntil applies both the invocation monotonic deadline and
497: // the grant's wall expiry at the last native boundary, after pre-signal waits.
498: // A failed deadline check returns with zero TerminateProcess calls; the caller's
499: // already-durable intent remains consumed.
500: func (h *OwnedCoreRetainedHandle) TerminateOnceAndWaitUntil(timeout time.Duration, invocationDeadline, wallExpiry time.Time) (OwnedCoreNativeExit, error) {
501: 	return h.TerminateOnceAndWaitUntilContext(context.Background(), timeout, invocationDeadline, wallExpiry)
502: }
503: 
504: func (h *OwnedCoreRetainedHandle) TerminateOnceAndWaitUntilContext(ctx context.Context, timeout time.Duration, invocationDeadline, wallExpiry time.Time) (OwnedCoreNativeExit, error) {
505: 	var out OwnedCoreNativeExit
506: 	if ctx == nil {
507: 		return out, errors.New("caller context required")
508: 	}
509: 	if timeout <= 0 || timeout > 30*time.Second {
510: 		return out, errors.New("bounded native wait required")
511: 	}
512: 	if earlier, ok := ctx.Deadline(); ok && earlier.Before(invocationDeadline) {
513: 		invocationDeadline = earlier
514: 	}
515: 	if err := ctx.Err(); err != nil {
