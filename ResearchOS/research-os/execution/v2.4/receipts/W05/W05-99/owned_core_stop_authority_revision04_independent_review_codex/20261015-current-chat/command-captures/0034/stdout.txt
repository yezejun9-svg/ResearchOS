45: // Every component is opened without FILE_SHARE_DELETE and retained. The final
46: // file additionally denies write sharing; its bytes are read and decoded from
47: // this one handle while the handles remain live through the operation.
48: func openHeldFile(path string, directory bool) (out *heldFile, err error) {
49: 	if err = cleanOwnedLocalPath(path); err != nil {
50: 		return nil, err
51: 	}
52: 	out = &heldFile{}
53: 	defer func() {
54: 		if err != nil {
55: 			out.Close()
56: 			out = nil
57: 		}
58: 	}()
59: 	volume := filepath.VolumeName(path)
60: 	parts := strings.Split(strings.TrimPrefix(path[len(volume):], string(filepath.Separator)), string(filepath.Separator))
61: 	current := volume + string(filepath.Separator)
62: 	paths := []string{current}
63: 	if path != current {
64: 		for _, part := range parts {
65: 			if part == "" || part == "." || part == ".." {
66: 				return nil, errors.New("noncanonical path component")
67: 			}
68: 			current = filepath.Join(current, part)
69: 			paths = append(paths, current)
70: 		}
71: 	}
72: 	for i, p := range paths {
73: 		last := i == len(paths)-1
74: 		wide, e := windows.UTF16PtrFromString(p)
75: 		if e != nil {
76: 			err = e
77: 			return
78: 		}
79: 		access := uint32(windows.FILE_READ_ATTRIBUTES | windows.READ_CONTROL)
80: 		share := uint32(windows.FILE_SHARE_READ | windows.FILE_SHARE_WRITE)
81: 		flags := uint32(windows.FILE_FLAG_OPEN_REPARSE_POINT | windows.FILE_FLAG_BACKUP_SEMANTICS)
82: 		if last && !directory {
83: 			access = windows.GENERIC_READ | windows.READ_CONTROL
84: 			share = windows.FILE_SHARE_READ
85: 		}
86: 		if last && directory {
87: 			share = windows.FILE_SHARE_READ
88: 		}
89: 		h, e := windows.CreateFile(wide, access, share, nil, windows.OPEN_EXISTING, flags, 0)
90: 		if e != nil {
91: 			err = e
92: 			return
93: 		}
94: 		var info windows.ByHandleFileInformation
95: 		if e = windows.GetFileInformationByHandle(h, &info); e != nil {
96: 			_ = windows.CloseHandle(h)
97: 			err = e
98: 			return
99: 		}
100: 		expectDir := !last || directory
101: 		if info.FileAttributes&windows.FILE_ATTRIBUTE_REPARSE_POINT != 0 || (info.FileAttributes&windows.FILE_ATTRIBUTE_DIRECTORY != 0) != expectDir || info.VolumeSerialNumber == 0 || uint64(info.FileIndexHigh)<<32|uint64(info.FileIndexLow) == 0 || (!expectDir && info.NumberOfLinks != 1) {
102: 			_ = windows.CloseHandle(h)
103: 			err = errors.New("reparse, alias, or native identity invalid")
104: 			return
105: 		}
106: 		if last {
107: 			sd, e := windows.GetSecurityInfo(h, windows.SE_FILE_OBJECT, windows.OWNER_SECURITY_INFORMATION)
108: 			if e != nil {
109: 				_ = windows.CloseHandle(h)
110: 				err = e
111: 				return
112: 			}
113: 			owner, _, e := sd.Owner()
114: 			sid, se := currentSID()
115: 			if e != nil || se != nil || owner == nil || owner.String() != sid.String() {
116: 				_ = windows.CloseHandle(h)
117: 				err = errors.New("file owner SID differs")
118: 				return
119: 			}
120: 			out.file = os.NewFile(uintptr(h), p)
121: 			out.info = info
122: 		} else {
123: 			out.ancestors = append(out.ancestors, h)
124: 		}
125: 	}
126: 	return out, nil
127: }

129: func readHeldBytes(h *heldFile, max int) ([]byte, error) {
130: 	if h == nil || h.file == nil || h.info.FileSizeHigh != 0 || h.info.FileSizeLow == 0 || h.info.FileSizeLow > uint32(max) {
131: 		return nil, errors.New("bounded held file required")
132: 	}
133: 	raw, err := io.ReadAll(io.LimitReader(h.file, int64(max)+1))
134: 	if err != nil || len(raw) != int(h.info.FileSizeLow) {
135: 		return nil, errors.New("held file read changed")
136: 	}
137: 	var after windows.ByHandleFileInformation
138: 	if err = windows.GetFileInformationByHandle(windows.Handle(h.file.Fd()), &after); err != nil || nativeFileIdentity(after, h.file.Name()) != nativeFileIdentity(h.info, h.file.Name()) || after.FileSizeHigh != h.info.FileSizeHigh || after.FileSizeLow != h.info.FileSizeLow || after.LastWriteTime != h.info.LastWriteTime {
139: 		return nil, errors.New("held file identity/content changed")
140: 	}
141: 	return raw, nil
142: }

195: func acquireOperationLock(g operationGrant, create bool) (*operationLock, error) {
196: 	if err := outsideDataRoot(g.OperationDir.Path, g.DataRoot.Path); err != nil {
197: 		return nil, err
198: 	}
199: 	dir, err := openHeldFile(g.OperationDir.Path, true)
200: 	if err != nil {
201: 		return nil, err
202: 	}
203: 	if nativeFileIdentity(dir.info, g.OperationDir.Path) != g.OperationDir {
204: 		dir.Close()
205: 		return nil, errors.New("pinned operation directory identity differs")
206: 	}
207: 	if err = verifyOwnedPrivate(g.OperationDir.Path, true); err != nil {
208: 		dir.Close()
209: 		return nil, err
210: 	}
211: 	lockPath := filepath.Join(g.OperationDir.Path, ".operation.lock")
212: 	wide, err := windows.UTF16PtrFromString(lockPath)
213: 	if err != nil {
214: 		dir.Close()
215: 		return nil, err
216: 	}
217: 	mode := uint32(windows.OPEN_EXISTING)
218: 	newLock := false
219: 	if create {
220: 		mode = windows.CREATE_NEW
221: 	}
222: 	h, err := windows.CreateFile(wide, windows.GENERIC_READ|windows.GENERIC_WRITE, 0, nil, mode, windows.FILE_FLAG_OPEN_REPARSE_POINT, 0)
223: 	if create && (errors.Is(err, windows.ERROR_FILE_EXISTS) || errors.Is(err, windows.ERROR_ALREADY_EXISTS)) {
224: 		h, err = windows.CreateFile(wide, windows.GENERIC_READ|windows.GENERIC_WRITE, 0, nil, windows.OPEN_EXISTING, windows.FILE_FLAG_OPEN_REPARSE_POINT, 0)
225: 	} else if create && err == nil {
226: 		newLock = true
227: 	}
228: 	if err != nil {
229: 		dir.Close()
230: 		return nil, fmt.Errorf("operation already locked or lock absent: %w", err)
231: 	}
232: 	l := &operationLock{dir: dir, file: os.NewFile(uintptr(h), lockPath)}
233: 	var info windows.ByHandleFileInformation
234: 	if err = windows.GetFileInformationByHandle(h, &info); err != nil || info.FileAttributes&(windows.FILE_ATTRIBUTE_REPARSE_POINT|windows.FILE_ATTRIBUTE_DIRECTORY) != 0 || info.NumberOfLinks != 1 {
235: 		l.Close()
236: 		return nil, errors.New("operation lock reparse/alias rejected")
237: 	}
238: 	if newLock {
239: 		err = privateOwnedPath(lockPath, false)
240: 	}
241: 	if err != nil {
242: 		l.Close()
243: 		return nil, err
244: 	}
245: 	if err = verifyOwnedPrivate(lockPath, false); err != nil {
246: 		l.Close()
247: 		return nil, err
248: 	}
249: 	return l, nil
250: }

306: func currentExecutableSHA256() (string, error) {
307: 	path, err := os.Executable()
308: 	if err != nil {
309: 		return "", err
310: 	}
311: 	h, err := openHeldFile(path, false)
312: 	if err != nil {
313: 		return "", err
314: 	}
315: 	defer h.Close()
316: 	if h.info.FileSizeHigh != 0 || h.info.FileSizeLow == 0 || h.info.FileSizeLow > 256<<20 {
317: 		return "", errors.New("bounded executable required")
318: 	}
319: 	stream := sha256.New()
320: 	n, err := io.Copy(stream, io.LimitReader(h.file, 256<<20+1))
321: 	if err != nil || n != int64(h.info.FileSizeLow) {
322: 		return "", errors.New("executable read differs")
323: 	}
324: 	return hex.EncodeToString(stream.Sum(nil)), nil
325: }

500: func privateOwnedPath(path string, dir bool) error {
501: 	sid, err := currentSID()
502: 	if err != nil {
503: 		return err
504: 	}
505: 	ace := "(A;;FA;;;" + sid.String() + ")"
506: 	if dir {

516: 	}

520: func verifyOwnedPrivate(path string, dir bool) error {
521: 	sid, err := currentSID()
522: 	if err != nil {
523: 		return err
524: 	}
525: 	sd, err := windows.GetNamedSecurityInfo(path, windows.SE_FILE_OBJECT, windows.OWNER_SECURITY_INFORMATION|windows.DACL_SECURITY_INFORMATION)
526: 	if err != nil {
527: 		return err
528: 	}
529: 	owner, _, err := sd.Owner()
530: 	if err != nil || owner == nil || owner.String() != sid.String() {
531: 		return errors.New("private path owner SID differs")
532: 	}
533: 	control, _, err := sd.Control()
534: 	if err != nil || control&windows.SE_DACL_PROTECTED == 0 {
535: 		return errors.New("protected private DACL missing")
536: 	}
537: 	acl, _, err := sd.DACL()
538: 	if err != nil || acl == nil || acl.AceCount != 1 {
539: 		return errors.New("single private ACE required")
540: 	}
541: 	var ace *windows.ACCESS_ALLOWED_ACE
542: 	if err = windows.GetAce(acl, 0, &ace); err != nil {
543: 		return err
544: 	}
545: 	flags := byte(0)
546: 	if dir {
547: 		flags = windows.OBJECT_INHERIT_ACE | windows.CONTAINER_INHERIT_ACE
548: 	}
549: 	if ace.Header.AceType != windows.ACCESS_ALLOWED_ACE_TYPE || ace.Header.AceFlags != flags || ace.Mask != 0x001f01ff || (*windows.SID)(unsafe.Pointer(&ace.SidStart)).String() != sid.String() {
550: 		return errors.New("private ACE differs")
551: 	}
552: 	return nil
553: }

594: func writeOwnedJournal(out, name string, body any) (string, error) {
595: 	if filepath.Base(name) != name || strings.ContainsAny(name, "/\\:") {
596: 		return "", errors.New("fixed journal filename required")
597: 	}
598: 	if err := verifyOwnedPrivate(out, true); err != nil {
599: 		return "", err
600: 	}
601: 	raw, err := canonicalJSON(body)
602: 	if err != nil {
603: 		return "", err
604: 	}
605: 	path := filepath.Join(out, name)
606: 	f, err := os.OpenFile(path, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
607: 	if err != nil {
608: 		return "", err
609: 	}
610: 	defer f.Close()
611: 	if err = privateOwnedPath(path, false); err != nil {
612: 		return "", err
613: 	}
614: 	if err = verifyOwnedPrivate(path, false); err != nil {
615: 		return "", err
616: 	}
617: 	if _, err = f.Write(raw); err != nil {
618: 		return "", err
619: 	}
620: 	if err = f.Sync(); err != nil {
621: 		return "", err
622: 	}
623: 	return digest(raw), nil
624: }

648: func readPlan(path string) (Plan, string, error) {
649: 	var plan Plan
650: 	if filepath.Base(path) != "plan.json" {
651: 		return plan, "", errors.New("canonical plan.json required")
652: 	}
653: 	raw, err := readOwnedJournal(filepath.Dir(path), "plan.json", 256<<10)
654: 	if err != nil {
655: 		return plan, "", err
656: 	}
657: 	if err = strictJSON(raw, &plan); err != nil {
658: 		return plan, "", err
659: 	}
660: 	if err = validatePlan(plan); err != nil {
661: 		return plan, "", err
662: 	}
663: 	if plan.OutputDir != filepath.Dir(path) {
664: 		return plan, "", errors.New("plan directory changed")
665: 	}
666: 	canonical, err := canonicalJSON(plan)
667: 	if err != nil || string(canonical) != string(raw) {
668: 		return plan, "", errors.New("plan serialization differs")
669: 	}

673: // Inspect is observation-only. It never opens a termination handle or repairs
674: // a missing native wait. Completed POSTFLIGHT is returned unchanged.
675: func Inspect(planPath string) (Result, error) {
676: 	g, err := loadCompiledOperationGrant()
677: 	if err != nil {
678: 		return Result{}, err
679: 	}
680: 	if _, err = g.checkAction("inspect", false); err != nil {
681: 		return Result{}, err
682: 	}
683: 	if planPath != filepath.Join(g.OperationDir.Path, "plan.json") {
684: 		return Result{}, errors.New("inspect plan differs from compiled grant")
685: 	}
686: 	lock, err := acquireOperationLock(g, false)
687: 	if err != nil {
688: 		return Result{Schema: ResultSchema, OperationID: g.OperationID, State: "IN_PROGRESS", Reason: "operation writer active or immutable lock absent"}, nil
689: 	}
690: 	defer lock.Close()
691: 	docs := &heldDocuments{}
692: 	defer docs.Close()
693: 	c, m, err := readPreflightComplete(docs, g)
694: 	if err != nil {
695: 		return Result{Schema: ResultSchema, OperationID: g.OperationID, State: "STOP_UNAVAILABLE", Reason: "immutable preflight completion absent"}, nil
696: 	}
697: 	planGrant := g
698: 	planGrant.PlanSHA256 = c.PlanSHA256
699: 	p, sha, err := readPinnedPlan(docs, planGrant)
700: 	if err != nil {
701: 		return Result{}, err
702: 	}
703: 	if err = validatePreflightOutputs(docs, g, c, m, p); err != nil {
704: 		return Result{}, err
705: 	}
706: 	if g.GrantKind == "PREFLIGHT_ONLY" {
707: 		return Result{Schema: ResultSchema, OperationID: g.OperationID, PlanSHA256: sha, State: "PREFLIGHT_COMPLETE"}, nil
708: 	}
709: 	if _, err = os.Stat(filepath.Join(g.OperationDir.Path, "STOP_ATTEMPT.json")); errors.Is(err, os.ErrNotExist) {
710: 		return Result{Schema: ResultSchema, OperationID: g.OperationID, PlanSHA256: sha, State: "STOP_UNAVAILABLE", Reason: "no STOP attempt"}, nil
711: 	} else if err != nil {
712: 		return Result{}, err
713: 	}
714: 	return readExistingStopOutcome(docs, g, p, sha)
715: }
