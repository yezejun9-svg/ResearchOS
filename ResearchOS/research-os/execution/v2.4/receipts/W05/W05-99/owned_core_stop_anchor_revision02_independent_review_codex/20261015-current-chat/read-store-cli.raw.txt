// OwnedStopCensus is a hash of the exact sqlite_master rows and every user or
// SQLite-owned table row. Each cell has a type and length delimiter; BLOBs are
// hashed as bytes. A row count is included for review, not as a substitute.
type OwnedStopCensus struct {
	SHA256       string `json:"sha256"`
	SchemaSHA256 string `json:"schema_sha256"`
	Tables       int    `json:"tables"`
	Rows         int64  `json:"rows"`
	QuickCheck   string `json:"quick_check"`
}

type ownedStopQueryer interface {
	QueryContext(context.Context, string, ...any) (*sql.Rows, error)
	QueryRowContext(context.Context, string, ...any) *sql.Row
}

var ownedStopIdentifier = regexp.MustCompile(`^[A-Za-z_][A-Za-z0-9_]*$`)

func ownedStopURI(path, mode string) string {
	u := url.URL{Scheme: "file", Opaque: filepath.ToSlash(path)}
	q := u.Query()
	q.Set("mode", mode)
	q.Set("_busy_timeout", "1500")
	u.RawQuery = q.Encode()
	return u.String()
}

func ownedStopFrame(h hash.Hash, kind byte, raw []byte) {
	_, _ = h.Write([]byte{kind})
	var n [8]byte
	binary.BigEndian.PutUint64(n[:], uint64(len(raw)))
	_, _ = h.Write(n[:])
	_, _ = h.Write(raw)
}
func ownedStopCell(h hash.Hash, v any) error {
	switch x := v.(type) {
	case nil:
		ownedStopFrame(h, 'N', nil)
	case []byte:
		ownedStopFrame(h, 'B', x)
	case string:
		ownedStopFrame(h, 'S', []byte(x))
	case int64:
		ownedStopFrame(h, 'I', []byte(fmt.Sprintf("%d", x)))
	case float64:
		ownedStopFrame(h, 'F', []byte(fmt.Sprintf("%016x", math.Float64bits(x))))
	case bool:
		if x {
			ownedStopFrame(h, 'T', nil)
		} else {
			ownedStopFrame(h, 'f', nil)
		}
	case time.Time:
		ownedStopFrame(h, 'D', []byte(x.UTC().Format(time.RFC3339Nano)))
	default:
		return fmt.Errorf("unsupported SQLite cell type %T", v)
	}
	return nil
}

// Fixed statement forms are constructed only from identifiers returned by
// sqlite_master and PRAGMA table_info. The connection is never returned.
func ownedStopCensus(ctx context.Context, q ownedStopQueryer) (OwnedStopCensus, error) {
	var out OwnedStopCensus
	if err := q.QueryRowContext(ctx, "PRAGMA quick_check").Scan(&out.QuickCheck); err != nil || out.QuickCheck != "ok" {
		return out, fmt.Errorf("quick_check: %s: %w", out.QuickCheck, err)
	}
	schema := sha256.New()
	all := sha256.New()
	rows, err := q.QueryContext(ctx, "SELECT type,name,tbl_name,sql FROM sqlite_master ORDER BY type,name,tbl_name,sql")
	if err != nil {
		return out, err
	}
	tables := []string{}
	for rows.Next() {
		var typ, name, tbl string
		var sqlText sql.NullString
		if err = rows.Scan(&typ, &name, &tbl, &sqlText); err != nil {
			break
		}
		for _, s := range []string{typ, name, tbl} {
			ownedStopFrame(schema, 'S', []byte(s))
		}
		if sqlText.Valid {
			ownedStopFrame(schema, 'S', []byte(sqlText.String))
		} else {
			ownedStopFrame(schema, 'N', nil)
		}
		if typ == "table" {
			tables = append(tables, name)
		}
	}
	if err == nil {
		err = rows.Err()
	}
	rows.Close()
	if err != nil {
		return out, err
	}
	out.SchemaSHA256 = hex.EncodeToString(schema.Sum(nil))
	ownedStopFrame(all, 'H', schema.Sum(nil))
	for _, table := range tables {
		if !ownedStopIdentifier.MatchString(table) {
			return out, errors.New("noncanonical SQLite table identifier")
		}
		colRows, e := q.QueryContext(ctx, "PRAGMA table_info(\""+table+"\")")
		if e != nil {
			return out, e
		}
		type key struct {
			name  string
			order int
		}
		var keys []key
		for colRows.Next() {
			var cid, notnull, pk int
			var name, typ string
			var defaultValue sql.NullString
			if e = colRows.Scan(&cid, &name, &typ, &notnull, &defaultValue, &pk); e != nil {
				break
			}
			if !ownedStopIdentifier.MatchString(name) {
				e = errors.New("noncanonical SQLite column identifier")
				break
			}
			if pk > 0 {
				keys = append(keys, key{name, pk})
			}
		}
		if e == nil {
			e = colRows.Err()
		}
		colRows.Close()
		if e != nil {
			return out, e
		}
		order := "rowid"
		if len(keys) > 0 {
			sort.Slice(keys, func(i, j int) bool { return keys[i].order < keys[j].order })
			parts := make([]string, len(keys))
			for i, k := range keys {
				parts[i] = "\"" + k.name + "\""
			}
			order = strings.Join(parts, ",")
		}
		ownedStopFrame(all, 'T', []byte(table))
		data, e := q.QueryContext(ctx, "SELECT * FROM \""+table+"\" ORDER BY "+order)
		if e != nil {
			return out, e
		}
		columns, e := data.Columns()
		if e != nil {
			data.Close()
			return out, e
		}
		for data.Next() {
			values := make([]any, len(columns))
			ptrs := make([]any, len(columns))
			for i := range values {
				ptrs[i] = &values[i]
			}
			if e = data.Scan(ptrs...); e != nil {
				break
			}
			ownedStopFrame(all, 'R', nil)
			for _, v := range values {
				if e = ownedStopCell(all, v); e != nil {
					break
				}
			}
			if e != nil {
				break
			}
			out.Rows++
		}
		if e == nil {
			e = data.Err()
		}
		data.Close()
		if e != nil {
			return out, e
		}
		out.Tables++
	}
	out.SHA256 = hex.EncodeToString(all.Sum(nil))
	return out, nil
}

// OnlineOwnedStopBackup creates a new destination outside the source root and
// uses the pinned native online-backup API. It returns only hashes and counts.
func OnlineOwnedStopBackup(ctx context.Context, source, destination string) (OwnedStopCensus, string, error) {
	if !filepath.IsAbs(source) || !filepath.IsAbs(destination) || filepath.Clean(destination) != destination {
		return OwnedStopCensus{}, "", errors.New("canonical backup paths required")
	}
	rel, e := filepath.Rel(filepath.Dir(source), destination)
	if e != nil || (!strings.HasPrefix(rel, ".."+string(filepath.Separator)) && rel != "..") {
		return OwnedStopCensus{}, "", errors.New("backup destination must be outside data root")
	}
	if _, err := os.Stat(source); err != nil {
		return OwnedStopCensus{}, "", err
	}
	f, err := os.OpenFile(destination, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	if err = f.Close(); err != nil {
		return OwnedStopCensus{}, "", err
	}
	src, err := sql.Open("sqlite", ownedStopURI(source, "rw"))
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	defer src.Close()
	src.SetMaxOpenConns(1)
	conn, err := src.Conn(ctx)
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	defer conn.Close()
	destURI := ownedStopURI(destination, "rw")
	err = conn.Raw(func(driverConn any) error {
		maker, ok := driverConn.(interface {
			NewBackup(string) (*sqlite.Backup, error)
		})
		if !ok {
			return errors.New("pinned native backup unavailable")
		}
		backup, e := maker.NewBackup(destURI)
		if e != nil {
			return e
		}
		var stepErr error
		complete := false
		for steps := 0; steps < 100000; steps++ {
			if e = ctx.Err(); e != nil {
				stepErr = e
				break
			}
			var more bool
			more, e = backup.Step(128)
			if e != nil {
				stepErr = e
				break
			}
			if !more {
				complete = true
				break
			}
		}
		if !complete && stepErr == nil {
			stepErr = errors.New("online backup step limit")
		}
		return errors.Join(stepErr, backup.Finish())
	})
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	closed, err := os.OpenFile(destination, os.O_RDWR, 0)
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	if err = closed.Sync(); err != nil {
		closed.Close()
		return OwnedStopCensus{}, "", err
	}
	h := sha256.New()
	_, err = io.Copy(h, closed)
	closeErr := closed.Close()
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	if closeErr != nil {
		return OwnedStopCensus{}, "", closeErr
	}
	ro, err := sql.Open("sqlite", ownedStopURI(destination, "ro"))
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	defer ro.Close()
	ro.SetMaxOpenConns(1)
	census, err := ownedStopCensus(ctx, ro)
	return census, hex.EncodeToString(h.Sum(nil)), err
}

// ReadOwnedStopBackup rereads a closed backup as a read-only database. It is
// used when committing a reviewed plan; it never refreshes the backup.
func ReadOwnedStopBackup(ctx context.Context, path string) (OwnedStopCensus, string, error) {
	if !filepath.IsAbs(path) || filepath.Clean(path) != path {
		return OwnedStopCensus{}, "", errors.New("canonical backup path required")
	}
	f, err := os.Open(path)
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	h := sha256.New()
	_, err = io.Copy(h, f)
	closeErr := f.Close()
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	if closeErr != nil {
		return OwnedStopCensus{}, "", closeErr
	}
	db, err := sql.Open("sqlite", ownedStopURI(path, "ro"))
	if err != nil {
		return OwnedStopCensus{}, "", err
	}
	defer db.Close()
	db.SetMaxOpenConns(1)
	census, err := ownedStopCensus(ctx, db)
	return census, hex.EncodeToString(h.Sum(nil)), err
}

func ReadOwnedStopCensus(ctx context.Context, path string) (OwnedStopCensus, error) {
	if !filepath.IsAbs(path) || filepath.Clean(path) != path {
		return OwnedStopCensus{}, errors.New("canonical existing SQLite path required")
	}
	db, err := sql.Open("sqlite", ownedStopURI(path, "ro"))
	if err != nil {
		return OwnedStopCensus{}, err
	}
	defer db.Close()
	db.SetMaxOpenConns(1)
	return ownedStopCensus(ctx, db)
}

package store

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"os"
	"path/filepath"
	"strings"
	"time"
)

// OwnedStopBarrier owns one SQLite connection and one zero-row writer reservation.
// It deliberately exposes no SQL executor or connection to its caller.
type OwnedStopBarrier struct {
	db       *sql.DB
	conn     *sql.Conn
	reserved bool
}

type OwnedStopSubject struct {
	Project     string `json:"project"`
	Run         string `json:"run"`
	Epoch       int    `json:"epoch"`
	DispatchKey string `json:"dispatch_key"`
	RequestKey  string `json:"request_key"`
}

type OwnedStopSafetySnapshot struct {
	RunStatus                 string `json:"run_status"`
	DispatchState             string `json:"dispatch_state"`
	DispatchKey               string `json:"dispatch_key"`
	UnresolvedOtherRuns       int    `json:"unresolved_other_runs"`
	UnresolvedOtherDispatches int    `json:"unresolved_other_dispatches"`
	UnresolvedExternalEffects int    `json:"unresolved_external_effects"`
	NonModelEffects           int    `json:"non_model_effects"`
	ConfirmedModelEffects     int    `json:"confirmed_model_effects"`
	PendingClientCommands     int    `json:"pending_client_commands"`
	PendingControlRequests    int    `json:"pending_control_requests"`
	SchemaVersion             int    `json:"schema_version"`
}

func (s OwnedStopSafetySnapshot) ExactOldSubjectSafe() bool {
	return s.SchemaVersion == 24 && s.RunStatus == "RUNNING" && s.DispatchState == "UNKNOWN" &&
		s.UnresolvedOtherRuns == 0 && s.UnresolvedOtherDispatches == 0 &&
		s.UnresolvedExternalEffects == 0 && s.NonModelEffects == 0 && s.ConfirmedModelEffects == 1 &&
		s.PendingClientCommands == 0 && s.PendingControlRequests == 0
}

func ownedStopExistingDB(path string) (string, error) {
	if !filepath.IsAbs(path) || filepath.Clean(path) != path || !strings.EqualFold(filepath.Ext(path), ".sqlite") {
		return "", errors.New("canonical existing SQLite path required")
	}
	actual, err := filepath.EvalSymlinks(path)
	if err != nil || !strings.EqualFold(filepath.Clean(actual), path) {
		return "", errors.New("SQLite path reparse or absence")
	}
	info, err := os.Stat(path)
	if err != nil || !info.Mode().IsRegular() {
		return "", errors.New("regular existing SQLite file required")
	}
	return path, nil
}

// AcquireOwnedStopBarrier does not call store.Open, create, bootstrap, migrate,
// checkpoint or change a persistent PRAGMA. The only SQLite writes it requests
// are BEGIN IMMEDIATE and ROLLBACK, with no product row write.
func AcquireOwnedStopBarrier(ctx context.Context, path string) (_ *OwnedStopBarrier, err error) {
	path, err = ownedStopExistingDB(path)
	if err != nil {
		return nil, err
	}
	db, err := sql.Open("sqlite", ownedStopURI(path, "rw"))
	if err != nil {
		return nil, err
	}
	db.SetMaxOpenConns(1)
	db.SetMaxIdleConns(1)
	b := &OwnedStopBarrier{db: db}
	defer func() {
		if err != nil {
			_ = b.Close()
		}
	}()
	b.conn, err = db.Conn(ctx)
	if err != nil {
		return nil, err
	}
	if _, err = b.conn.ExecContext(ctx, "BEGIN IMMEDIATE"); err != nil {
		return nil, fmt.Errorf("reserve writer: %w", err)
	}
	b.reserved = true
	if _, err = b.conn.ExecContext(ctx, "PRAGMA query_only=ON"); err != nil {
		return nil, fmt.Errorf("enable query_only: %w", err)
	}
	var readback int
	if err = b.conn.QueryRowContext(ctx, "PRAGMA query_only").Scan(&readback); err != nil || readback != 1 {
		return nil, errors.New("query_only readback failed")
	}
	return b, nil
}

// Census reads on the reserved connection, so a Core transaction committed
// while BEGIN IMMEDIATE waited is included in this exact snapshot.
func (b *OwnedStopBarrier) Census(ctx context.Context) (OwnedStopCensus, error) {
	if b == nil || b.conn == nil || !b.reserved {
		return OwnedStopCensus{}, errors.New("writer reservation unavailable")
	}
	return ownedStopCensus(ctx, b.conn)
}

// SafetySnapshot executes a closed catalog of parameterized SELECTs on the
// reserved connection. The one historical RUNNING/UNKNOWN subject is allowed;
// every other unresolved subject or external effect is rejected.
func (b *OwnedStopBarrier) SafetySnapshot(ctx context.Context, subject OwnedStopSubject) (OwnedStopSafetySnapshot, error) {
	if b == nil || b.conn == nil || !b.reserved {
		return OwnedStopSafetySnapshot{}, errors.New("reserved connection required")
	}
	return ownedStopSafetySnapshot(ctx, b.conn, subject)
}

// ReadOwnedStopSafetyPreflight is informational. Commit repeats this exact
// catalog after acquiring the writer reservation and uses that result.
func ReadOwnedStopSafetyPreflight(ctx context.Context, path string, subject OwnedStopSubject) (OwnedStopSafetySnapshot, error) {
	path, err := ownedStopExistingDB(path)
	if err != nil {
		return OwnedStopSafetySnapshot{}, err
	}
	db, err := sql.Open("sqlite", ownedStopURI(path, "ro"))
	if err != nil {
		return OwnedStopSafetySnapshot{}, err
	}
	defer db.Close()
	db.SetMaxOpenConns(1)
	return ownedStopSafetySnapshot(ctx, db, subject)
}

func ownedStopSafetySnapshot(ctx context.Context, source ownedStopQueryer, subject OwnedStopSubject) (OwnedStopSafetySnapshot, error) {
	var s OwnedStopSafetySnapshot
	if subject.Project == "" || subject.Run == "" || subject.Epoch < 1 || subject.DispatchKey == "" || subject.RequestKey == "" {
		return s, errors.New("exact subject required")
	}
	var n, minVersion int
	if err := source.QueryRowContext(ctx, `SELECT max(version),min(version),count(DISTINCT version) FROM schema_migrations`).Scan(&s.SchemaVersion, &minVersion, &n); err != nil || s.SchemaVersion != 24 || minVersion != 1 || n != 24 {
		return s, errors.New("schema 24 required")
	}
	if err := source.QueryRowContext(ctx, `SELECT json_extract(body,'$.status') FROM domain_objects WHERE project=? AND kind='Run' AND id=?`, subject.Project, subject.Run).Scan(&s.RunStatus); err != nil {
		return s, err
	}
	if err := source.QueryRowContext(ctx, `SELECT state,dispatch_key FROM runtime_dispatches WHERE project=? AND run=? AND epoch=?`, subject.Project, subject.Run, subject.Epoch).Scan(&s.DispatchState, &s.DispatchKey); err != nil {
		return s, err
	}
	queries := []struct {
		sql  string
		args []any
		dst  *int
	}{
		{`SELECT count(*) FROM domain_objects WHERE kind='Run' AND NOT(project=? AND id=?) AND coalesce(json_extract(body,'$.status'),'UNKNOWN') NOT IN ('SETTLED','FAILED','CANCELLED')`, []any{subject.Project, subject.Run}, &s.UnresolvedOtherRuns},
		{`SELECT count(*) FROM runtime_dispatches WHERE NOT(project=? AND run=? AND epoch=?) AND state IN ('PREPARED','STARTED','UNKNOWN')`, []any{subject.Project, subject.Run, subject.Epoch}, &s.UnresolvedOtherDispatches},
		{`SELECT count(*) FROM resource_effects x WHERE sequence=(SELECT max(y.sequence) FROM resource_effects y WHERE y.project=x.project AND y.run=x.run AND y.effect_key=x.effect_key) AND json_extract(body,'$.state') IN ('PREPARED','STARTED','UNKNOWN')`, nil, &s.UnresolvedExternalEffects},
		{`SELECT count(*) FROM resource_effects x WHERE sequence=(SELECT max(y.sequence) FROM resource_effects y WHERE y.project=x.project AND y.run=x.run AND y.effect_key=x.effect_key) AND (coalesce(json_extract(body,'$.payload.capability'),'')!='MODEL' OR json_extract(body,'$.state')!='CONFIRMED' OR NOT(x.project=? AND x.run=?))`, []any{subject.Project, subject.Run}, &s.NonModelEffects},
		{`SELECT count(*) FROM resource_effects x WHERE sequence=(SELECT max(y.sequence) FROM resource_effects y WHERE y.project=x.project AND y.run=x.run AND y.effect_key=x.effect_key) AND json_extract(body,'$.payload.capability')='MODEL' AND json_extract(body,'$.state')='CONFIRMED' AND x.project=? AND x.run=?`, []any{subject.Project, subject.Run}, &s.ConfirmedModelEffects},
		{`SELECT count(*) FROM client_command_queue WHERE state NOT IN ('CONSUMED','DROPPED')`, nil, &s.PendingClientCommands},
		{`SELECT count(*) FROM client_control_requests WHERE state!='CONFIRMED'`, nil, &s.PendingControlRequests},
	}
	for _, q := range queries {
		if err := source.QueryRowContext(ctx, q.sql, q.args...).Scan(q.dst); err != nil {
			return s, err
		}
	}
	if s.DispatchKey != subject.DispatchKey {
		return s, errors.New("dispatch identity mismatch")
	}
	var requestCount int
	if err := source.QueryRowContext(ctx, `SELECT count(*) FROM idempotency WHERE project=? AND operation='host:managed-runtime-qualification' AND key=?`, subject.Project, subject.RequestKey).Scan(&requestCount); err != nil {
		return s, err
	}
	if requestCount != 1 {
		return s, errors.New("exact original request not found")
	}
	return s, nil
}

// Close is idempotent and always rolls back; no COMMIT path exists.
func (b *OwnedStopBarrier) Close() error {
	if b == nil {
		return nil
	}
	var errs []error
	if b.conn != nil {
		if b.reserved {
			ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
			_, err := b.conn.ExecContext(ctx, "ROLLBACK")
			cancel()
			if err != nil {
				errs = append(errs, err)
			}
			b.reserved = false
		}
		if err := b.conn.Close(); err != nil {
			errs = append(errs, err)
		}
		b.conn = nil
	}
	if b.db != nil {
		if err := b.db.Close(); err != nil {
			errs = append(errs, err)
		}
		b.db = nil
	}
	return errors.Join(errs...)
}

package main

import (
	"context"
	"encoding/json"
	"errors"
	"fmt"
	"os"
	"strings"
	"time"

	"researchos.local/research-os/internal/ownedcorestop"
)

func main() { os.Exit(run(os.Args[1:])) }

func parseCommand(args []string) (string, map[string]string, error) {
	if len(args) == 0 {
		return "", nil, errors.New("one action required")
	}
	action := args[0]
	var required []string
	switch action {
	case "preflight":
		required = []string{"--input", "--operation-id", "--out"}
	case "commit":
		required = []string{"--plan", "--expect-plan-sha256"}
	case "inspect":
		required = []string{"--plan"}
	default:
		return "", nil, errors.New("unknown action")
	}
	if len(args) != 1+2*len(required) {
		return "", nil, errors.New("exact action flags required")
	}
	allowed := make(map[string]bool, len(required))
	for _, k := range required {
		allowed[k] = true
	}
	parsed := make(map[string]string, len(required))
	for i := 1; i < len(args); i += 2 {
		key, value := args[i], args[i+1]
		_, seen := parsed[key]
		if !allowed[key] || seen || value == "" || strings.HasPrefix(value, "--") || strings.ContainsRune(value, '\x00') {
			return "", nil, errors.New("mixed, duplicate, or malformed flags")
		}
		parsed[key] = value
	}
	if len(parsed) != len(required) {
		return "", nil, errors.New("missing action flag")
	}
	return action, parsed, nil
}

func run(args []string) int {
	action, flags, parseErr := parseCommand(args)
	if parseErr != nil {
		fmt.Fprintln(os.Stderr, parseErr)
		return 2
	}
	ctx, cancel := context.WithTimeout(context.Background(), 45*time.Second)
	defer cancel()
	var result any
	var err error
	switch action {
	case "preflight":
		var plan ownedcorestop.Plan
		var sha string
		plan, sha, err = ownedcorestop.Preflight(ctx, flags["--input"], flags["--operation-id"], flags["--out"])
		if err == nil {
			result = struct {
				Plan       ownedcorestop.Plan `json:"plan"`
				PlanSHA256 string             `json:"plan_sha256"`
			}{plan, sha}
		}
	case "commit":
		result, err = ownedcorestop.Commit(ctx, flags["--plan"], flags["--expect-plan-sha256"])
	case "inspect":
		result, err = ownedcorestop.Inspect(flags["--plan"])
	}
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		return 1
	}
	if err = json.NewEncoder(os.Stdout).Encode(result); err != nil {
		fmt.Fprintln(os.Stderr, err)
		return 1
	}
	if r, ok := result.(ownedcorestop.Result); ok && r.State != "STOPPED_LEASE_VERIFIED" && r.State != "PREFLIGHT_COMPLETE" {
		return 3
	}
	return 0
}

func (d *heldDocuments) Close() {
	for i := len(d.files) - 1; i >= 0; i-- {
		d.files[i].Close()
	}
	d.files = nil
}
func (d *heldDocuments) read(ref grantDocument, max int) ([]byte, error) {
	if !validGrantDocument(ref) {
		return nil, errors.New("invalid pinned document")
	}
	raw, err := d.readUnpinned(ref.Path, max)
	if err != nil {
		return nil, err
	}
	if digest(raw) != ref.SHA256 {
		return nil, errors.New("pinned document digest differs")
	}
	return raw, nil
}
func (d *heldDocuments) readUnpinned(path string, max int) ([]byte, error) {
	h, err := openHeldFile(path, false)
	if err != nil {
		return nil, err
	}
	d.files = append(d.files, h)
	raw, err := readHeldBytes(h, max)
	if err != nil {
		return nil, err
	}
	return raw, nil
}

type operationLock struct {
	dir  *heldFile
	file *os.File
}

func (l *operationLock) Close() {
	if l == nil {
		return
	}
	if l.file != nil {
		_ = l.file.Close()
	}
	if l.dir != nil {
		l.dir.Close()
	}
}
