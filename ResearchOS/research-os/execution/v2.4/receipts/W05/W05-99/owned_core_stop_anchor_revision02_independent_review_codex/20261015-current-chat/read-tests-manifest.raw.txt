func TestOwnedCoreStopPostflightRealLeaseBlocksLogicalDrift(t *testing.T) {
	fixture := os.Getenv("RESEARCHOS_OWNED_STOP_FIXTURE_EXE")
	if fixture == "" {
		t.Fatal("new owned fixture exe required")
	}
	base := t.TempDir()
	root := filepath.Join(base, "data")
	if err := os.Mkdir(root, 0700); err != nil {
		t.Fatal(err)
	}
	dbPath := filepath.Join(root, "researchos.sqlite")
	db, err := sql.Open("sqlite", dbPath)
	if err != nil {
		t.Fatal(err)
	}
	if _, err = db.Exec(`PRAGMA journal_mode=WAL;CREATE TABLE stop_fixture_rows(id TEXT PRIMARY KEY,payload BLOB)`); err != nil {
		db.Close()
		t.Fatal(err)
	}
	if err = db.Close(); err != nil {
		t.Fatal(err)
	}
	markers := filepath.Join(base, "markers")
	if err = os.Mkdir(markers, 0700); err != nil {
		t.Fatal(err)
	}
	ready := filepath.Join(markers, "ready")
	cmd := exec.Command(fixture, "--root", root, "--ready", ready, "--attempt", filepath.Join(markers, "never-attempt"), "--attempted", filepath.Join(markers, "never-attempted"))
	if err = cmd.Start(); err != nil {
		t.Fatal(err)
	}
	defer func() { _ = cmd.Process.Kill(); _ = cmd.Wait() }()
	found := false
	for deadline := time.Now().Add(6 * time.Second); time.Now().Before(deadline); time.Sleep(20 * time.Millisecond) {
		if _, e := os.Stat(ready); e == nil {
			found = true
			break
		}
	}
	if !found {
		t.Fatal("fixture ready missing")
	}
	baseline, err := store.ReadOwnedStopCensus(context.Background(), dbPath)
	if err != nil || baseline.Rows != 1 {
		t.Fatalf("committed baseline: %+v %v", baseline, err)
	}
	id, err := supervisor.ObserveOwnedCoreReadOnly(uint32(cmd.Process.Pid))
	if err != nil {
		t.Fatal(err)
	}
	h, err := supervisor.OpenOwnedCoreRetained(id)
	if err != nil {
		t.Fatal(err)
	}
	defer h.Close()
	exit, err := h.TerminateOnceAndWait(5 * time.Second)
	if err != nil || !exit.ExitObserved || exit.TerminateCalls != 1 {
		t.Fatalf("native fixture exit: %+v %v", exit, err)
	}
	db, err = sql.Open("sqlite", dbPath)
	if err != nil {
		t.Fatal(err)
	}
	if _, err = db.Exec(`INSERT INTO stop_fixture_rows VALUES('poststop-drift',x'ff')`); err != nil {
		db.Close()
		t.Fatal(err)
	}
	if err = db.Close(); err != nil {
		t.Fatal(err)
	}
	out := filepath.Join(base, "postflight")
	if err = createPrivateOutput(out, root); err != nil {
		t.Fatal(err)
	}
	plan := Plan{OperationID: strings.Repeat("c", 32), OutputDir: out, Input: Input{DataRoot: root}, Baseline: baseline}
	plan.Host.Database = dbPath
	result, err := finishAfterNativeExit(context.Background(), plan, strings.Repeat("d", 64), exit)
	if err != nil {
		t.Fatal(err)
	}
	if result.State != "STOP_OBSERVED_MAINTENANCE_BLOCKED" || result.Postflight == nil || result.Postflight.Rows != 2 || *result.Postflight == baseline {
		t.Fatalf("poststop logical drift escaped maintenance block: %+v", result)
	}
	if _, err := os.Stat(filepath.Join(out, "POSTFLIGHT.json")); err != nil {
		t.Fatalf("durable blocked receipt missing: %v", err)
	}
}

	}

func TestOwnedCoreTerminateFailureIsSingleUnknownAttempt(t *testing.T) {
	core := ownedCoreChild(t)
	defer func() { _ = core.Process.Kill(); _ = core.Wait() }()
	id, err := ObserveOwnedCoreReadOnly(uint32(core.Process.Pid))

	if err != nil {
		t.Fatal(err)
	}
	defer h.Close()
	called := 0
	h.terminate = func(windows.Handle, uint32) error { called++; return errors.New("injected native failure") }
	exit, err := h.TerminateOnceAndWait(time.Second)
	if err == nil || exit.TerminateCalls != 1 || exit.ExitObserved || called != 1 {
		t.Fatalf("one-call failure was not UNKNOWN: %+v calls=%d err=%v", exit, called, err)
	}
	if again, err := h.TerminateOnceAndWait(time.Second); err == nil || again.TerminateCalls != 0 || called != 1 {
		t.Fatalf("native failure retried: %+v calls=%d err=%v", again, called, err)
	}
	if _, err := ObserveOwnedCoreReadOnly(uint32(core.Process.Pid)); err != nil {
		t.Fatalf("injected failure affected target before cleanup: %v", err)
	}
}

func TestOwnedCoreRetainedHandleSingleChildAndUnrelatedSurvives(t *testing.T) {
	core := ownedCoreChild(t)
	unrelated := ownedCoreChild(t)
	defer func() { _ = core.Wait(); _ = unrelated.Wait() }()
	observed, err := ObserveOwnedCoreReadOnly(uint32(core.Process.Pid))
	if err != nil {
		t.Fatal(err)
	}
	if observed.PID != uint32(core.Process.Pid) || observed.CreatedNS == 0 || observed.SID == "" || observed.ParentPID != uint32(os.Getpid()) || observed.ImageIndex == 0 || observed.ParentImageIndex == 0 {
		t.Fatalf("incomplete retained identity: %+v", observed)
	}
	wrong := observed
	wrong.CreatedNS++
	if h, e := OpenOwnedCoreRetained(wrong); e == nil {
		h.Close()
		t.Fatal("PID reuse identity accepted")
	}
	wrong = observed
	wrong.ImageSHA256 = strings.Repeat("0", 64)
	if h, e := OpenOwnedCoreRetained(wrong); e == nil {
		h.Close()
		t.Fatal("replacement image identity accepted")
	}
	h, err := OpenOwnedCoreRetained(observed)
	if err != nil {
		t.Fatal(err)
	}
	defer h.Close()
	if h.Identity() != observed {
		t.Fatal("retained identity changed")
	}
	exit, err := h.TerminateOnceAndWait(5 * time.Second)
	if err != nil || exit.TerminateCalls != 1 || !exit.ExitObserved || exit.ExitCode != OwnedCoreMaintenanceExitCode || exit.BeforeSignal {
		t.Fatalf("single exact termination failed: %+v %v", exit, err)
	}
	if second, e := h.TerminateOnceAndWait(time.Second); e == nil || second.TerminateCalls != 0 {
		t.Fatalf("second termination accepted: %+v %v", second, e)
	}
	other, e := ObserveOwnedCoreReadOnly(uint32(unrelated.Process.Pid))
	if e != nil || other.PID != uint32(unrelated.Process.Pid) {
		t.Fatalf("same-console unrelated child lost: %+v %v", other, e)
	}
