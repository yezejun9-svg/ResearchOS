64: // Checker runs a real verification command in a real directory and reports its exit code plus
65: // combined output. A nil error means the process ran; a non-zero exit is a failing check, not a
66: // failure to check.
67: type Checker func(ctx context.Context, dir string) (exitCode int, output string, err error)
68: 
69: func CommandChecker(argv ...string) Checker {
70: 	return func(ctx context.Context, dir string) (int, string, error) {
71: 		if len(argv) == 0 {
72: 			return -1, "", errors.New("NO_CHECK_COMMAND_CONFIGURED")
73: 		}
74: 		cmd := exec.CommandContext(ctx, argv[0], argv[1:]...)
75: 		cmd.Dir = dir
76: 		out, err := cmd.CombinedOutput()
77: 		var ee *exec.ExitError
78: 		if errors.As(err, &ee) {
79: 			// The check ran and said no. That is a failing check, not a failure to check.
80: 			return ee.ExitCode(), string(out), nil
81: 		}
82: 		if err != nil {
83: 			// The check never ran. Reporting exit -1 here would let an unstartable command pose as
84: 			// a failing verification and drive the repair loop on nothing.
85: 			return -1, string(out), fmt.Errorf("CHECK_EXEC_FAILED: %w", err)
86: 		}
87: 		return 0, string(out), nil
88: 	}
89: }
