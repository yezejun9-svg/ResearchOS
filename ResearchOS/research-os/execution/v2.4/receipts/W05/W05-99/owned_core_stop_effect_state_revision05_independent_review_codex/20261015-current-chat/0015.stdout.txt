	raw, err := json.Marshal(g)
	if err != nil {
		t.Fatal(err)
	}
	raw, err = canonicalMapJSON(raw)
	if err != nil {
		t.Fatal(err)
	}
	parsed, err := parseCompiledOperationGrant(base64.StdEncoding.EncodeToString(raw))
	if err != nil {
		t.Fatal(err)
	}
	if parsed.digest != digest(raw) {
		t.Fatal("compiled grant digest differs")
	}
	if _, err = parsed.checkAction("commit", true); err == nil {
		t.Fatal("preflight grant allowed commit")
	}
	parsed.expires = time.Now().Add(-time.Second)
	if _, err = parsed.checkAction("preflight", true); err == nil {
		t.Fatal("expired preflight allowed action")
	}
	if _, err = parsed.checkAction("inspect", false); err != nil {
		t.Fatalf("expired read-only inspect denied: %v", err)
	}
	for _, bad := range []string{"", base64.StdEncoding.EncodeToString([]byte(`{"schema":"a","schema":"b"}`)), base64.StdEncoding.EncodeToString(raw) + "\n"} {
		if _, err = parseCompiledOperationGrant(bad); err == nil {
			t.Fatalf("malformed grant accepted: %q", bad)
		}
	}
	var unknown map[string]any
	if err = json.Unmarshal(raw, &unknown); err != nil {
		t.Fatal(err)
	}
	unknown["wildcard"] = true
	unknownRaw, err := json.Marshal(unknown)
	if err != nil {
		t.Fatal(err)
	}
	unknownRaw, err = canonicalMapJSON(unknownRaw)
	if err != nil {
		t.Fatal(err)
	}
	if _, err = parseCompiledOperationGrant(base64.StdEncoding.EncodeToString(unknownRaw)); err == nil {
		t.Fatal("unknown grant field accepted")
	}
	g.Proposal.SHA256 = strings.Repeat("0", 64)
	zeroRaw, err := json.Marshal(g)
	if err != nil {
		t.Fatal(err)
	}
	zeroRaw, err = canonicalMapJSON(zeroRaw)
	if err != nil {
		t.Fatal(err)
	}
	if _, err = parseCompiledOperationGrant(base64.StdEncoding.EncodeToString(zeroRaw)); err == nil {
		t.Fatal("zero digest accepted")
	}
}

func TestNativeExitProofRequiresExplicitNumericCapturePerClass(t *testing.T) {
	sha := strings.Repeat("a", 64)
	for _, kind := range []string{"worker_native_exit", "verifier_native_exit"} {
		for _, tc := range []struct {
			name  string
			field string
			value any
			omit  bool
			want  bool
		}{
			{"explicit exit zero", "native_exit_code", 0, false, true},
			{"explicit exit nonzero", "native_exit_code", 137, false, true},
			{"missing exit", "native_exit_code", nil, true, false},
			{"null exit", "native_exit_code", nil, false, false},
			{"negative exit", "native_exit_code", -1, false, false},
			{"overflow exit", "native_exit_code", int64(1) << 32, false, false},
			{"fractional exit", "native_exit_code", 0.5, false, false},
			{"explicit wait zero", "native_wait_result", 0, false, true},
			{"missing wait", "native_wait_result", nil, true, false},
			{"null wait", "native_wait_result", nil, false, false},
			{"nonzero wait", "native_wait_result", 258, false, false},
			{"negative wait", "native_wait_result", -1, false, false},
			{"overflow wait", "native_wait_result", int64(1) << 32, false, false},
			{"fractional wait", "native_wait_result", 0.5, false, false},
		} {
			t.Run(kind+"/"+tc.name, func(t *testing.T) {
				fields := map[string]any{
					"kind": kind, "native_pid": 8, "native_created_ns": 123,
					"native_wait_api": "WaitForSingleObject", "native_wait_result": 0,
					"native_exit_api": "GetExitCodeProcess", "native_exit_code": 0,
					"native_handle_record_sha256": sha,
				}
				if tc.omit {
					delete(fields, tc.field)
				} else {
					fields[tc.field] = tc.value
				}
				raw, err := json.Marshal(fields)
				if err != nil {
					t.Fatal(err)
				}
				var proof historicalEvidence
				err = decodeCanonicalAuthority(raw, &proof)
				got := err == nil && nativeExitEvidenceComplete(proof, 9)
				if got != tc.want {
					t.Fatalf("native proof %s accepted=%v want=%v decode=%v: %s", tc.name, got, tc.want, err, raw)
				}
			})
		}
	}
}

func TestOwnedCoreStopPostflightRealLeaseBlocksLogicalDrift(t *testing.T) {
	fixture := os.Getenv("RESEARCHOS_OWNED_STOP_FIXTURE_EXE")
	if fixture == "" {
		t.Fatal("new owned fixture exe required")
	}
	base := t.TempDir()
	root := filepath.Join(base, "data")
	if err := os.Mkdir(root, 0700); err != nil {
		t.Fatal(err)
	}
	dbPath := filepath.Join(root, "researchos.sqlite")
	db, err := sql.Open("sqlite", dbPath)
	if err != nil {
		t.Fatal(err)
	}
	if _, err = db.Exec(`PRAGMA journal_mode=WAL;CREATE TABLE stop_fixture_rows(id TEXT PRIMARY KEY,payload BLOB)`); err != nil {
		db.Close()
		t.Fatal(err)
	}
	if err = db.Close(); err != nil {
		t.Fatal(err)
	}
	markers := filepath.Join(base, "markers")
	if err = os.Mkdir(markers, 0700); err != nil {
		t.Fatal(err)
	}
	ready := filepath.Join(markers, "ready")
	cmd := exec.Command(fixture, "--root", root, "--ready", ready, "--attempt", filepath.Join(markers, "never-attempt"), "--attempted", filepath.Join(markers, "never-attempted"))
	if err = cmd.Start(); err != nil {
		t.Fatal(err)
	}
	defer func() { _ = cmd.Process.Kill(); _ = cmd.Wait() }()
	found := false
	for deadline := time.Now().Add(6 * time.Second); time.Now().Before(deadline); time.Sleep(20 * time.Millisecond) {
		if _, e := os.Stat(ready); e == nil {
			found = true
			break
		}
	}
	if !found {
		t.Fatal("fixture ready missing")
	}
	baseline, err := store.ReadOwnedStopCensus(context.Background(), dbPath)
	if err != nil || baseline.Rows != 1 {
		t.Fatalf("committed baseline: %+v %v", baseline, err)
	}
	id, err := supervisor.ObserveOwnedCoreReadOnly(uint32(cmd.Process.Pid))
	if err != nil {
		t.Fatal(err)
	}
	h, err := supervisor.OpenOwnedCoreRetained(id)
	if err != nil {
		t.Fatal(err)
	}
	defer h.Close()
	exit, err := h.TerminateOnceAndWait(5 * time.Second)
	if err != nil || !exit.ExitObserved || exit.TerminateCalls != 1 {
		t.Fatalf("native fixture exit: %+v %v", exit, err)
	}
	db, err = sql.Open("sqlite", dbPath)
	if err != nil {
		t.Fatal(err)
	}
	if _, err = db.Exec(`INSERT INTO stop_fixture_rows VALUES('poststop-drift',x'ff')`); err != nil {
		db.Close()
		t.Fatal(err)
	}
	if err = db.Close(); err != nil {
		t.Fatal(err)
	}
	out := filepath.Join(base, "postflight")
	if err = createPrivateOutput(out, root); err != nil {
		t.Fatal(err)
	}
	plan := Plan{OperationID: strings.Repeat("c", 32), OutputDir: out, Input: Input{DataRoot: root}, Baseline: baseline}
	plan.Host.Database = dbPath
	result, err := finishAfterNativeExit(context.Background(), plan, strings.Repeat("d", 64), exit)
	if err != nil {
		t.Fatal(err)
	}
	if result.State != "STOP_OBSERVED_MAINTENANCE_BLOCKED" || result.Postflight == nil || result.Postflight.Rows != 2 || *result.Postflight == baseline {
		t.Fatalf("poststop logical drift escaped maintenance block: %+v", result)
	}
	if _, err := os.Stat(filepath.Join(out, "POSTFLIGHT.json")); err != nil {
		t.Fatalf("durable blocked receipt missing: %v", err)
	}
}

func TestCommitSignalBoundaryHonorsCallerCancellationAndEarlierDeadline(t *testing.T) {
	now := time.Now()
	g := operationGrant{AllowedActions: []string{"commit"}, notBefore: now.Add(-time.Minute), expires: now.Add(time.Minute)}
	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	if err := checkCommitSignalBoundary(ctx, g); err != context.Canceled {
		t.Fatalf("canceled caller reached intent: %v", err)
	}
	ctx, cancel = context.WithDeadline(context.Background(), now.Add(-time.Second))
	defer cancel()
	if err := checkCommitSignalBoundary(ctx, g); err != context.DeadlineExceeded {
		t.Fatalf("earlier caller deadline reached intent: %v", err)
	}
	if err := checkCommitSignalBoundary(context.Background(), g); err != nil {
		t.Fatalf("valid caller/grant denied: %v", err)
	}
}

func TestHeldAuthorityAndOperationLockDenyReplacement(t *testing.T) {
	base := t.TempDir()
	data := filepath.Join(base, "data")
	if err := os.Mkdir(data, 0700); err != nil {
		t.Fatal(err)
	}
	out := filepath.Join(base, "operation")
	if err := createPrivateOutput(out, data); err != nil {
		t.Fatal(err)
	}
	h, err := openHeldFile(out, true)
	if err != nil {
		t.Fatal(err)
	}
	id := nativeFileIdentity(h.info, out)
	h.Close()
	g := operationGrant{DataRoot: grantFileIdentity{Path: data}, OperationDir: id}
	lock, err := acquireOperationLock(g, true)
	if err != nil {
		t.Fatal(err)
	}
	if second, err := acquireOperationLock(g, false); err == nil {
		second.Close()
		t.Fatal("parallel operation acquired exclusive lock")
	}
	if _, err := writeOwnedJournal(out, "authority.json", map[string]string{"role": "fixture"}); err != nil {
		t.Fatal(err)
	}
	doc, err := openHeldFile(filepath.Join(out, "authority.json"), false)
	if err != nil {
		t.Fatal(err)
	}
	if f, e := os.OpenFile(filepath.Join(out, "authority.json"), os.O_WRONLY, 0); e == nil {
		f.Close()
		t.Fatal("held authority file allowed write sharing")
	}
	if err := os.Rename(out, filepath.Join(base, "replacement")); err == nil {
		t.Fatal("held ancestor directory renamed")
	}
	doc.Close()
	lock.Close()
	replay, err := acquireOperationLock(g, false)
	if err != nil {
		t.Fatal(err)
	}
	replay.Close()
}

func TestOwnedCoreStopInspectJournalCrashBoundaries(t *testing.T) {
	base := t.TempDir()
	dataRoot := filepath.Join(base, "data")
	if err := os.Mkdir(dataRoot, 0700); err != nil {
		t.Fatal(err)
	}
	out := filepath.Join(base, "new-output")
	if err := createPrivateOutput(out, dataRoot); err != nil {
		t.Fatal(err)
	}
	subject := store.OwnedStopSubject{Run: OldRun, Epoch: 1, DispatchKey: OldDispatch, RequestKey: OldRequest, Project: "fixture"}
	sid := "fixture-sid"
	op := strings.Repeat("a", 32)
	plan := Plan{
		Schema: PlanSchema, OperationID: op, SourceBaseCommit: SourceBaseCommit,
		OutputDir: out, BackupPath: filepath.Join(out, "backup.sqlite"),
		InputSHA256: strings.Repeat("b", 64), BackupSHA256: strings.Repeat("c", 64),
		Input:       Input{Schema: InputSchema, OperationID: op, SourceRepository: base, DataRoot: dataRoot, CorePID: 1, Subject: subject},
		Native:      supervisor.OwnedCoreNativeIdentity{PID: 1, SID: sid, ParentSID: sid},
		Host:        host.OwnedCoreStopPublicIdentity{DataRoot: dataRoot, SID: sid, SchemaVersion: 24},
		Baseline:    store.OwnedStopCensus{SHA256: strings.Repeat("d", 64), QuickCheck: "ok"},
		Eligibility: "STOP_UNAVAILABLE",
	}
	planSHA, err := writeOwnedJournal(out, "plan.json", plan)
	if err != nil {
		t.Fatal(err)
	}
	grantSHA := strings.Repeat("e", 64)
	g := operationGrant{OperationID: op, OperationDir: grantFileIdentity{Path: out}, digest: grantSHA}
	if _, err := writeOwnedJournal(out, "STOP_ATTEMPT.json", stopAttemptRecord{Schema: "researchos.owned-core-stop-attempt.v1", OperationID: op, GrantSHA256: grantSHA, PlanSHA256: planSHA}); err != nil {
		t.Fatal(err)
	}
	check := func(state string) {
		t.Helper()
		docs := &heldDocuments{}
		got, err := readExistingStopOutcome(docs, g, plan, planSHA)
		docs.Close()
		if err != nil || got.State != state || got.PlanSHA256 != planSHA {
			t.Fatalf("inspect %s: %+v %v", state, got, err)
		}
	}
	check("UNKNOWN") // before BEGIN
	if _, err := writeOwnedJournal(out, "BEGIN.json", ownedBeginRecord{Schema: "researchos.owned-core-stop-begin.v1", OperationID: op, PlanSHA256: planSHA, GrantSHA256: grantSHA}); err != nil {
		t.Fatal(err)
	}
	check("UNKNOWN") // after BEGIN, before termination intent
	if _, err := writeOwnedJournal(out, "TERMINATE_INTENT.json", ownedIntentRecord{Schema: "researchos.owned-core-stop-terminate-intent.v1", OperationID: op, PlanSHA256: planSHA, GrantSHA256: grantSHA, Native: plan.Native}); err != nil {
		t.Fatal(err)
	}
	check("UNKNOWN") // before/after native call without durable wait proof
	boundaryGrant := operationGrant{AllowedActions: []string{"commit"}, notBefore: time.Now().Add(-time.Minute), expires: time.Now().Add(time.Minute)}
	canceled, cancel := context.WithCancel(context.Background())
	cancel()
	if err := checkCommitSignalBoundary(canceled, boundaryGrant); err != context.Canceled {
		t.Fatalf("cancellation after durable intent was ignored: %v", err)
	}
	check("UNKNOWN") // consumed intent remains, including after cancellation
	boundaryGrant.expires = time.Now().Add(-time.Second)
	if err := checkCommitSignalBoundary(context.Background(), boundaryGrant); err == nil {
		t.Fatal("grant expiry after durable intent was ignored")
	}
	check("UNKNOWN") // expired grant cannot refund consumed intent
	nativeExit := supervisor.OwnedCoreNativeExit{TerminateCalls: 1, ExitObserved: true, ExitCode: supervisor.OwnedCoreMaintenanceExitCode}
	if _, err := writeOwnedJournal(out, "NATIVE_EXIT.json", ownedExitRecord{Schema: "researchos.owned-core-stop-native-exit.v1", OperationID: op, PlanSHA256: planSHA, GrantSHA256: grantSHA, Native: nativeExit}); err != nil {
		t.Fatal(err)
	}
	check("UNKNOWN") // after exit, before immutable terminal
	final := stopResult(plan, planSHA, "STOP_OBSERVED_MAINTENANCE_BLOCKED", "lease unavailable")
	final.StopGrantSHA256 = grantSHA
	final.Native = &nativeExit
	if _, err := writeOwnedJournal(out, "POSTFLIGHT.json", final); err != nil {
		t.Fatal(err)
	}
	check(final.State)
	check(final.State) // repeat inspection is identical and observation only
	if err := os.WriteFile(filepath.Join(out, "POSTFLIGHT.json"), []byte("{\"schema\":\"forged\"}"), 0600); err != nil {
		t.Fatal(err)
	}
	docs := &heldDocuments{}
	_, err = readExistingStopOutcome(docs, g, plan, planSHA)
	docs.Close()
	if err == nil {
		t.Fatal("corrupt terminal receipt silently fell back to earlier journal")
	}
}
