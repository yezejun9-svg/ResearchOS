func f08AbsenceDecode(t *testing.T, rows []any) *f08Sources {
	t.Helper(); s,e:=f08DecodeSources(rows); if e!=nil { t.Fatal(e) }
	v:=f08AbsenceValidator(t)
	for _,r:=range s.rows["domain_versions"] { b,_,e:=f08RawObject(r[4]);if e!=nil{t.Fatal(e)}; if e=v.entities.ValidateDefInput(r[1].(string),b);e!=nil{t.Fatal("fixture schema",e)} }
	return s
}
func f08AbsenceRow(rows []any, category string) []any {

func TestF08NoSessionAbsenceFullTypedSources(t *testing.T) {
	v:=f08AbsenceValidator(t)
	for _, other:=range []bool{false,true} {
		t.Run(fmt.Sprintf("positive_unrelated_history_%v",other),func(t *testing.T){s:=f08AbsenceDecode(t,f08AbsenceFixture(t,other));q,e:=v.queueAdmission(s,"p","A");if e!=nil||q["variant"]!="NO_SESSION"{t.Fatal(q,e)};final,e:=v.emptyQueueFinal(s,"p","A",q);if e!=nil||!f08EmptyArray(final["entry_refs"])||!f08EmptyArray(final["receipt_refs"]){t.Fatal(final,e)}})
	}
	t.Run("positive_unrelated_receipt_not_current_pointer",func(t *testing.T){rows:=f08AbsenceFixture(t,true);f08AbsenceRow(rows,"queue")[16]=nil;s:=f08AbsenceDecode(t,rows);q,e:=v.queueAdmission(s,"p","A");if e!=nil{t.Fatal(e)};if _,e=v.emptyQueueFinal(s,"p","A",q);e!=nil{t.Fatal(e)}})
	for _,state:=range []string{"ENQUEUED","SENDING","SDK_ACCEPTED","UNKNOWN","CONSUMED","DROP_REQUESTED","DROPPED"} {
		t.Run("cross_task_generation_"+state,func(t *testing.T){rows:=f08AbsenceFixture(t,true);q:=f08AbsenceRow(rows,"queue");q[4]="A";q[14]=state;s:=f08AbsenceDecode(t,rows);if _,e:=v.queueAdmission(s,"p","A");e==nil{t.Fatal("generation FK hid Task A queue")};before,e:=v.queueAdmission(f08AbsenceDecode(t,f08AbsenceFixture(t,false)),"p","A");if e!=nil{t.Fatal(e)};if _,e=v.emptyQueueFinal(s,"p","A",before);e==nil{t.Fatal("final omitted Task A history")}})
	}
	cases:=[]struct{name string; mutate func([]any)}{
		{"receipt_task_changed_rehash",func(rows []any){f08AbsenceReceiptBody(t,rows,map[string]any{"task_id":"A","from":"ENQUEUED","to":"DROPPED"})}},
		{"orphan_generation_rehash",func(rows []any){f08AbsenceRow(rows,"receipts")[2]="missing-generation"}},
		{"orphan_command",func(rows []any){f08AbsenceRow(rows,"receipts")[3]="missing-command"}},
		{"null_command",func(rows []any){f08AbsenceRow(rows,"receipts")[3]=nil}},
		{"cross_project_receipt",func(rows []any){f08AbsenceRow(rows,"receipts")[1]="other-project"}},
		{"cross_session_queue",func(rows []any){f08AbsenceRow(rows,"queue")[3]="other-session"}},
		{"cross_goal_queue",func(rows []any){f08AbsenceRow(rows,"queue")[5]="goal-A"}},
		{"receipt_cycle",func(rows []any){f08AbsenceRow(rows,"receipts")[7]="receipt-B"}},
		{"missing_predecessor",func(rows []any){f08AbsenceRow(rows,"receipts")[7]="missing-receipt"}},
		{"receipt_content_hash",func(rows []any){f08AbsenceRow(rows,"receipts")[6]="sha256:"+strings.Repeat("0",64)}},
		{"queue_content_hash",func(rows []any){f08AbsenceRow(rows,"queue")[13]="sha256:"+strings.Repeat("0",64)}},
		{"invalid_transition_rehash",func(rows []any){f08AbsenceReceiptBody(t,rows,map[string]any{"task_id":"B","from":"SENDING","to":"DROPPED"})}},
		{"unknown_kind_not_absence",func(rows []any){f08AbsenceRow(rows,"receipts")[4]="UNCLASSIFIED"}},
	}
	for _,tc:=range cases {t.Run(tc.name,func(t *testing.T){rows:=f08AbsenceFixture(t,true);f08AbsenceRow(rows,"queue")[16]=nil;tc.mutate(rows);s:=f08AbsenceDecode(t,rows);if _,e:=v.queueAdmission(s,"p","A");e==nil{t.Fatal("admission accepted corrupt association")};before,e:=v.queueAdmission(f08AbsenceDecode(t,f08AbsenceFixture(t,false)),"p","A");if e!=nil{t.Fatal(e)};if _,e=v.emptyQueueFinal(s,"p","A",before);e==nil{t.Fatal("final accepted corrupt association")}})}
	t.Run("receipt_only_task_A_correct_raw_hash",func(t *testing.T){rows:=f08AbsenceFixture(t,true);filtered:=[]any{};for _,x:=range rows{if x.([]any)[0]!="queue"{filtered=append(filtered,x)}};f08AbsenceReceiptBody(t,filtered,map[string]any{"task_id":"A","from":"SENDING","to":"DROP_REQUESTED"});s:=f08AbsenceDecode(t,filtered);if _,e:=v.queueAdmission(s,"p","A");e==nil{t.Fatal("receipt-only Task A hidden")};before,e:=v.queueAdmission(f08AbsenceDecode(t,f08AbsenceFixture(t,false)),"p","A");if e!=nil{t.Fatal(e)};if _,e=v.emptyQueueFinal(s,"p","A",before);e==nil{t.Fatal("receipt-only final")}})
}
