type f08Scalar func(*sqlite.FunctionContext, []driver.Value) (driver.Value, error)

// The private schema is captured by the three closures. It is not exported or
// installed into backupConfigurationSchemaDoc or any Host/Service/cache.
// This initializer is a candidate, not an assertion that registration was run.
func init() {
	functions, err := f08BuildCallbacks(backupConfigurationProtocol, backupConfigurationDomain)
	if err != nil {
		panic("F08_SCHEMA_INITIALIZATION_FAILED")
	}
	for _, registration := range []struct {
		name string
		arity int32
		call f08Scalar
	}{
		{"researchos_f08_cancel_request_v1", 2, functions[0]},
		{"researchos_f08_cancel_response_v1", 5, functions[1]},
		{"researchos_f08_proof_v1", 2, functions[2]},
	} {
		if err := sqlite.RegisterFunction(registration.name, &sqlite.FunctionImpl{
			NArgs: registration.arity,
			Deterministic: true,
			Innocuous: true, // DOES NOT COMPILE against the unpatched pinned driver.
			Scalar: registration.call,
			VolatileArgs: false,
		}); err != nil {
			panic("F08_FUNCTION_REGISTRATION_FAILED")
		}
	}
}

func f08BuildCallbacks(protocol, external []byte) ([3]f08Scalar, error) {
	var zero [3]f08Scalar
	p, e := bytes.Clone(protocol), bytes.Clone(external)
	if f08RawHash(p) != f08ProtocolHash || f08RawHash(e) != f08ExternalHash {
		return zero, f08Schema
	}
	doc, err := LoadSchemaWithExternals(p, map[string][]byte{"domain.schema.json": e})
	if err != nil {
		return zero, f08Schema
	}
	validator, err := f08NewProofValidator(doc, e)
	if err != nil { return zero, err }
	return [3]f08Scalar{
		f08Boundary(func(args []driver.Value) (driver.Value, error) {
			if len(args) != 2 { return nil, f08Arguments }
			header, err := f08Blob(args[0]); if err != nil { return nil, err }
			identity, err := f08Blob(args[1]); if err != nil { return nil, err }
			if _, err = f08Request(doc, header, identity); err != nil { return nil, err }
			return int64(1), nil
		}),
		f08Boundary(func(args []driver.Value) (driver.Value, error) {
			if len(args) != 5 { return nil, f08Arguments }
			header, err := f08Blob(args[0]); if err != nil { return nil, err }
			binding, err := f08Blob(args[1]); if err != nil { return nil, err }
			var old []byte
			if args[2] != nil { old, err = f08Blob(args[2]); if err != nil { return nil, err } }
			current, err := f08Blob(args[3]); if err != nil { return nil, err }
			mode, ok := args[4].(int64); if !ok { return nil, f08Arguments }
			if err = f08Response(doc, header, binding, old, current, mode); err != nil { return nil, err }
			return int64(1), nil
		}),
		f08Boundary(func(args []driver.Value) (driver.Value, error) {
			if len(args) != 2 { return nil, f08Arguments }
			mode, ok := args[0].(int64); if !ok { return nil, f08Arguments }
			input, err := f08Blob(args[1]); if err != nil { return nil, err }
			switch mode {
			case -2:
				// This mode accepts any BLOB bytes, including empty/NUL/non-UTF8.
				// Its digest has no authority to authorize a database mutation.
				return f08RawHash(input), nil
			case -1:
				_, canonicalBytes, err := f08Strict(input)
				if err != nil { return nil, err }
				return f08RawHash(canonicalBytes), nil
			case 9:
				// SQL view's type-violation branch is unconditional refusal.
				return nil, f08StorageClass
			case 0, 1, 2, 3, 4, 5, 6, 7, 8:
				return validator.evaluate(mode, input)
			default:
				return nil, f08Mode
			}
		}),
	}, nil
}

// No FunctionContext, SQL argument, or input-derived closure survives a call.
// Recoverable programmer/parser panic becomes a fixed error, never SQL NULL
// with nil error. Go fatal OOM/stack exhaustion is not claimed recoverable.
func f08Boundary(call func([]driver.Value) (driver.Value, error)) f08Scalar {
	return func(_ *sqlite.FunctionContext, args []driver.Value) (result driver.Value, err error) {
		defer func() {
			if recover() != nil { result, err = nil, f08Panic }
		}()
		result, err = call(args)
		if err != nil { return nil, err }
		if result == nil { return nil, f08Panic }
		return result, nil
	}
}

func f08Blob(value driver.Value) ([]byte, error) {
	b, ok := value.([]byte)
	if !ok || b == nil { return nil, f08Arguments }
	return b, nil
}

func f08RawHash(raw []byte) string {
	h := sha256.Sum256(raw)
	return "sha256:" + hex.EncodeToString(h[:])
}

// Strict validation precedes map/struct parsing. Preserve the original numeric
// lexemes with UseNumber and compare values to canonical output before hashing.
// There is no response/witness byte ceiling invented here. Future qualification
// must measure existing parser/number-library resource behaviour explicitly.
func f08Strict(raw []byte) (any, []byte, error) {
	if !utf8.Valid(raw) || len(raw) == 0 { return nil, nil, f08JSON }
	c, err := canonical.JSON(raw)
	if err != nil { return nil, nil, f08JSON }
	decode := func(b []byte) (any, error) {
		d := json.NewDecoder(bytes.NewReader(b)); d.UseNumber()
		var v any
		if err := d.Decode(&v); err != nil { return nil, f08JSON }
		var trailing any
		if err := d.Decode(&trailing); !errors.Is(err, io.EOF) { return nil, f08JSON }
		return v, nil
	}
	v, err := decode(raw); if err != nil { return nil, nil, err }
	w, err := decode(c); if err != nil || !sameNumbers(v, w) { return nil, nil, f08JSON }
	return v, c, nil
}

func f08Object(v any, keys ...string) (map[string]any, error) {
	m, ok := v.(map[string]any)
	if !ok || len(m) != len(keys) { return nil, f08Shape }
	for _, key := range keys { if _, ok := m[key]; !ok { return nil, f08Shape } }
	return m, nil
}

func f08String(v any) (string, error) {
	s, ok := v.(string)
	if !ok || s == "" || !utf8.ValidString(s) { return "", f08Shape }
	return s, nil
}

func f08Integer(v any, minimum int64) (int64, error) {
	n, ok := v.(json.Number)
	if !ok { return 0, f08Shape }
	r, ok := new(big.Rat).SetString(string(n))
	if !ok || !r.IsInt() || !r.Num().IsInt64() { return 0, f08Shape }
	i := r.Num().Int64()
	if i < minimum { return 0, f08Shape }
	return i, nil
}

func f08HashString(v any) (string, error) {
	s, err := f08String(v)
	if err != nil || len(s) != 71 || !strings.HasPrefix(s, "sha256:") { return "", f08Shape }
	b, err := hex.DecodeString(s[7:])
	if err != nil || len(b) != 32 || hex.EncodeToString(b) != s[7:] { return "", f08Shape }
	return s, nil
}

func f08Encode(v any) ([]byte, error) {
	b, err := json.Marshal(v)
	if err != nil { return nil, f08JSON }
	return b, nil
}

func f08ValueHash(v any) (string, error) {
	b, err := f08Encode(v); if err != nil { return "", err }
	_, c, err := f08Strict(b); if err != nil { return "", err }
	return f08RawHash(c), nil
}

type f08CancelRequest struct {
	envelope *Envelope
	identity map[string]any
	command string
	project string
	session string
	client string
	target string
	principal string
	requestHash string

// f08Request is not authentication. The immutable SQL row producer supplies
// principal; this function checks the exact original tuple, never caller role.
func f08Request(doc *SchemaDoc, header, identityRaw []byte) (f08CancelRequest, error) {
	var zero f08CancelRequest
	if doc == nil || len(header) == 0 || len(header) > f08HeaderMaximum { return zero, f08Header }
	v, _, err := f08Strict(header); if err != nil { return zero, err }
	h, err := f08Object(v, "schema", "encoding", "raw_envelope_base64"); if err != nil { return zero, err }
	encoded, ok := h["raw_envelope_base64"].(string)
	if !ok || h["schema"] != f08HeaderSchema || h["encoding"] != "base64" || len(encoded) > f08Base64Maximum { return zero, f08Header }
	stored, err := json.Marshal(struct {
		Schema string `json:"schema"`
		Encoding string `json:"encoding"`
		Raw string `json:"raw_envelope_base64"`
	}{f08HeaderSchema, "base64", encoded})
	if err != nil || !bytes.Equal(stored, header) { return zero, f08Header }
	raw, err := base64.StdEncoding.Strict().DecodeString(encoded)
	if err != nil || len(raw) == 0 || len(raw) > f08EnvelopeMaximum || base64.StdEncoding.EncodeToString(raw) != encoded { return zero, f08Header }
	v, _, err = f08Strict(raw); if err != nil { return zero, err }
	em, err := f08Object(v, "protocol_version", "command_id", "client_id", "project_id", "session_id", "expected_revision", "writer_epoch", "method", "payload")
	if err != nil { return zero, err }
	if em["protocol_version"] != ProtocolV2 || em["method"] != "session.control" { return zero, f08Binding }
	if err = doc.ValidateDefInput("CommandEnvelope", v); err != nil { return zero, f08Schema }
	payload, err := f08Object(em["payload"], "action", "target_id", "reason"); if err != nil { return zero, err }
	if payload["action"] != "CANCEL" { return zero, f08Binding }
	if err = doc.ValidateDefInput("ControlPayload", payload); err != nil { return zero, f08Schema }
	// Restrict method before DecodeEnvelope so unrelated host configuration or
	// method-specific branches cannot enter the callback's reachable graph.
	env, err := DecodeEnvelope(raw); if err != nil { return zero, f08Schema }
	q, err := env.ContentHash(); if err != nil { return zero, f08Binding }
	iv, _, err := f08Strict(identityRaw); if err != nil { return zero, err }
	im, err := f08Object(iv, "format", "principal", "project", "operation", "key", "request_hash", "command_id", "method", "client_id", "session_id", "writer_epoch", "expected_revision", "target_id")
	if err != nil { return zero, err }
	if im["format"] != f08IdentityFormat || im["method"] != "session.control" || im["operation"] != "v2:session.control" || im["request_hash"] != q { return zero, f08Binding }
	for _, field := range []string{"format", "principal", "project", "operation", "key", "request_hash", "command_id", "method", "client_id", "session_id", "target_id"} {
		if _, err = f08String(im[field]); err != nil { return zero, err }
	}
	for ik, ek := range map[string]string{"project":"project_id", "command_id":"command_id", "client_id":"client_id", "session_id":"session_id", "method":"method"} {
		if im[ik] != em[ek] { return zero, f08Binding }
	}
	if im["key"] != im["command_id"] || im["target_id"] != payload["target_id"] { return zero, f08Binding }
	revision, err := f08Integer(em["expected_revision"], 1); if err != nil { return zero, err }
	epoch, err := f08Integer(em["writer_epoch"], 1); if err != nil { return zero, err }
	ir, err := f08Integer(im["expected_revision"], 1); if err != nil || ir != revision { return zero, f08Binding }
	ie, err := f08Integer(im["writer_epoch"], 1); if err != nil || ie != epoch { return zero, f08Binding }
	return f08CancelRequest{env, im, im["command_id"].(string), im["project"].(string), im["session_id"].(string), im["client_id"].(string), im["target_id"].(string), im["principal"].(string), q, f08RawHash(header), revision, epoch}, nil
}

type f08ValidatedResponse struct {
	revision int64
	result map[string]any
	snapshot map[string]any
	eventSequence int64
}

func f08ResponseBody(doc *SchemaDoc, req f08CancelRequest, raw []byte, status string) (f08ValidatedResponse, error) {
	var zero f08ValidatedResponse
	v, _, err := f08Strict(raw); if err != nil { return zero, err }
	m, err := f08Object(v, "protocol_version", "command_id", "method", "status", "session_id", "revision", "error_code", "error", "result")
	if err != nil { return zero, err }
	if m["protocol_version"] != ProtocolV2 || m["command_id"] != req.command || m["method"] != "session.control" || m["status"] != status || m["session_id"] != req.session || m["error"] != nil || m["error_code"] != nil { return zero, f08Binding }
	if err = doc.ValidateDef("CommandResponse", v); err != nil { return zero, f08Schema }
	parsed, err := DecodeResponse(raw); if err != nil { return zero, f08Schema }
	if err = ValidateResultAgainstMethod("session.control", parsed, doc); err != nil { return zero, f08Schema }
	rev, err := f08Integer(m["revision"], req.revision); if err != nil { return zero, err }
	result, ok := m["result"].(map[string]any); if !ok { return zero, f08Shape }
	if status == "ACKNOWLEDGED" {
		if err = doc.ValidateDef("PendingResult", result); err != nil { return zero, f08Schema }
		if result["operation_id"] != "op-"+req.command { return zero, f08Binding }
		sequence, err := f08Integer(result["after_sequence"], 0); if err != nil { return zero, err }
		return f08ValidatedResponse{revision: rev, result: result, eventSequence: sequence}, nil
	}
	if status != "COMPLETED" { return zero, f08Mode }
	if err = doc.ValidateDef("ControlResult", result); err != nil { return zero, f08Schema }
	if result["action"] != "CANCEL" || result["target_id"] != req.target || result["control_request_id"] != "ctrl-"+req.command || result["settled"] != true || result["children_stopped"] != true || result["queue_empty"] != true { return zero, f08Binding }
	snapshot, ok := result["snapshot"].(map[string]any); if !ok { return zero, f08Shape }
	if err = doc.ValidateDef("SessionSnapshot", snapshot); err != nil { return zero, f08Schema }
	if snapshot["session_id"] != req.session || snapshot["project_id"] != req.project || snapshot["protocol_version"] != ProtocolV2 { return zero, f08Binding }
	sr, err := f08Integer(snapshot["revision"], 1); if err != nil || sr != rev { return zero, f08Binding }
	sequence, err := f08Integer(snapshot["event_sequence"], 0); if err != nil { return zero, err }
	pending, ok := snapshot["pending_command_ids"].([]any); if !ok { return zero, f08Shape }
	for _, id := range pending { if id == req.command { return zero, f08Binding } }
	return f08ValidatedResponse{rev, result, snapshot, sequence}, nil
}

// This validates local wire bindings. It cannot establish that response_proof
// originated in real SQL source rows: proof modes 0/1/5/6 must recompute those
// rows and preserve every other pending command before calling this helper.
// Until that evaluator exists, this helper's success NEVER grants an UPDATE.
func f08Response(doc *SchemaDoc, header, binding, old, current []byte, mode int64) error {
	if mode < 0 || mode > 2 { return f08Mode }
	if (mode == 2) != (old != nil) { return f08Arguments }
	v, _, err := f08Strict(binding); if err != nil { return err }
	m, err := f08Object(v, "identity", "response_proof"); if err != nil { return err }
	identity, err := f08Encode(m["identity"]); if err != nil { return err }
	req, err := f08Request(doc, header, identity); if err != nil { return err }
	proof, err := f08Object(m["response_proof"], "admitted_root_hash", "original_response_digest", "final_root_hash", "final_response_digest", "final_snapshot_binding")
	if err != nil { return err }
	if _, err = f08HashString(proof["admitted_root_hash"]); err != nil { return err }
	originalDigest, err := f08HashString(proof["original_response_digest"]); if err != nil { return err }
	if mode == 0 {
		if proof["final_root_hash"] != nil || proof["final_response_digest"] != nil || proof["final_snapshot_binding"] != nil || originalDigest != f08RawHash(current) { return f08Binding }
		_, err = f08ResponseBody(doc, req, current, "ACKNOWLEDGED")
		return err
	}
	if _, err = f08HashString(proof["final_root_hash"]); err != nil { return err }
	finalDigest, err := f08HashString(proof["final_response_digest"]); if err != nil || finalDigest != f08RawHash(current) { return f08Binding }
	response, err := f08ResponseBody(doc, req, current, "COMPLETED"); if err != nil { return err }
	if mode == 1 {
		if originalDigest != finalDigest { return f08Binding }
	} else {
		if originalDigest != f08RawHash(old) || bytes.Equal(old, current) { return f08Binding }
		prior, err := f08ResponseBody(doc, req, old, "ACKNOWLEDGED"); if err != nil { return err }
		if response.revision < prior.revision || response.eventSequence < prior.eventSequence { return f08Binding }
	}
	sb, err := f08Object(proof["final_snapshot_binding"], "session_id", "project_id", "revision", "event_sequence", "pending_command_ids_digest")
	if err != nil { return err }
	if sb["session_id"] != req.session || sb["project_id"] != req.project { return f08Binding }
	sr, err := f08Integer(sb["revision"], 1); if err != nil || sr != response.revision { return f08Binding }
	es, err := f08Integer(sb["event_sequence"], 0); if err != nil || es != response.eventSequence { return f08Binding }
	digest, err := f08ValueHash(response.snapshot["pending_command_ids"]); if err != nil { return err }
	if sb["pending_command_ids_digest"] != digest { return f08Binding }
	return nil
}

// Structural precondition only. This does not turn an arbitrary source bundle
// into authority. Complete mode-specific evaluation is still missing; see the
// explicit refusal in f08BuildCallbacks and mode-implementation-matrix.json.
func f08ProofFrame(mode int64, raw []byte) error {
	v, _, err := f08Strict(raw); if err != nil { return err }
	m, err := f08Object(v, "action", "row", "old", "sources"); if err != nil { return err }
	action, err := f08String(m["action"]); if err != nil { return err }
	allowed := false
	switch mode {
	case 0: allowed = action == "admitted.insert"
	case 1: allowed = action == "final.insert"
	case 2: allowed = action == "origin.insert"
	case 3: allowed = false // No mode3 action literal exists in the frozen SQL; caller contract must be supplied.
	case 4: allowed = action == "dispatch.update"
	case 5: allowed = action == "journal.insert" || action == "journal.update" || action == "commit.insert"
	case 6: allowed = action == "control.insert" || action == "control.update"
	case 7: allowed = action == "admitted_member.insert" || action == "final_member.insert" || action == "pin.insert"
	case 8: allowed = action == "origin.derive"
	}
	if !allowed { return f08Mode }
	row, ok := m["row"].([]any); if !ok || len(row) == 0 { return f08Shape }
	if strings.HasSuffix(action, ".update") {
		old, ok := m["old"].([]any); if !ok || len(old) != len(row) { return f08Shape }
	} else if m["old"] != nil { return f08Shape }
	sources, ok := m["sources"].([]any); if !ok { return f08Shape }
	for _, item := range sources {
		tuple, ok := item.([]any); if !ok || len(tuple) != 2 { return f08Shape }
		if _, err := f08String(tuple[0]); err != nil { return err }
		columns, ok := tuple[1].([]any); if !ok || len(columns) == 0 { return f08Shape }
	}
	return nil
}

// SQL tuple lexicographic ordering compares scalar columns, not encoded JSON
// strings (which would put numeric revision10 before2). String order is UTF8.
func f08ScalarCompare(a,b any) int {

}

	for _,row:=range rows{if e:=v.materialization(s,action,row);e!=nil{return e}}
	pins:=s.selectRows("pins",[]int{0,1,2,3,4},append(append([]any(nil),g.row[:4]...),g.phase))
	if len(pins)!=len(g.pins){return f08Binding};for _,row:=range pins{if e:=v.materialization(s,"pin.insert",row);e!=nil{return e}}
	return nil
}

}

// Source-derived original request and NO_RUN admission. Native admission is
// routed to the separately implemented origin checker, never inferred from a
// stage label or absence of a dispatch row.
func (v *f08ProofValidator) groupRequest(s *f08Sources,g *f08Group)(f08CancelRequest,error){
	var zero f08CancelRequest
	if g.phase!="ADMITTED"{return zero,f08Mode}
	r:=g.root;binding,e:=f08Object(r["session_binding"],"session_id","accepted_revision","writer_client_id","writer_epoch","lease_expires_at","event_watermark");if e!=nil{return zero,e}
	target,e:=f08Object(r["target_binding"],"kind","id","version_ref");if e!=nil{return zero,e}
	id,e:=s.one("identities",[]int{0,1},[]any{g.row[0],g.row[2]});if e!=nil{return zero,e}
	if id[2]!=ProtocolV2 || id[3]!=g.row[1] || id[4]!="session.control" || id[5]!=g.row[4] || id[6]!=g.row[3]{return zero,f08Binding}
	header,e:=f08Hex(g.row[5]);if e!=nil{return zero,e}
	identity:=map[string]any{"format":f08IdentityFormat,"principal":id[0],"project":id[3],"operation":id[5],"key":id[1],"request_hash":id[6],"command_id":id[1],"method":id[4],"client_id":r["client_id"],"session_id":binding["session_id"],"writer_epoch":binding["writer_epoch"],"expected_revision":binding["accepted_revision"],"target_id":target["id"]}
	b,e:=f08Encode(identity);if e!=nil{return zero,e};req,e:=f08Request(v.protocol,header,b);if e!=nil{return zero,e}
	if r["private_header_digest"]!=req.headerDigest || binding["writer_client_id"]!=req.client{return zero,f08Binding}
	return req,nil
}
// Pure absence check over the complete typed source vector. It does not prove
// SDK settlement and does not infer absence from a missing current Session.
// A retained fence is the original Task/Session association; a Session's current
// task_id may have advanced, so it is not used to reassign old queue history.
func (v *f08ProofValidator) noSessionQueueHistory(s *f08Sources, project, task string) error {
	if len(s.selectRows("sessions", []int{1, 6}, []any{project, task})) != 0 || len(s.selectRows("fences", []int{0, 1}, []any{project, task})) != 0 { return f08Binding }
	for _, q := range s.rows["queue"] {
		fs := s.selectRows("fences", []int{3}, []any{q[6]})
		relevant := q[1] == project
		for _, f := range fs { relevant = relevant || f[0] == project }
		if !relevant { continue }
		// Even DROPPED/CONSUMED history belongs to its original Task. SQL22's
		// generation-only FK cannot justify the other denormalized columns.
		if q[1] == project && q[4] == task { return f08Binding }
		if len(fs) != 1 { return f08Binding }; f := fs[0]
		if f[0] != q[1] || f[1] != q[4] || f[2] != q[3] { return f08Binding }
		if _, e := s.one("sessions", []int{0, 1}, []any{q[3], q[1]}); e != nil { return e }
		owner, e := v.head(s, project, "Task", q[4].(string)); if e != nil || owner.body["goal_id"] != q[5] { return f08Binding }
		if _, e = v.head(s, project, "Goal", q[5].(string)); e != nil { return e }
		n, e := f08Integer(q[7], 1); if e != nil { return e }
		a, e := f08Integer(f[5], n); if e != nil || n > a { return f08Binding }
		_, raw, e := f08RawObject(q[12]); if e != nil || f08RawHash(raw) != q[13] { return f08Binding }
		if q[16] != nil {
			if _, e = s.one("receipts", []int{0,1,2,3}, []any{q[16],q[1],q[6],q[2]}); e != nil { return e }
		}
	}
	// Receipts have no FK. Every same-project receipt must be attributable to
	// an original, different Task, rather than silently discarded as unrelated.
	for _, r := range s.rows["receipts"] {
		fs := s.selectRows("fences", []int{3}, []any{r[2]})
		relevant := r[1] == project
		for _, f := range fs { relevant = relevant || f[0] == project }
		if !relevant { continue }
		body, raw, e := f08RawObject(r[5]); if e != nil || f08RawHash(raw) != r[6] { return f08Binding }
		if len(fs) != 1 || fs[0][0] != r[1] || fs[0][1] == task { return f08Binding }; f := fs[0]
		if _, e = s.one("sessions", []int{0,1}, []any{f[2],r[1]}); e != nil { return e }
		// Only the exact, known CANCEL_FENCE producer is classified here. Other
		// receipt kinds need their own complete association proof; a kind label,
		// NULL command, or unknown payload cannot confer an absence exemption.
		if r[4] != "CANCEL_FENCE" { return f08NativeProofGap }
		if _, e = f08Object(body, "task_id", "from", "to"); e != nil { return e }
		if body["task_id"] != f[1] || body["task_id"] == task || r[3] == nil { return f08Binding }
		q, e := s.one("queue", []int{1,2,6}, []any{r[1],r[3],r[2]}); if e != nil { return e }
		if q[3] != f[2] || q[4] != f[1] { return f08Binding }
		from, to := body["from"], body["to"]
		if !(from == "ENQUEUED" && to == "DROPPED" || (from == "SENDING" || from == "SDK_ACCEPTED") && to == "DROP_REQUESTED") { return f08Binding }
		if r[7] != nil {
			// Resolve the entire retained chain, including nullable/duplicate IDs,
			// cycle and cross-command/generation substitutions. This proves only
			// association, never the SDK semantic meaning of a predecessor.
			seen := map[string]bool{}
			id, e := f08String(r[0]); if e != nil { return e }; seen[id] = true
			previous := r[7]
			for previous != nil {
				id, e = f08String(previous); if e != nil || seen[id] { return f08Binding }; seen[id] = true
				p, e := s.one("receipts", []int{0,1,2,3}, []any{id,r[1],r[2],r[3]}); if e != nil { return e }
				_, bytes, e := f08RawObject(p[5]); if e != nil || f08RawHash(bytes) != p[6] { return f08Binding }
				previous = p[7]
			}
		}
	}
	return nil
}

func (v *f08ProofValidator) queueAdmission(s *f08Sources,project,task string)(map[string]any,error){
	q:=map[string]any{"variant":"NO_SESSION","session_id":nil,"generation":nil,"revision":nil,"state":nil,"accepted_through":nil,"sealed_through":nil,"run_id":nil,"runtime_epoch":nil,"dispatch_key":nil,"manifest_hash":nil,"process_identity":nil,"restore_fenced":nil}
	sessions:=s.selectRows("sessions",[]int{1,6},[]any{project,task})
	fences:=s.selectRows("fences",[]int{0,1},[]any{project,task})
	if len(sessions)==0{if e:=v.noSessionQueueHistory(s,project,task);e!=nil{return nil,e};return q,nil}
	if len(sessions)!=1{return nil,f08Binding};session:=sessions[0];q["session_id"]=session[0]
	if len(fences)==0{q["variant"]="REAL_SESSION_NO_ORIGINAL_FENCE";return q,nil}
	if len(fences)!=1 || fences[0][2]!=session[0]{return nil,f08Binding}
	f:=fences[0];q["variant"]="ORIGINAL_FENCE"
	for key,i:=range map[string]int{"generation":3,"revision":13,"state":4,"accepted_through":5,"sealed_through":6,"run_id":7,"runtime_epoch":8,"dispatch_key":9,"manifest_hash":10,"restore_fenced":12}{q[key]=f[i]}
	// The registry fixes actual fence tuple positions; process_identity is raw
	// nullable JSON, and revision is the last column (index13 in SQL22).
	q["revision"]=f[13]
	if f[11]!=nil{identity,_,e:=f08JSONHex(f[11]);if e!=nil{return nil,e};q["process_identity"]=identity}
	accepted,e:=f08Integer(f[5],0);if e!=nil{return nil,e};sealed,e:=f08Integer(f[6],0);if e!=nil || sealed>accepted{return nil,f08Binding}
	for _,entry:=range s.selectRows("queue",[]int{1,4,6},[]any{project,task,f[3]}){
		seq,e:=f08Integer(entry[7],1);if e!=nil || seq>accepted || entry[3]!=session[0]{return nil,f08Binding}
		_,contentRaw,e:=f08JSONHex(entry[12]);if e!=nil{return nil,e};if f08RawHash(contentRaw)!=entry[13]{return nil,f08Binding}
		if entry[16]!=nil{
			r,e:=s.one("receipts",[]int{0,1,2,3},[]any{entry[16],project,f[3],entry[2]});if e!=nil{return nil,e}
			_,receiptRaw,e:=f08JSONHex(r[5]);if e!=nil || f08RawHash(receiptRaw)!=r[6]{return nil,f08Binding}
		}
	}
	return q,nil
}
func (v *f08ProofValidator) noRun(s *f08Sources,task f08Version)error{
	if task.body["active_run_id"]!=nil{return f08Binding}
	for _,row:=range s.rows["domain_versions"]{
		if row[0]!=task.project || row[1]!="Run"{continue}
		body,_,e:=f08RawObject(row[4]);if e!=nil{return e};if body["task_id"]==task.id{return f08Binding}
	}
	for _,row:=range s.rows["dispatches"]{
		if row[0]!=task.project{continue};body,_,e:=f08RawObject(row[4]);if e!=nil{return e};if body["task_id"]==task.id{return f08Binding}
	}
	// Effects bind a Run, not a Task. Every same-project effect must resolve a
	// real Run version; an orphan effect cannot be ignored as "not this Task".
	for _,row:=range s.rows["effects"]{
		if row[0]!=task.project{continue};runs:=s.selectRows("domain_versions",[]int{0,1,2},[]any{task.project,"Run",row[1]});if len(runs)==0{return f08Binding}
		for _,run:=range runs{body,_,e:=f08RawObject(run[4]);if e!=nil || body["task_id"]==task.id{return f08Binding}}
	}
	return nil
}
func (v *f08ProofValidator) admitted(s *f08Sources,row []any,current bool)(*f08Group,error){

	root:=g.root;binding:=root["session_binding"].(map[string]any);target:=root["target_binding"].(map[string]any)

	targetVersion,e:=v.version(s,req.project,target["version_ref"]);if e!=nil || targetVersion.kind!=target["kind"] || targetVersion.id!=target["id"]{return nil,f08Binding}
	if current{
		if len(s.selectRows("admitted",[]int{0,1,2},row[:3]))!=0 || len(s.selectRows("idempotency",[]int{0,1,2,3},[]any{row[0],row[1],row[4],row[2]}))!=0{return nil,f08Binding}

		for key,index:=range map[string]int{"accepted_revision":2,"writer_client_id":9,"writer_epoch":8,"lease_expires_at":10,"event_watermark":11}{if !f08Equal(binding[key],session[index]){return nil,f08Binding}}

		if targetVersion.kind=="Goal" && session[5]!=targetVersion.id || targetVersion.kind=="Run" && session[7]!=targetVersion.id{return nil,f08Binding}
	}

		for _,o:=range s.rows["domain_objects"]{

		};sort.Slice(wantIDs,func(i,j int)bool{return wantIDs[i].(string)<wantIDs[j].(string)})

		if targetVersion.kind=="Goal" && goal.id!=targetVersion.id || targetVersion.kind=="Run" && targetVersion.body["task_id"]!=task.id{return nil,f08Binding}
		if current{head,e:=v.head(s,req.project,"Task",task.id);if e!=nil || !f08SameVersion(task,head){return nil,f08Binding}}

	}

}
func f08QueueAdmissionShape(value any)error{
	q,e:=f08Object(value,"variant","session_id","generation","revision","state","accepted_through","sealed_through","run_id","runtime_epoch","dispatch_key","manifest_hash","process_identity","restore_fenced");if e!=nil{return e}
	nullKeys:=[]string{"generation","revision","state","accepted_through","sealed_through","run_id","runtime_epoch","dispatch_key","manifest_hash","process_identity","restore_fenced"}
	if q["variant"]=="NO_SESSION"{if q["session_id"]!=nil || !f08NullFields(q,nullKeys...){return f08Binding};return nil}
	if _,e=f08String(q["session_id"]);e!=nil{return e}
	if q["variant"]=="REAL_SESSION_NO_ORIGINAL_FENCE"{if !f08NullFields(q,nullKeys...){return f08Binding};return nil}
	if q["variant"]!="ORIGINAL_FENCE"{return f08Shape}
	for _,key:=range []string{"generation","state"}{if _,e=f08String(q[key]);e!=nil{return e}}
	if _,e=f08Integer(q["revision"],1);e!=nil{return e};accepted,e:=f08Integer(q["accepted_through"],0);if e!=nil{return e};sealed,e:=f08Integer(q["sealed_through"],0);if e!=nil || sealed>accepted{return f08Binding}
	restored,e:=f08Integer(q["restore_fenced"],0);if e!=nil || restored>1{return f08Shape}
	if q["runtime_epoch"]!=nil{if _,e=f08Integer(q["runtime_epoch"],1);e!=nil{return e}}
	for _,key:=range []string{"run_id","dispatch_key"}{if q[key]!=nil{if _,e=f08String(q[key]);e!=nil{return e}}}
	if q["manifest_hash"]!=nil{if _,e=f08HashString(q["manifest_hash"]);e!=nil{return e}}
	if q["process_identity"]!=nil{id,ok:=q["process_identity"].(map[string]any);if !ok{return f08Shape};if e=f08BoundIdentity(id);e!=nil{return e}}
	return nil
}


// Dispatch transition checks operate on actual SQL OLD/NEW/source tuples.

func f08RetainedColumns(old,row []any,mutable ...int)bool{
	allowed:=map[int]bool{};for _,i:=range mutable{allowed[i]=true}
	if len(old)!=len(row){return false};for i:=range row{if !allowed[i] && !f08Equal(old[i],row[i]){return false}};return true
}
func (v *f08ProofValidator) dispatchUpdate(s *f08Sources,old,row []any)error{
	if e:=f08ValidateTuple("dispatches",old);e!=nil{return e};if e:=f08ValidateTuple("dispatches",row);e!=nil{return e}
	actual,e:=s.one("dispatches",[]int{0,1,2},old[:3]);if e!=nil || !f08Equal(actual,old){return f08Binding}
	if !f08RetainedColumns(old,row,6,7,8,9,10,11,13,14){return f08Binding}
	before,e:=f08Identity(old[9]);if e!=nil{return e};after,e:=f08Identity(row[9]);if e!=nil{return e}
	oldSequence,e:=f08Integer(old[10],0);if e!=nil{return e};sequence,e:=f08Integer(row[10],0);if e!=nil{return e}
	state0,state:=old[6].(string),row[6].(string)
	switch {
	case state0=="PREPARED" && state=="STARTED":
		if old[7]!="" || old[8]!="" || len(before)!=0 || oldSequence!=0 || old[11]!="" || row[7]=="" || row[8]!=row[3] || len(after)!=0 || sequence!=0 || row[11]!="" || row[14]!=old[14]{return f08Binding}
		run,e:=v.head(s,row[0].(string),"Run",row[1].(string));if e!=nil{return e}
		if run.body["status"]!="RUNNING" && run.body["status"]!="QUEUED"{return f08Binding}
		if !f08NullFields(run.body,"cancel_requested_at","cancel_confirmed_at","ended_at") || !f08Equal(run.body["lease_epoch"],row[2]){return f08Binding}
		taskID,e:=f08String(run.body["task_id"]);if e!=nil{return e};task,e:=v.head(s,run.project,"Task",taskID);if e!=nil{return e}
		if _,e=v.dispatchManifest(s,row,run,task);e!=nil{return e}
		goalID,e:=f08String(task.body["goal_id"]);if e!=nil{return e};goal,e:=v.head(s,run.project,"Goal",goalID);if e!=nil || goal.body["status"]!="ACTIVE"{return f08Binding}
		for _,member:=range s.rows["final_members"]{
			if member[1]!=row[0]{continue};m,_,e:=f08RawObject(member[8]);if e!=nil{return e}
			if m["proof_kind"]!="NEVER_LAUNCHED_NATIVE"{continue}
			d,ok:=m["dispatch_final_binding"].(map[string]any);if !ok{return f08Shape}
			if d["run_id"]==row[1] && f08Equal(d["epoch"],row[2]){return f08Binding}
		}
		return nil
	case state0=="STARTED" && state=="STARTED":
		if !f08RetainedColumns(old,row,9,13) || old[7]=="" || old[8]!=old[3] || sequence!=0 || row[11]!=""{return f08Binding}
		if len(before)==0 {if len(after)==0{return f08Binding}} else if !f08Equal(before,after){return f08Binding}
		return f08BoundIdentity(after)
	case (state0=="PREPARED" || state0=="STARTED" || state0=="UNKNOWN") && state=="UNKNOWN":
		if !f08RetainedColumns(old,row,6,13,14){return f08Binding};return nil
	case (state0=="STARTED" || state0=="UNKNOWN" || state0=="OBSERVED") && state=="OBSERVED":
		if !f08RetainedColumns(old,row,6,10,11,13) || row[7]=="" || row[8]!=row[3] || len(after)==0 || sequence<oldSequence || sequence<=0{return f08Binding}
		if sequence==oldSequence && row[11]!=old[11]{return f08Binding}
		if _,e=f08HashString(row[11]);e!=nil{return e}
		return f08BoundIdentity(after)
	default:return f08Binding
	}
}

func (v *f08ProofValidator) eventFor(s *f08Sources,aggregate f08Version,eventType string)(f08Version,map[string]any,error){

func (v *f08ProofValidator) effectHistories(s *f08Sources,project,runID string,through int64)([]f08EffectHistory,[]f08Version,error){

	for _,key:=range keys{

		if q["session_id"]!=before["session_id"] || q["variant"]!="ORIGINAL_FENCE"{return nil,f08Binding}
		if q["state"]!="CANCEL_FENCED" && q["state"]!="SEALED"{return nil,f08Binding}
		if !f08Equal(q["restore_fenced"],f08Number(0)) || !f08NullFields(q,"run_id","runtime_epoch","dispatch_key","manifest_hash","process_identity"){return nil,f08Binding}
		accepted,e:=f08Integer(q["accepted_through"],0);if e!=nil{return nil,e};sealed,e:=f08Integer(q["sealed_through"],0);if e!=nil || sealed!=accepted{return nil,f08Binding}
		// Nonempty canonical delivery/receipt histories require the distinct
		// full queue-chain evaluator, never an empty-chain shortcut.
		if accepted!=0 || len(s.selectRows("queue",[]int{1,4},[]any{project,task}))!=0 || len(s.selectRows("receipts",[]int{1,2},[]any{project,q["generation"]}))!=0{return nil,f08NativeProofGap}
		if before["variant"]=="REAL_SESSION_NO_ORIGINAL_FENCE"{
			q["variant"]="CREATED_SAME_TRANSACTION"
		}else if before["variant"]=="ORIGINAL_FENCE"{
			if before["generation"]!=q["generation"] || !f08Equal(before["accepted_through"],f08Number(0)) || !f08Equal(before["sealed_through"],f08Number(0)){return nil,f08Binding}
			prior,e:=f08Integer(before["revision"],1);if e!=nil{return nil,e};now,e:=f08Integer(q["revision"],prior);if e!=nil || now<prior{return nil,f08Binding}
		}else{return nil,f08Shape}
	}
	q["entry_refs"]=[]any{};q["receipt_refs"]=[]any{};return q,nil
}

func (v *f08ProofValidator) finalGroup(s *f08Sources,row []any,current bool)(*f08Group,error){
	g,e:=f08ParseGroup("final",row);if e!=nil{return nil,e}
	ar,e:=s.one("admitted",[]int{0,1,2,3},row[:4]);if e!=nil{return nil,e};a,e:=v.admitted(s,ar,false);if e!=nil{return nil,e}
	if e=v.completeMaterializations(s,a);e!=nil{return nil,e}
	if row[4]!=a.rootHash || len(g.members)!=len(a.members) || g.root["control_request_id"]!="ctrl-"+row[2].(string){return nil,f08Binding}
	req,e:=v.groupRequest(s,a);if e!=nil{return nil,e}
	control,e:=s.one("controls",[]int{0,1,2,5},row[:4]);if e!=nil{return nil,e}
	if control[3]!=req.session || control[4]!=g.root["control_request_id"] || control[6]!=req.target{return nil,f08Binding}
	taskIDs,_,e:=f08JSONHex(control[7]);if e!=nil || !f08Equal(taskIDs,a.root["member_task_ids"]){return nil,f08Binding}
	switch g.root["completion_mode"]{
	case "ACK_TO_COMPLETED":
		original,e:=f08Hex(control[9]);if e!=nil || f08RawHash(original)!=g.root["original_response_digest"]{return nil,f08Binding}
		if _,e=v.responseBody(req,original,"ACKNOWLEDGED");e!=nil{return nil,e}
	case "ORIGINAL_COMPLETED":
		if g.root["original_response_digest"]!=g.root["final_response_digest"]{return nil,f08Binding}
		if control[9]!=nil{raw,e:=f08Hex(control[9]);if e!=nil || f08RawHash(raw)!=g.root["original_response_digest"]{return nil,f08Binding}}
	default:return nil,f08Shape
	}
	for _,key:=range []string{"original_response_digest","final_response_digest"}{if _,e=f08HashString(g.root[key]);e!=nil{return nil,e}}
	sb,e:=f08Object(g.root["final_snapshot_binding"],"session_id","project_id","revision","event_sequence","pending_command_ids_digest");if e!=nil{return nil,e}
	if sb["session_id"]!=req.session || sb["project_id"]!=req.project{return nil,f08Binding}
	if current{
		session,e:=s.one("sessions",[]int{0,1},[]any{req.session,req.project});if e!=nil{return nil,e}
		if !f08Equal(sb["revision"],session[2]) || !f08Equal(sb["event_sequence"],session[11]){return nil,f08Binding}
		h,e:=f08Hash(f08Pending(s,req.project,req.session,req.command));if e!=nil || sb["pending_command_ids_digest"]!=h{return nil,f08Binding}
	}
	all:=map[string]any{}
	for i,item:=range g.members{
		m:=item.(map[string]any);original:=a.members[i].(map[string]any)
		originalTask,e:=v.version(s,req.project,original["task_ref"]);if e!=nil{return nil,e};mh,e:=f08Hash(original);if e!=nil{return nil,e}
		if !f08Equal(m["admitted_member_ref"],f08MemberReference("ADMITTED",originalTask.id,mh)) || !f08Equal(m["task_admitted_ref"],original["task_ref"]) || !f08Equal(m["run_admitted_ref"],original["run_ref"]){return nil,f08Binding}
		if original["dispatch_stage"]!="NO_RUN" || m["proof_kind"]!="NO_EXECUTION"{return nil,f08NativeProofGap}
		if !f08NullFields(m,"run_admitted_ref","run_before_settlement_ref","run_final_ref","dispatch_origin_ref","dispatch_final_binding","observation_ref","run_event_ref","prelaunch_checkpoint_carry") || !f08EmptyArray(m["native_artifact_refs"]) || !f08EmptyArray(m["sdk_receipt_refs"]) || !f08EmptyArray(m["prior_epoch_closure"]){return nil,f08Binding}
		task,e:=v.version(s,req.project,m["task_final_ref"]);if e!=nil || task.kind!="Task" || task.id!=originalTask.id || task.body["goal_id"]!=originalTask.body["goal_id"] || task.revision<originalTask.revision{return nil,f08Binding}
		if task.body["status"]!="CANCELLED" && task.body["status"]!="ACCEPTED"{return nil,f08Binding}
		if current{head,e:=v.head(s,req.project,"Task",task.id);if e!=nil || !f08SameVersion(head,task){return nil,f08Binding};if e=v.noRun(s,task);e!=nil{return nil,e}}
		if !f08Equal(m["effect_closure"],map[string]any{"run_id":nil,"epoch":nil,"histories":[]any{},"non_effect_reservations":[]any{}}){return nil,f08Binding}
		// Historical queue-binding validation still requires an immutable
		// chain evaluator. Do not substitute today's fence for history.
		if !current{return nil,f08NativeProofGap}
		queue,e:=v.emptyQueueFinal(s,req.project,task.id,original["queue_admission"]);if e!=nil || !f08Equal(queue,m["queue_final_binding"]){return nil,f08Binding}
		event,e:=v.eventClaim(s,req.project,m["task_event_ref"],task,"TASK_CHANGED");if e!=nil{return nil,e}
		if task.revision>originalTask.revision && event.body["correlation_id"]!=req.command{return nil,f08Binding}
		goal,e:=v.version(s,req.project,original["goal_ref"]);if e!=nil{return nil,e}
		closure,anchors,e:=v.referenceClosure(s,[]f08Version{originalTask,goal,task,event});if e!=nil || !f08Equal(closure,m["reference_closure"]){return nil,f08Binding}
		for _,anchor:=range anchors{k,_:=f08Key(anchor);all[k]=anchor}
	}
	if !f08Equal(g.pins,f08Pins(f08SortedAnchors(all))){return nil,f08Binding};return g,nil
}

// Sole-journal closure. All raw digests below consume the hex-decoded original
// SQL column bytes; re-encoding a response never substitutes for those bytes.
func (v *f08ProofValidator) journalEnvelope(s *f08Sources,row []any)(map[string]any,map[string]any,error){
	if e:=f08ValidateTuple("idempotency",row);e!=nil{return nil,nil,e}
	if row[2]!="v2:session.control" || row[10]!="text" || row[9]!="text" && row[9]!="blob"{return nil,nil,f08Binding}
	hraw,e:=f08Hex(row[7]);if e!=nil || len(hraw)>f08HeaderMaximum{return nil,nil,f08Header}
	h,_,e:=f08Strict(hraw);if e!=nil{return nil,nil,e};hm,e:=f08Object(h,"schema","encoding","raw_envelope_base64");if e!=nil{return nil,nil,e}
	encoded,ok:=hm["raw_envelope_base64"].(string);if !ok || len(encoded)>f08Base64Maximum || hm["schema"]!=f08HeaderSchema || hm["encoding"]!="base64"{return nil,nil,f08Header}
	canonicalHeader,e:=json.Marshal(struct{Schema string `json:"schema"`;Encoding string `json:"encoding"`;Raw string `json:"raw_envelope_base64"`}{f08HeaderSchema,"base64",encoded});if e!=nil || !bytes.Equal(canonicalHeader,hraw){return nil,nil,f08Header}
	raw,e:=base64.StdEncoding.Strict().DecodeString(encoded);if e!=nil || len(raw)==0 || len(raw)>f08EnvelopeMaximum || base64.StdEncoding.EncodeToString(raw)!=encoded{return nil,nil,f08Header}
	parsed,_,e:=f08Strict(raw);if e!=nil{return nil,nil,e};env,e:=f08Object(parsed,"protocol_version","command_id","client_id","project_id","session_id","expected_revision","writer_epoch","method","payload");if e!=nil{return nil,nil,e}
	if env["method"]!="session.control" || env["project_id"]!=row[1] || env["command_id"]!=row[3] || env["protocol_version"]!=ProtocolV2{return nil,nil,f08Binding}
	if e=v.protocol.ValidateDefInput("CommandEnvelope",env);e!=nil{return nil,nil,f08Schema}
	decoded,e:=DecodeEnvelope(raw);if e!=nil{return nil,nil,f08Schema};q,e:=decoded.ContentHash();if e!=nil || q!=row[4]{return nil,nil,f08Binding}
	id,e:=s.one("identities",[]int{0,1},[]any{row[0],row[3]});if e!=nil{return nil,nil,e}
	if id[2]!=ProtocolV2 || id[3]!=row[1] || id[4]!=env["method"] || id[5]!=row[2] || id[6]!=row[4]{return nil,nil,f08Binding}
	body,rawBody,e:=f08RawObject(row[6]);if e!=nil{return nil,nil,e}
	if e=v.protocol.ValidateDef("CommandResponse",body);e!=nil{return nil,nil,f08Schema}
	response,e:=DecodeResponse(rawBody);if e!=nil{return nil,nil,f08Schema};if e=ValidateResultAgainstMethod("session.control",response,v.protocol);e!=nil{return nil,nil,f08Schema}
	for _,key:=range []string{"protocol_version","command_id","method","session_id"}{if !f08Equal(body[key],env[key]){return nil,nil,f08Binding}}
	return env,body,nil
}

func (v *f08ProofValidator) control(s *f08Sources,action string,old,row []any)error{
	if e:=f08ValidateTuple("controls",row);e!=nil{return e}
	if action=="control.update"{
		if e:=f08ValidateTuple("controls",old);e!=nil{return e};actual,e:=s.one("controls",[]int{0,1,2},old[:3]);if e!=nil || !f08Equal(actual,old){return f08Binding}
		if !f08RetainedColumns(old,row,8,9,10){return f08Binding}
		if f08Equal(row,old){return nil}
		// Existing genuine recovery can move only its pending state. It does
		// not create original membership, change bytes or finalize legacy ACK.
		if old[8]=="REQUESTED" && row[8]=="RECONCILING" && f08RetainedColumns(old,row,8){return nil}
		if old[8]=="CONFIRMED" || old[9]!=nil && old[9]!=row[9] || old[10]!=nil && old[10]!=row[10]{return f08Binding}
	}else if action!="control.insert"{return f08Mode}
	a,e:=f08FindGroup(s,"admitted",f08ControlLocator(row));if e!=nil{return e}
	if _,e=v.admitted(s,a.row,false);e!=nil{return e};if e=v.completeMaterializations(s,a);e!=nil{return e}
	req,e:=v.groupRequest(s,a);if e!=nil{return e}
	tasks,_,e:=f08JSONHex(row[7]);if e!=nil || !f08Equal(tasks,a.root["member_task_ids"]) || row[3]!=req.session || row[4]!="ctrl-"+req.command || row[6]!=req.target{return f08Binding}
	if action=="control.insert"{if row[8]!="REQUESTED" || row[9]!=nil || row[10]!=nil || len(s.selectRows("controls",[]int{0,1,2},row[:3]))!=0{return f08Binding};return nil}
	if row[8]!="REQUESTED" && row[8]!="RECONCILING" && row[8]!="CONFIRMED"{return f08Shape}
	if row[10]==nil{
		if row[8]=="CONFIRMED" || row[8]!=old[8] || old[9]!=nil || row[9]==nil || len(s.selectRows("final",[]int{0,1,2,3},a.row[:4]))!=0{return f08Binding}
		raw,e:=f08Hex(row[9]);if e!=nil{return e};_,e=v.responseBody(req,raw,"ACKNOWLEDGED");return e
	}
	f,e:=f08FindGroup(s,"final",a.row[:4]);if e!=nil{return e};if _,e=v.finalGroup(s,f.row,true);e!=nil{return e}
	if e=v.completeMaterializations(s,f);e!=nil{return e}
	raw,e:=f08Hex(row[10]);if e!=nil{return e};var prior []byte
	if f.root["completion_mode"]=="ACK_TO_COMPLETED"{prior,e=f08Hex(row[9]);if e!=nil{return e}}else if row[9]!=row[10]{return f08Binding}
	if e=v.finalResponse(s,f,raw,prior);e!=nil{return e}
	// Both staged final bytes (same pending state) and the subsequent genuine
	// CONFIRMED transition bind this already-complete group. SQL deferred
	// commit obligation still requires the sole journal completed write.
	if row[8]!=old[8] && row[8]!="CONFIRMED"{return f08Binding};return nil
}

func (v *f08ProofValidator) journal(s *f08Sources,action string,old,row []any)error{
	if action=="commit.insert"{
		if e:=f08ValidateTuple("commits",row);e!=nil{return e};f,e:=f08FindGroup(s,"final",row[:4]);if e!=nil{return e}
		if row[4]!=f.rootHash || row[5]!=f.root["original_response_digest"] || row[6]!=f.root["final_response_digest"]{return f08Binding}
		j,e:=s.one("idempotency",[]int{0,1,2,3},[]any{row[0],row[1],"v2:session.control",row[2]});if e!=nil{return e}
		return v.completedJournal(s,j,f)
	}
	if action!="journal.insert" && action!="journal.update"{return f08Mode}
	env,response,e:=v.journalEnvelope(s,row);if e!=nil{return e}
	if action=="journal.update"{
		if e=f08ValidateTuple("idempotency",old);e!=nil{return e};actual,e:=s.one("idempotency",[]int{0,1,2,3},old[:4]);if e!=nil || !f08Equal(actual,old){return f08Binding}
		if !f08RetainedColumns(old,row,6) || old[6]==row[6]{return f08Binding}
	}else if len(s.selectRows("idempotency",[]int{0,1,2,3},row[:4]))!=0{return f08Binding}
	payload,ok:=env["payload"].(map[string]any);if !ok{return f08Shape};locator:=f08JournalLocator(row)
	if payload["action"]!="CANCEL" || response["status"]=="REJECTED"{
		// F08 does not confer PAUSE/RESUME authority or alter typed rejection
		// business semantics. No witness/control may survive this branch.
		if action!="journal.insert" || len(s.selectRows("admitted",[]int{0,1,2,3},locator))!=0 || len(s.selectRows("final",[]int{0,1,2,3},locator))!=0 || len(s.selectRows("controls",[]int{0,1,2,5},locator))!=0{return f08Binding};return nil
	}
	a,e:=f08FindGroup(s,"admitted",locator);if e!=nil{return e};if _,e=v.admitted(s,a.row,false);e!=nil{return e};if e=v.completeMaterializations(s,a);e!=nil{return e}
	if row[7]!=a.row[5]{return f08Binding};req,e:=v.groupRequest(s,a);if e!=nil{return e}
	c,e:=s.one("controls",[]int{0,1,2,5},locator);if e!=nil{return e}
	if response["status"]=="ACKNOWLEDGED"{
		if action!="journal.insert" || c[8]=="CONFIRMED" || c[9]!=row[6] || c[10]!=nil || len(s.selectRows("final",[]int{0,1,2,3},locator))!=0{return f08Binding}
		raw,e:=f08Hex(row[6]);if e!=nil{return e};_,e=v.responseBody(req,raw,"ACKNOWLEDGED");return e
	}
	if response["status"]!="COMPLETED"{return f08Binding}
	f,e:=f08FindGroup(s,"final",locator);if e!=nil{return e}
	if action=="journal.update"{if f.root["completion_mode"]!="ACK_TO_COMPLETED" || old[6]!=c[9]{return f08Binding}}else if f.root["completion_mode"]!="ORIGINAL_COMPLETED"{return f08Binding}
	return v.completedJournal(s,row,f)
}
func (v *f08ProofValidator) completedJournal(s *f08Sources,row []any,f *f08Group)error{
	_,response,e:=v.journalEnvelope(s,row);if e!=nil || response["status"]!="COMPLETED"{return f08Binding}
	if _,e=v.finalGroup(s,f.row,true);e!=nil{return e};if e=v.completeMaterializations(s,f);e!=nil{return e}
	c,e:=s.one("controls",[]int{0,1,2,5},f.row[:4]);if e!=nil || c[8]!="CONFIRMED" || c[10]!=row[6] || c[9]==nil{return f08Binding}
	a,e:=f08FindGroup(s,"admitted",f.row[:4]);if e!=nil || row[7]!=a.row[5]{return f08Binding}
	raw,e:=f08Hex(row[6]);if e!=nil{return e};var prior []byte
	if f.root["completion_mode"]=="ACK_TO_COMPLETED"{prior,e=f08Hex(c[9]);if e!=nil{return e}}else if c[9]!=row[6]{return f08Binding}
	return v.finalResponse(s,f,raw,prior)
}

			b,raw,e:=f08RawObject(row[3]);if e!=nil{return out,e};if b["kind"]=="SESSION_REBUILT"{continue}
			to,ok:=b["to_runtime_ref"].(map[string]any);if !ok || to["id"]!=run.body["runtime_id"] || b["checkpoint_artifact_id"]!=run.body["checkpoint_artifact_id"] || b["actor"]!=rebuild["actor"]{continue}
			if _,e=f08Object(b,"from_runtime_ref","to_runtime_ref","grant","run_id","task_id","acceptance_contract_hash","context_bundle_id","checkpoint_artifact_id","at","actor");e!=nil{return out,e}
			fromRuntime,e:=v.version(s,run.project,b["from_runtime_ref"]);if e!=nil || fromRuntime.kind!="RuntimeRegistration" || fromRuntime.id!=previous.body["runtime_id"]{return out,f08Binding}
			toRuntime,e:=v.version(s,run.project,to);if e!=nil || toRuntime.kind!="RuntimeRegistration" || !f08Equal(to,admission["runtime_ref"]){return out,f08Binding}
			grant,ok:=b["grant"].(map[string]any);if !ok || !f08Equal(grant["admission"],admission) || b["run_id"]!=run.id || b["task_id"]!=task.id || b["context_bundle_id"]!=task.body["context_bundle_id"] || b["acceptance_contract_hash"]!=task.body["acceptance_contract"].(map[string]any)["contract_hash"] || row[0]==nil{return out,f08Binding}
			route=b;anchors=append(anchors,f08RowAnchor("routes",[]any{row[0]},raw));count++
		};if count!=1{return out,f08Binding}
	default:return out,f08Binding
	}
	carryRow,e:=s.one("checkpoint_carries",[]int{0,1,2},d[:3]);if e!=nil{return out,e};carry,carryRaw,e:=f08RawObject(carryRow[3]);if e!=nil{return out,e}
	if _,e=f08Object(carry,"actor","artifact_ref","from_use_receipt_hash","from_epoch","to_epoch","new_upload_key","reason","at");e!=nil{return out,e}
	if carry["actor"]!=rebuild["actor"] || carry["reason"]!="CHECKPOINT_CARRY_NOT_NEW_RESULT" || !f08Equal(carry["to_epoch"],d[2]){return out,f08Binding}
	fromEpoch,e:=f08Integer(carry["from_epoch"],1);if e!=nil || fromEpoch>previousEpoch{return out,f08Binding}
	artifact,e:=v.version(s,run.project,carry["artifact_ref"]);if e!=nil || artifact.kind!="Artifact" || artifact.id!=run.body["checkpoint_artifact_id"]{return out,f08Binding}
	oldUse,e:=s.one("artifact_uses",[]int{0,1,2,3,6},[]any{run.project,artifact.id,run.id,carry["from_epoch"],carry["from_use_receipt_hash"]});if e!=nil{return out,e}
	newUse,e:=s.one("artifact_uses",[]int{0,1,2,3,5},[]any{run.project,artifact.id,run.id,d[2],carry["new_upload_key"]});if e!=nil{return out,e}
	for _,use:=range [][]any{oldUse,newUse}{body,_,e:=f08RawObject(use[7]);if e!=nil || Receipt(body)!=use[6]{return out,f08Binding};content,e:=s.one("content",[]int{0,1},[]any{use[5],run.project});if e!=nil{return out,e};raw,e:=f08Hex(content[3]);if e!=nil || f08RawHash(raw)!=artifact.body["content_hash"] || content[4]!=artifact.body["content_hash"] || !f08Equal(content[6],f08Number(1)){return out,f08Binding}}
	anchors=append(anchors,f08RowAnchor("checkpoint_carries",d[:3],carryRaw))
	return f08LaterParts{run,previous,task,rebuild,route,carry,anchors},nil
}


// One stateless dispatch boundary for the actual frozen SQL action literals.
// Successful subset evaluation is not permission to install SQL24: remaining
// legitimate native/legacy/history branches intentionally fail closed.
func(v *f08ProofValidator) evaluate(mode int64,raw []byte)(driver.Value,error){
	if mode==3{return nil,f08Unwired}
	if e:=f08ProofFrame(mode,raw);e!=nil{return nil,e};decoded,_,e:=f08Strict(raw);if e!=nil{return nil,e}
	frame:=decoded.(map[string]any);s,e:=f08DecodeSources(frame["sources"]);if e!=nil{return nil,e};action:=frame["action"].(string);row:=frame["row"].([]any)
	var old []any;if frame["old"]!=nil{old=frame["old"].([]any)}
	switch mode{
	case 0:_,e=v.admitted(s,row,true)
	case 1:_,e=v.finalGroup(s,row,true)
	case 2:e=v.originInsert(s,row)
	case 4:e=v.dispatchUpdate(s,old,row)
	case 5:e=v.journal(s,action,old,row)
	case 6:e=v.control(s,action,old,row)
	case 7:e=v.materialization(s,action,row)
	case 8:
		if e=f08ValidateTuple("dispatches",row);e!=nil{return nil,e}
		run,re:=v.head(s,row[0].(string),"Run",row[1].(string));if re!=nil{return nil,re}
		if run.revision>1{if _,e=v.laterParts(s,row);e!=nil{return nil,e};return nil,f08NativeProofGap}
		origin,e:=v.deriveInitialOrigin(s,row);if e!=nil{return nil,e};b,e:=f08Encode(origin);if e!=nil{return nil,e};_,canonical,e:=f08Strict(b);if e!=nil{return nil,e};return string(canonical),nil
	default:return nil,f08Mode
	}
	if e!=nil{return nil,e};return int64(1),nil
}
