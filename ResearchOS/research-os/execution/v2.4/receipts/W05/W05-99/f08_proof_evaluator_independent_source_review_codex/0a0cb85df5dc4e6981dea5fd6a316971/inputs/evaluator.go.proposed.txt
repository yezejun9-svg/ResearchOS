// One stateless dispatch boundary for the actual frozen SQL action literals.
// Successful subset evaluation is not permission to install SQL24: remaining
// legitimate native/legacy/history branches intentionally fail closed.
func(v *f08ProofValidator) evaluate(mode int64,raw []byte)(driver.Value,error){
	if mode==3{return nil,f08Unwired}
	if e:=f08ProofFrame(mode,raw);e!=nil{return nil,e};decoded,_,e:=f08Strict(raw);if e!=nil{return nil,e}
	frame:=decoded.(map[string]any);s,e:=f08DecodeSources(frame["sources"]);if e!=nil{return nil,e};action:=frame["action"].(string);row:=frame["row"].([]any)
	var old []any;if frame["old"]!=nil{old=frame["old"].([]any)}
	switch mode{
	case 0:_,e=v.admitted(s,row,true)
	case 1:_,e=v.finalGroup(s,row,true)
	case 2:e=v.originInsert(s,row)
	case 4:e=v.dispatchUpdate(s,old,row)
	case 5:e=v.journal(s,action,old,row)
	case 6:e=v.control(s,action,old,row)
	case 7:e=v.materialization(s,action,row)
	case 8:
		if e=f08ValidateTuple("dispatches",row);e!=nil{return nil,e}
		run,re:=v.head(s,row[0].(string),"Run",row[1].(string));if re!=nil{return nil,re}
		if run.revision>1{if _,e=v.laterParts(s,row);e!=nil{return nil,e};return nil,f08NativeProofGap}
		origin,e:=v.deriveInitialOrigin(s,row);if e!=nil{return nil,e};b,e:=f08Encode(origin);if e!=nil{return nil,e};_,canonical,e:=f08Strict(b);if e!=nil{return nil,e};return string(canonical),nil
	default:return nil,f08Mode
	}
	if e!=nil{return nil,e};return int64(1),nil
}
