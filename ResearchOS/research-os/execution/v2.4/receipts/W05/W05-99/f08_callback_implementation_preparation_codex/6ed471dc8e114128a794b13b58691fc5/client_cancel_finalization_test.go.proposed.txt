type f08TestFixture struct {
	functions [3]f08Scalar
	header []byte
	identity []byte
	identityObject map[string]any
}

func f08TestJSON(t *testing.T, v any) []byte {
	t.Helper()
	b, err := json.Marshal(v)
	if err != nil { t.Fatal(err) }
	return b
}

func f08TestHeader(t *testing.T, raw []byte) []byte {
	t.Helper()
	return f08TestJSON(t, struct {
		Schema string `json:"schema"`
		Encoding string `json:"encoding"`
		Raw string `json:"raw_envelope_base64"`
	}{f08HeaderSchema, "base64", base64.StdEncoding.EncodeToString(raw)})
}

func f08TestSetup(t *testing.T) f08TestFixture {
	t.Helper()
	functions, err := f08BuildCallbacks(backupConfigurationProtocol, backupConfigurationDomain)
	if err != nil { t.Fatal(err) }
	raw := []byte(`{"protocol_version":"researchos-client/2","command_id":"cancel-one","client_id":"client-one","project_id":"project-one","session_id":"session-one","expected_revision":7,"writer_epoch":3,"method":"session.control","payload":{"action":"CANCEL","target_id":"goal-one","reason":"fixture"}}`)
	envelope, err := DecodeEnvelope(raw)
	if err != nil { t.Fatal(err) }
	hash, err := envelope.ContentHash()
	if err != nil { t.Fatal(err) }
	identity := map[string]any{"format":f08IdentityFormat, "principal":"owner-one", "project":"project-one", "operation":"v2:session.control", "key":"cancel-one", "request_hash":hash, "command_id":"cancel-one", "method":"session.control", "client_id":"client-one", "session_id":"session-one", "writer_epoch":3, "expected_revision":7, "target_id":"goal-one"}
	return f08TestFixture{functions, f08TestHeader(t, raw), f08TestJSON(t, identity), identity}
}

func f08TestRequireOne(t *testing.T, fn f08Scalar, args ...driver.Value) {
	t.Helper()
	v, err := fn(nil, args)
	if err != nil || v != int64(1) { t.Fatalf("expected int64(1), value type=%T error=%v", v, err) }
}

func f08TestRequireError(t *testing.T, fn f08Scalar, expected f08Error, args ...driver.Value) {
	t.Helper()
	v, err := fn(nil, args)
	if err == nil || v != nil { t.Fatalf("expected fail-closed nil/error, type=%T err=%v", v, err) }
	if expected != "" && err != expected { t.Fatalf("wrong fixed error: %v", err) }
}

func f08TestBinding(t *testing.T, f f08TestFixture, old, current []byte, snapshot map[string]any) []byte {
	t.Helper()
	p := map[string]any{"admitted_root_hash":"sha256:"+strings.Repeat("a",64), "original_response_digest":f08RawHash(current), "final_root_hash":nil, "final_response_digest":nil, "final_snapshot_binding":nil}
	if snapshot != nil {
		pendingHash, err := f08ValueHash(snapshot["pending_command_ids"])
		if err != nil { t.Fatal(err) }
		p["final_root_hash"] = "sha256:"+strings.Repeat("b",64)
		p["final_response_digest"] = f08RawHash(current)
		p["final_snapshot_binding"] = map[string]any{"session_id":"session-one", "project_id":"project-one", "revision":snapshot["revision"], "event_sequence":snapshot["event_sequence"], "pending_command_ids_digest":pendingHash}
	}
	if old != nil { p["original_response_digest"] = f08RawHash(old) }
	return f08TestJSON(t, map[string]any{"identity":f.identityObject, "response_proof":p})
}

func TestF08ProposedRequestOriginalTuple(t *testing.T) {
	f := f08TestSetup(t)
	f08TestRequireOne(t, f.functions[0], f.header, f.identity)
	for _, key := range []string{"principal", "project", "key", "request_hash", "command_id", "method", "client_id", "session_id", "target_id"} {
		t.Run(key, func(t *testing.T) {
			m := map[string]any{}
			for k,v := range f.identityObject { m[k]=v }
			m[key] = "changed"
			if key == "principal" {
				// Principal is SQL producer authority, not present in Envelope.
				// Local callback must not invent authentication from this field.
				f08TestRequireOne(t, f.functions[0], f.header, f08TestJSON(t,m))
				return
			}
			f08TestRequireError(t, f.functions[0], "", f.header, f08TestJSON(t,m))
		})
	}
}

func TestF08ProposedResponseFullSchemaAndRawDigest(t *testing.T) {
	f:=f08TestSetup(t)
	ack:=f08TestACK(t)
	complete,snapshot:=f08TestCompleted(t)
	binding:=f08TestBinding(t,f,ack,complete,snapshot)
	for name,raw:=range map[string][]byte{
		"wrong-session":bytes.Replace(complete,[]byte(`"session-one"`),[]byte(`"session-two"`),1),
		"unsettled":bytes.Replace(complete,[]byte(`"settled":true`),[]byte(`"settled":false`),1),
		"null-revision":bytes.Replace(complete,[]byte(`"revision":8`),[]byte(`"revision":null`),1),
		"duplicate-escaped":bytes.Replace(complete,[]byte(`{`),[]byte(`{"\u0073tatus":"COMPLETED",`),1),
		"raw-whitespace-changed":append([]byte(" "),complete...),
	} {
		t.Run(name,func(t *testing.T){
			candidateBinding:=binding
			if name!="raw-whitespace-changed" {
				// Recompute transport digest so these controls reach full raw
				// schema/identity checking rather than failing only a digest.
				candidateBinding=f08TestBinding(t,f,ack,raw,snapshot)
			}
			f08TestRequireError(t,f.functions[1],"",f.header,candidateBinding,ack,raw,int64(2))
		})
	}
	t.Run("invalid-ack-operation",func(t *testing.T){
		bad:=bytes.Replace(ack,[]byte(`"op-cancel-one"`),[]byte(`"op-other"`),1)
		f08TestRequireError(t,f.functions[1],f08Binding,f.header,f08TestBinding(t,f,nil,bad,nil),nil,bad,int64(0))
	})
}

func TestF08ProposedDigestAndTypeViolationModes(t *testing.T) {
	f:=f08TestSetup(t)
	raw:=[]byte{0,0xff,0x61}
	v,err:=f.functions[2](nil,[]driver.Value{int64(-2),raw})
	if err!=nil || v!=f08RawHash(raw) { t.Fatal("raw digest altered bytes") }
	a,err:=f.functions[2](nil,[]driver.Value{int64(-1),[]byte(`{"b":2,"a":1}`)})
	if err!=nil {t.Fatal(err)}
	b,err:=f.functions[2](nil,[]driver.Value{int64(-1),[]byte(` { "a":1,"b":2 } `)})
	if err!=nil || a!=b {t.Fatal("canonical digest mismatch")}
	f08TestRequireError(t,f.functions[2],f08StorageClass,int64(9),[]byte("F08_STORAGE_CLASS"))
	f08TestRequireError(t,f.functions[2],f08Mode,int64(10),[]byte(`{}`))
}

func TestF08ProposedIncompleteProofNeverAuthorizes(t *testing.T) {
	f:=f08TestSetup(t)
	for mode, action:=range map[int64]string{0:"admitted.insert",1:"final.insert",2:"origin.insert",4:"dispatch.update",5:"journal.insert",6:"control.insert",7:"pin.insert",8:"origin.derive"} {
		t.Run(fmt.Sprint(mode),func(t *testing.T){
			var old any
			if mode==4 {old=[]any{"old"}}
			input:=f08TestJSON(t,map[string]any{"action":action,"row":[]any{"not-proof"},"old":old,"sources":[]any{}})
			f08TestRequireError(t,f.functions[2],f08Unimplemented,mode,input)
		})
	}
	// Mode3 has no concrete action literal in the current SQL; no guessed alias.
	f08TestRequireError(t,f.functions[2],f08Mode,int64(3),[]byte(`{"action":"legacy.insert","row":[1],"old":null,"sources":[]}`))
}
