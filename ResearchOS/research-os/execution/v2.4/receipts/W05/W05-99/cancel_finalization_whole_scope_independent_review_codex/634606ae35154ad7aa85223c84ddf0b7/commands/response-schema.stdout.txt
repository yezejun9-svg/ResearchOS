116: // v2EnvelopeFor validates locally, exactly as DecodeEnvelope does on the far side, so a malformed command never
117: // leaves the process: the refusal names the missing field instead of arriving as an unexplained 400.
118: func v2EnvelopeFor(ctx *v2Context, method string, payload map[string]any) (*v2Envelope, string) {
119: 	c, known := v2Contract[method]
120: 	if !known {
121: 		return nil, "UNKNOWN_METHOD"
122: 	}
123: 	if ctx.clientID == "" {
124: 		return nil, "MISSING_FIELD:client_id"
125: 	}
126: 	if ctx.credential == "" {
127: 		return nil, "CREDENTIAL_ABSENT"
128: 	}
129: 	if !commandIDPattern.MatchString(ctx.commandID) {
130: 		return nil, "MISSING_FIELD:command_id"
131: 	}
132: 	raw, err := json.Marshal(payload)
133: 	if err != nil {
134: 		return nil, "PAYLOAD_NOT_OBJECT"
135: 	}
136: 	var keys map[string]json.RawMessage
137: 	if err := json.Unmarshal(raw, &keys); err != nil {
138: 		return nil, "PAYLOAD_NOT_OBJECT"
139: 	}
140: 	for _, k := range c.required {
141: 		if _, has := keys[k]; !has {
142: 			return nil, "MISSING_PAYLOAD_FIELD:" + k
143: 		}
144: 	}
145: 	if c.payload == "EmptyPayload" && len(keys) > 0 {
146: 		return nil, "UNEXPECTED_PAYLOAD_KEYS:" + strings.Join(sortedKeys(keys), ",")
147: 	}
148: 	env := &v2Envelope{ProtocolVersion: clientProtocol, CommandID: ctx.commandID, ClientID: ctx.clientID,
149: 		Method: method, Payload: raw}
