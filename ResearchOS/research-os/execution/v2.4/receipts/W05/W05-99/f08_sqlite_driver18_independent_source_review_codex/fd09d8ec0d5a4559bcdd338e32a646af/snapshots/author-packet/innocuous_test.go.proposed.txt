var driver18RegistrationError error

func driver18Positive(_ *FunctionContext, args []driver.Value) (driver.Value, error) {
	if len(args) != 1 { return nil, fmt.Errorf("driver18 arity") }
	x, ok := args[0].(int64)
	if !ok { return nil, fmt.Errorf("driver18 requires integer") }
	if x > 0 { return int64(1), nil }
	return int64(0), nil
}

type driver18Sum struct { sum int64; reject bool; null bool }
func (s *driver18Sum) Step(_ *FunctionContext, args []driver.Value) error {
	if s.reject { return errors.New("driver18 aggregate callback rejected") }
	x, ok := args[0].(int64); if !ok { return errors.New("driver18 sum requires integer") }
	s.sum += x; return nil
}
func (s *driver18Sum) WindowInverse(_ *FunctionContext, args []driver.Value) error {
	x, ok := args[0].(int64); if !ok { return errors.New("driver18 inverse requires integer") }
	s.sum -= x; return nil
}
func (s *driver18Sum) WindowValue(_ *FunctionContext) (driver.Value, error) {
	if s.null { return nil, nil }; return s.sum, nil
}
func (s *driver18Sum) Final(_ *FunctionContext) {}

func driver18Register(name string, impl *FunctionImpl) {
	if err := RegisterFunction(name, impl); err != nil && driver18RegistrationError == nil {
		driver18RegistrationError = fmt.Errorf("register %s: %w", name, err)
	}
}
func driver18Aggregate(deterministic, innocuous bool) *FunctionImpl {
	return &FunctionImpl{NArgs: 1, Deterministic: deterministic, Innocuous: innocuous,
		MakeAggregate: func(FunctionContext) (AggregateFunction, error) { return &driver18Sum{}, nil }}
}

type driver18DB struct { db *sql.DB; conn *sql.Conn; ctx context.Context; cancel context.CancelFunc }
func driver18Open(t *testing.T, dsn string) *driver18DB {
	t.Helper()
	if driver18RegistrationError != nil { t.Fatal(driver18RegistrationError) }
	ctx, cancel := context.WithTimeout(context.Background(), 20*time.Second)
	db, err := sql.Open("sqlite", dsn); if err != nil { cancel(); t.Fatal(err) }
	db.SetMaxOpenConns(2); db.SetMaxIdleConns(2)
	c, err := db.Conn(ctx); if err != nil { db.Close(); cancel(); t.Fatal(err) }
	f := &driver18DB{db: db, conn: c, ctx: ctx, cancel: cancel}
	t.Cleanup(func() {
		if err := c.Close(); err != nil { t.Errorf("connection close: %v", err) }
		if err := db.Close(); err != nil { t.Errorf("database close: %v", err) }
		cancel()
	})
	driver18Off(t, ctx, c)
	return f
}
func driver18Off(t *testing.T, ctx context.Context, c *sql.Conn) {
	t.Helper()
	if _, err := c.ExecContext(ctx, "PRAGMA trusted_schema=OFF"); err != nil { t.Fatal(err) }
	var got int
	if err := c.QueryRowContext(ctx, "PRAGMA trusted_schema").Scan(&got); err != nil || got != 0 { t.Fatalf("trusted_schema got=%d err=%v", got, err) }
}
func (f *driver18DB) exec(t *testing.T, q string, args ...any) {
	t.Helper(); if _, err := f.conn.ExecContext(f.ctx, q, args...); err != nil { t.Fatalf("setup/positive SQL %q: %v", q, err) }
}
func (f *driver18DB) integer(t *testing.T, q string, want int64, args ...any) {
	t.Helper(); var got int64
	if err := f.conn.QueryRowContext(f.ctx, q, args...).Scan(&got); err != nil || got != want { t.Fatalf("SQL %q got=%d want=%d err=%v", q, got, want, err) }
}
func driver18Error(t *testing.T, err error, code int, text string) {
	t.Helper(); var native *Error
	if err == nil || !errors.As(err, &native) || native.Code() != code || !strings.Contains(err.Error(), text) {
		t.Fatalf("expected SQLite code=%d containing %q; got %T %v", code, text, err, err)
	}
}
func driver18Oracle(t *testing.T, ctx context.Context, c *sql.Conn, name, typ string, arity, flags int) {
	t.Helper()
	rows, err := c.QueryContext(ctx, "SELECT name,builtin,type,enc,narg,flags FROM pragma_function_list WHERE name=?", name)
	if err != nil { t.Fatal(err) }; defer func(){ if err := rows.Close(); err != nil { t.Error(err) } }()
	count := 0
	for rows.Next() {
		var n, k, enc string; var builtin, na, fl int
		if err := rows.Scan(&n, &builtin, &k, &enc, &na, &fl); err != nil { t.Fatal(err) }
		count++
		if n != name || builtin != 0 || k != typ || enc != "utf8" || na != arity || fl != flags { t.Fatalf("function_list %q builtin=%d type=%s enc=%s arity=%d flags=%d", n, builtin, k, enc, na, fl) }
	}
	if err := rows.Err(); err != nil { t.Fatal(err) }
	if count != 1 { t.Fatalf("function_list %s count=%d", name, count) }
}
func driver18UnsafeView(t *testing.T, name string, flags int) {
	f := driver18Open(t, ":memory:")
	driver18Oracle(t, f.ctx, f.conn, name, "s", 1, flags)
	f.integer(t, "SELECT "+name+"(7)", 1)
	f.exec(t, "CREATE VIEW v AS SELECT "+name+"(7) AS result")
	var got int
	err := f.conn.QueryRowContext(f.ctx, "SELECT result FROM v").Scan(&got)
	driver18Error(t, err, 1, "unsafe use of "+name+"()")
}

func driver18AggregateView(t *testing.T, name string, window, allowed bool) {
	f := driver18Open(t, ":memory:")
	flags := 2048; if allowed { flags += 2097152 }
	// Public MakeAggregate is installed with sqlite3_create_window_function;
	// function_list.type is w even when the invocation uses aggregate syntax.
	driver18Oracle(t, f.ctx, f.conn, name, "w", 1, flags)
	f.exec(t, "CREATE TABLE data(x INTEGER)"); f.exec(t, "INSERT INTO data VALUES(1),(2),(3)")
	f.integer(t, "SELECT "+name+"(x) FROM data", 6)
	query := "SELECT "+name+"(x) AS value FROM data"
	if window { query = "SELECT x,"+name+"(x) OVER(ORDER BY x ROWS BETWEEN 1 PRECEDING AND CURRENT ROW) AS value FROM data" }
	f.exec(t, "CREATE VIEW v AS "+query)
	if !allowed {
		rows, err := f.conn.QueryContext(f.ctx, "SELECT * FROM v")
		if rows != nil { rows.Close() }
		driver18Error(t, err, 1, "unsafe use of "+name+"()")
		return
	}
	if !window { f.integer(t, "SELECT value FROM v", 6); return }
	rows, err := f.conn.QueryContext(f.ctx, "SELECT value FROM v ORDER BY x")
	if err != nil { t.Fatal(err) }; defer func(){ if err := rows.Close(); err != nil { t.Error(err) } }()
	var got []int64
	for rows.Next() { var x int64; if err := rows.Scan(&x); err != nil { t.Fatal(err) }; got=append(got,x) }
	if err := rows.Err(); err != nil { t.Fatal(err) }
	if !reflect.DeepEqual(got, []int64{1,3,5}) { t.Fatalf("moving sum=%v", got) }
}
