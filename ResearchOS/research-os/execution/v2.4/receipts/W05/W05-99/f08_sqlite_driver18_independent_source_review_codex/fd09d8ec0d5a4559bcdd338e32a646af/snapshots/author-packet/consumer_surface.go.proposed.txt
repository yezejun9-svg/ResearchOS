package driver18consumer

import (
	"database/sql/driver"

	sqlite "modernc.org/sqlite"
)

func ScalarDefinition() *sqlite.FunctionImpl {
	return &sqlite.FunctionImpl{NArgs: 1, Deterministic: true, Innocuous: true,
		Scalar: func(_ *sqlite.FunctionContext, args []driver.Value) (driver.Value, error) { return args[0], nil }}
}
var RegisterPublic func(string, *sqlite.FunctionImpl) error = sqlite.RegisterFunction
var DefaultIsNotInnocuous = !(&sqlite.FunctionImpl{}).Innocuous
