// Full deterministic public logical image, captured through a read-only
// immutable connection before store.Open and again after its final close.
// Rows retain SQLite value types and rowid where it is stored authority.
func providerSemanticVector(path, output string) (Object, error) {
	u := url.URL{Scheme: "file", Opaque: filepath.ToSlash(path)}
	q := u.Query()
	q.Set("mode", "ro")
	q.Set("immutable", "1")
	u.RawQuery = q.Encode()
	db, err := sql.Open("sqlite", u.String())
	if err != nil {
		return nil, err
	}
	defer db.Close()
	read := func(query string) (Object, error) {
		rows, e := db.Query(query)
		if e != nil {
			return nil, e
		}
		defer rows.Close()
		columns, e := rows.Columns()
		if e != nil {
			return nil, e
		}
		encoded := []string{}
		for rows.Next() {
			values := make([]any, len(columns))
			args := make([]any, len(columns))
			for i := range values {
				args[i] = &values[i]
			}
			if e = rows.Scan(args...); e != nil {
				return nil, e
			}
			typed := []any{}
			for _, value := range values {
				switch v := value.(type) {
				case nil:
					typed = append(typed, []any{"null"})
				case int64:
					typed = append(typed, []any{"integer", fmt.Sprint(v)})
				case float64:
					typed = append(typed, []any{"real-ieee754", fmt.Sprintf("%016x", math.Float64bits(v))})
				case string:
					typed = append(typed, []any{"text", v})
				case []byte:
					typed = append(typed, []any{"blob-hex", fmt.Sprintf("%x", v)})
				default:
					return nil, fmt.Errorf("unrecorded SQLite value type %T", value)
				}
			}
			encoded = append(encoded, string(Bytes(typed)))
		}
		if e = rows.Err(); e != nil {
			return nil, e
		}
		sort.Strings(encoded)
		return Object{"columns": columns, "row_count": len(encoded), "typed_row_multiset": encoded}, nil
	}
	schema, err := read(`SELECT type,name,tbl_name,sql FROM sqlite_schema`)
	if err != nil {
		return nil, err
	}
	rows, err := db.Query(`SELECT name,coalesce(sql,'') FROM sqlite_schema WHERE type='table' ORDER BY name`)
	if err != nil {
		return nil, err
	}
	tableSQL := map[string]string{}
	for rows.Next() {
		var name, statement string
		if err = rows.Scan(&name, &statement); err != nil {
			rows.Close()
			return nil, err
		}
		tableSQL[name] = statement
	}
	err = rows.Err()
	rows.Close()
	if err != nil {
		return nil, err
	}
	tables := Object{}
	for name, statement := range tableSQL {
		quoted := `"` + strings.ReplaceAll(name, `"`, `""`) + `"`
		projection := "rowid,*"
		if strings.Contains(strings.ToUpper(statement), "WITHOUT ROWID") || strings.Contains(strings.ToUpper(statement), "CREATE VIRTUAL TABLE") {
			projection = "*"
		}
		data, e := read("SELECT " + projection + " FROM " + quoted)
		if e != nil {
			return nil, e
		}
		columns, e := read("PRAGMA table_xinfo(" + quoted + ")")
		if e != nil {
			return nil, e
		}
		tables[name] = Object{"rows": data, "column_metadata": columns}
	}
	vector := Object{"format": "researchos.sqlite-public-semantic.v1", "schema": schema, "tables": tables}
	f, err := os.OpenFile(output, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if err != nil {
		return nil, err
	}
	z := gzip.NewWriter(f)
	_, err = z.Write(Bytes(vector))
	ce := z.Close()
	fe := f.Close()
	if err != nil {
		return nil, err
	}
	if ce != nil {
		return nil, ce
	}
	if fe != nil {
		return nil, fe
	}
	return vector, nil
}

// These helpers only mutate newly copied, explicitly synthetic/corrupt fixture
// databases after the harness has saved every original trigger and logical row.
// They never generate model traffic or manufacture original accepted history.
func providerCopiedBodies(db *sql.DB, table string, where string) ([]Object, error) {
	rows, err := db.Query(`SELECT rowid,body FROM ` + table + ` WHERE ` + where + ` ORDER BY rowid`)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	items := []Object{}
	for rows.Next() {
		var row int64
		var raw string
		if err = rows.Scan(&row, &raw); err != nil {
			return nil, err
		}
		body, e := Decode([]byte(raw))
		if e != nil {
			return nil, e
		}
		items = append(items, Object{"rowid": row, "body": body})
	}
	return items, rows.Err()
}

func providerCopiedCost(ledger Object) Object {
	cost := Object{}
	for _, key := range []string{"policy_id", "run_id", "operation", "money", "provider_request_id", "provider_event_id", "reservation_id", "cost_basis", "occurred_at"} {
		cost[key] = ledger[key]
	}
	return cost
}

func providerRehashCopiedEffects(db *sql.DB) error {
	items, err := providerCopiedBodies(db, "resource_effects", "1=1")
	if err != nil {
		return err
	}
	for _, item := range items {
		e := item["body"].(map[string]any)
		var raw string
		if err = db.QueryRow(`SELECT body FROM domain_objects WHERE project=? AND kind='BudgetLedgerEntry' AND id=?`, e["project_id"], e["reservation_id"]).Scan(&raw); err != nil {
			return err
		}
		ledger, er := Decode([]byte(raw))
		if er != nil {
			return er
		}
		e["request_hash"] = Hash([]any{e["payload"], providerCopiedCost(ledger)})
		if _, err = db.Exec(`UPDATE resource_effects SET body=? WHERE rowid=?`, string(Bytes(e)), item["rowid"]); err != nil {
			return err
		}
	}
	return nil
}

func providerWriteCopiedLedger(db *sql.DB, ledger Object) error {
	ledger["receipt_hash"] = Receipt(ledger)
	for _, table := range []string{"domain_objects", "domain_versions"} {
		r, err := db.Exec(`UPDATE `+table+` SET body=?,content_hash=? WHERE project=? AND kind='BudgetLedgerEntry' AND id=? AND revision=?`, string(Bytes(ledger)), ledger["receipt_hash"], ledger["project_id"], ledger["id"], ledger["revision"])
		if err != nil {
			return err
		}
		n, err := r.RowsAffected()
		if err != nil || n != 1 {
			return fmt.Errorf("synthetic ledger exact row: %d %v", n, err)
		}
	}
	return nil
}

func TestProviderBindingOriginalEpochCausalControl(t *testing.T) {
	mode := os.Getenv("RESEARCHOS_PROVIDER_EPOCH_MODE")
	dir := os.Getenv("RESEARCHOS_PROVIDER_EPOCH_DIR")
	if dir == "" || os.Mkdir(dir, 0700) != nil {
		t.Fatal("new explicit epoch control directory required")
	}
	db, err := store.Open(filepath.Join(dir, "epoch.sqlite"))
	if err != nil {
		t.Fatal(err)
	}
	defer db.Close()
	// Exact existing domain schema, no product admission or history creation.
	schema, err := os.ReadFile(filepath.Join("..", "..", "migrations", "sqlite", "0002_domain.sql"))
	if err != nil {
		t.Fatal(err)
	}
	if _, err = db.Exec(string(schema)); err != nil {
		t.Fatal(err)
	}
	for rev := 1; rev <= 3; rev++ {
		runtime, epoch := "original-runtime", 1
		if rev == 3 {
			runtime, epoch = "later-runtime", 2
		}
		run := Object{"project_id": "synthetic", "id": "run", "revision": rev, "task_id": "task", "runtime_id": runtime, "lease_epoch": epoch, "created_at": "synthetic", "executor_principal_id": "synthetic", "status": fmt.Sprint(rev)}
		if _, err = db.Exec(`INSERT INTO domain_versions(project,kind,id,revision,body) VALUES('synthetic','Run','run',?,?)`, rev, string(Bytes(run))); err != nil {
			t.Fatal(err)
		}
	}
	effect, err := Decode([]byte(`{"epoch":1,"task_id":"task"}`))
	if err != nil {
		t.Fatal(err)
	}
	want := ""
	switch mode {
	case "decoded-number":
		if _, ok := effect["epoch"].(json.Number); !ok {
			t.Fatal("control must exercise real Decode json.Number")
		}
	case "int64":
		effect["epoch"] = int64(1)
	case "missing":
		delete(effect, "epoch")
		want = "provider original Run epoch invalid"
	case "string", "fraction", "zero", "negative", "overflow":
		effect["epoch"] = map[string]any{"string": "1", "fraction": json.Number("1.5"), "zero": json.Number("0"), "negative": json.Number("-1"), "overflow": json.Number("9223372036854775808")}[mode]
		want = "provider original Run epoch invalid"

		effect["epoch"] = int64(3)
		want = "provider original Run epoch unavailable"
	case "ambiguous":
		run := Object{"project_id": "synthetic", "id": "run", "revision": 4, "task_id": "task", "runtime_id": "ambiguous-runtime", "lease_epoch": 1, "created_at": "synthetic", "executor_principal_id": "synthetic"}

			t.Fatal(err)
		}
		want = "provider ambiguous original Run epoch"
	default:
		t.Fatal("unregistered epoch control")
	}
	tx, err := db.Begin()
	if err != nil {
		t.Fatal(err)
	}
	run, e := providerOriginalRun(tx, "synthetic", "run", effect)
	if err = tx.Rollback(); err != nil {
		t.Fatal(err)
	}
	result := Object{"label": "SYNTHETIC_NOT_PRODUCT", "mode": mode, "epoch_type": fmt.Sprintf("%T", effect["epoch"]), "epoch_repr": fmt.Sprint(effect["epoch"]), "run": run, "required_cause": want, "error": nil, "models": 0, "workers": 0, "http": 0}
	if e != nil {
		result["error"] = e.Error()
	}
	if err = os.WriteFile(filepath.Join(dir, "actual.json"), Bytes(result), 0600); err != nil {
		t.Fatal(err)
	}
	if want != "" {
		if e == nil || !strings.HasSuffix(e.Error(), want) {
			t.Fatal("incorrect epoch refusal layer", e, want)
		}
	} else if e != nil || run["runtime_id"] != "original-runtime" || Hash(run["lease_epoch"]) != Hash(1) {
		t.Fatal("original epoch not selected", e, run)
	}
}
