67: func nativeSecurityDescriptor(h windows.Handle, includeSACL bool) ([]byte, string, string, error) {
68: 	if err := ntQuerySecurityObject.Find(); err != nil {
69: 		return nil, "", "", err
70: 	}
71: 	raw := make([]byte, 64<<10)
72: 	var needed uint32
73: 	flags := windows.OWNER_SECURITY_INFORMATION | windows.GROUP_SECURITY_INFORMATION | windows.DACL_SECURITY_INFORMATION
74: 	if includeSACL {
75: 		flags |= windows.SACL_SECURITY_INFORMATION
76: 	}
77: 	status, _, _ := ntQuerySecurityObject.Call(uintptr(h), uintptr(flags), uintptr(unsafe.Pointer(&raw[0])), uintptr(len(raw)), uintptr(unsafe.Pointer(&needed)))
78: 	if result := windows.NTStatus(status); result != 0 || needed < 20 || needed > uint32(len(raw)) {
79: 		if result != 0 {
80: 			return nil, "", "", &nativeSecurityQueryError{status: result, needed: needed}
81: 		}
82: 		return nil, "", "", errors.New("NtQuerySecurityObject length invalid")
83: 	}
84: 	raw = raw[:needed]
85: 	semantic, err := nativeSecuritySemanticSHA256(raw)
86: 	if err != nil {
87: 		return nil, "", "", err
88: 	}
89: 	sd := (*windows.SECURITY_DESCRIPTOR)(unsafe.Pointer(&raw[0]))
90: 	if !sd.IsValid() {
91: 		return nil, "", "", errors.New("native security descriptor invalid")
92: 	}
93: 	sddl := sd.String()
94: 	runtime.KeepAlive(raw)
95: 	return raw, sddl, semantic, nil
96: }

98: func nativeSecurityFromSDDL(sddl string) ([]byte, string, error) {
99: 	sd, err := windows.SecurityDescriptorFromString(sddl)
100: 	if err != nil {
101: 		return nil, "", err
102: 	}
103: 	raw := append([]byte(nil), unsafe.Slice((*byte)(unsafe.Pointer(sd)), sd.Length())...)
104: 	semantic, err := nativeSecuritySemanticSHA256(raw)
105: 	runtime.KeepAlive(sd)
106: 	return raw, semantic, err
107: }

120: func finalPath(h windows.Handle) (string, error) {
121: 	buf := make([]uint16, 512)
122: 	for len(buf) <= 32768 {
123: 		n, e := windows.GetFinalPathNameByHandle(h, &buf[0], uint32(len(buf)), 0)
124: 		if e != nil {
125: 			return "", e
126: 		}
127: 		if n == 0 {
128: 			return "", errors.New("empty final path")
129: 		}
130: 		if int(n) < len(buf) {
131: 			return windows.UTF16ToString(buf[:n]), nil
132: 		}
133: 		buf = make([]uint16, int(n)+1)
134: 	}
135: 	return "", errors.New("final path exceeds Windows bound")
136: }

142: func openNativeAccess(path string, directory bool, hashFile bool, extraAccess uint32) (nativeHold, error) {
143: 	if !filepath.IsAbs(path) || filepath.Clean(path) != path {
144: 		return nativeHold{}, errors.New("native path is not canonical absolute")
145: 	}
146: 	if strings.Contains(path[len(filepath.VolumeName(path)):], ":") {
147: 		return nativeHold{}, errors.New("alternate data stream forbidden")
148: 	}
149: 	w, e := windows.UTF16PtrFromString(path)
150: 	if e != nil {
151: 		return nativeHold{}, e
152: 	}
153: 	// No FILE_SHARE_DELETE: the live owner pins every opened component against
154: 	// rename/delete until the caller's actual outer transaction has returned.
155: 	access := uint32(windows.READ_CONTROL | windows.FILE_READ_ATTRIBUTES | windows.FILE_READ_DATA | extraAccess)
156: 	h, e := windows.CreateFile(w, access|windows.ACCESS_SYSTEM_SECURITY,
157: 		windows.FILE_SHARE_READ|windows.FILE_SHARE_WRITE, nil, windows.OPEN_EXISTING,
158: 		windows.FILE_FLAG_OPEN_REPARSE_POINT|windows.FILE_FLAG_BACKUP_SEMANTICS, 0)
159: 	saclStatus := "READABLE"
160: 	var saclUnavailableStage string
161: 	var saclUnavailableWin32 uint32
162: 	var saclUnavailableNTSTATUS uint32
163: 	if errors.Is(e, windows.ERROR_ACCESS_DENIED) || errors.Is(e, windows.ERROR_PRIVILEGE_NOT_HELD) {
164: 		var errno syscall.Errno
165: 		if !errors.As(e, &errno) {
166: 			return nativeHold{}, e
167: 		}
168: 		saclUnavailableWin32 = uint32(errno)
169: 		saclUnavailableStage = "OPEN"
170: 		saclStatus = "UNAVAILABLE"
171: 		h, e = windows.CreateFile(w, access,
172: 			windows.FILE_SHARE_READ|windows.FILE_SHARE_WRITE, nil, windows.OPEN_EXISTING,
173: 			windows.FILE_FLAG_OPEN_REPARSE_POINT|windows.FILE_FLAG_BACKUP_SEMANTICS, 0)
174: 	}
175: 	if e != nil {
176: 		return nativeHold{}, e
177: 	}
178: 	fail := func(e error) (nativeHold, error) { windows.CloseHandle(h); return nativeHold{}, e }
179: 	rawSD, sddl, semantic, e := nativeSecurityDescriptor(h, saclStatus == "READABLE")
180: 	if e != nil && saclStatus == "READABLE" && (errors.Is(e, windows.ERROR_ACCESS_DENIED) || errors.Is(e, windows.ERROR_PRIVILEGE_NOT_HELD)) {
181: 		var queryErr *nativeSecurityQueryError
182: 		var errno syscall.Errno
183: 		if !errors.As(e, &queryErr) || !errors.As(e, &errno) {
184: 			return fail(e)
185: 		}
186: 		saclStatus = "UNAVAILABLE"
187: 		saclUnavailableStage = "QUERY"
188: 		saclUnavailableWin32 = uint32(errno)
189: 		saclUnavailableNTSTATUS = uint32(queryErr.status)
190: 		if closeErr := windows.CloseHandle(h); closeErr != nil {
191: 			return nativeHold{}, closeErr
192: 		}
193: 		h, e = windows.CreateFile(w, access,
194: 			windows.FILE_SHARE_READ|windows.FILE_SHARE_WRITE, nil, windows.OPEN_EXISTING,
195: 			windows.FILE_FLAG_OPEN_REPARSE_POINT|windows.FILE_FLAG_BACKUP_SEMANTICS, 0)
196: 		if e != nil {
197: 			return nativeHold{}, e
198: 		}
199: 		rawSD, sddl, semantic, e = nativeSecurityDescriptor(h, false)
200: 	}
201: 	if e != nil {
202: 		return fail(e)
203: 	}
204: 	var info windows.ByHandleFileInformation
205: 	if e = windows.GetFileInformationByHandle(h, &info); e != nil {
206: 		return fail(e)
207: 	}
208: 	if info.FileAttributes&windows.FILE_ATTRIBUTE_REPARSE_POINT != 0 || (info.FileAttributes&windows.FILE_ATTRIBUTE_DIRECTORY != 0) != directory {
209: 		return fail(errors.New("reparse point or file kind mismatch"))
210: 	}
211: 	if !directory && info.NumberOfLinks != 1 {
212: 		return fail(errors.New("hard-linked boundary file forbidden"))
213: 	}
214: 	v, id, e := nativeFileID(h)
215: 	if e != nil {
216: 		return fail(e)
217: 	}
218: 	final, e := finalPath(h)
219: 	if e != nil {
220: 		return fail(e)
221: 	}
222: 	x := FileIdentity{Path: path, FinalUTF16: final, Volume: v, FileID: id, SecurityDescriptor: sddl, NativeSecurityDescriptor: rawSD, NativeSecuritySHA256: semantic, SACLReadStatus: saclStatus, SACLUnavailableStage: saclUnavailableStage, SACLUnavailableWin32: saclUnavailableWin32, SACLUnavailableNTSTATUS: saclUnavailableNTSTATUS, Directory: directory}
223: 	if hashFile {
224: 		if directory {
225: 			return fail(errors.New("cannot hash directory"))
226: 		}
227: 		s := sha256.New()
228: 		var total int64
229: 		buf := make([]byte, 64<<10)
230: 		for {
231: 			var n uint32
232: 			e = windows.ReadFile(h, buf, &n, nil)
233: 			if e != nil {
234: 				return fail(e)
235: 			}
236: 			if n == 0 {
237: 				break
238: 			}
239: 			total += int64(n)
240: 			if total > 128<<20 {
241: 				return fail(errors.New("hash input exceeds bound"))
242: 			}
243: 			_, _ = s.Write(buf[:n])
244: 		}
245: 		x.SHA256 = hex.EncodeToString(s.Sum(nil))
246: 	}
247: 	return nativeHold{path: path, handle: h, identity: x}, nil
248: }

271: func pathAncestors(path string) []string {
272: 	var xs []string
273: 	for p := filepath.Dir(path); ; p = filepath.Dir(p) {
274: 		xs = append(xs, p)
275: 		if filepath.Dir(p) == p {
276: 			break
277: 		}

279: 	for i, j := 0, len(xs)-1; i < j; i, j = i+1, j-1 {
280: 		xs[i], xs[j] = xs[j], xs[i]
281: 	}
282: 	return xs

285: func openPlanHolds(p EntryFencePlan) ([]nativeHold, error) {
286: 	all := []FileIdentity{p.DataRoot, p.Database, p.Helper}
287: 	all = append(all, p.Ancestors...)
288: 	for _, t := range p.Targets {
289: 		all = append(all, t.File)
290: 	}
291: 	want := map[string]FileIdentity{}
292: 	for _, x := range all {
293: 		semantic, e := nativeSecuritySemanticSHA256(x.NativeSecurityDescriptor)
294: 		if e != nil || semantic != x.NativeSecuritySHA256 {
295: 			return nil, fmt.Errorf("native security preimage invalid for %s: %w", x.Path, e)
296: 		}
297: 		sd := (*windows.SECURITY_DESCRIPTOR)(unsafe.Pointer(&x.NativeSecurityDescriptor[0]))
298: 		if !sd.IsValid() || sd.String() != x.SecurityDescriptor {
299: 			return nil, fmt.Errorf("native security preimage and SDDL disagree for %s", x.Path)
300: 		}
301: 		want[strings.ToLower(x.Path)] = x
302: 	}
303: 	for _, target := range p.Targets {
304: 		_, semantic, e := nativeSecurityFromSDDL(target.InstalledSD)
305: 		if e != nil || semantic != target.InstalledNativeSecuritySHA256 {
306: 			return nil, fmt.Errorf("installed native descriptor digest invalid for %s: %w", target.File.Path, e)
307: 		}
308: 	}
309: 	for _, x := range all {
310: 		for _, a := range pathAncestors(x.Path) {
311: 			if _, ok := want[strings.ToLower(a)]; !ok {
312: 				return nil, fmt.Errorf("ancestor absent from frozen inventory: %s", a)
313: 			}
314: 		}
315: 	}
316: 	keys := make([]string, 0, len(want))
317: 	for k := range want {
318: 		keys = append(keys, k)
319: 	}
320: 	sort.Strings(keys)
321: 	var holds []nativeHold
322: 	for _, k := range keys {
323: 		x := want[k]
324: 		h, e := openNative(x.Path, x.Directory, !x.Directory)
325: 		if e != nil {
326: 			closeHolds(holds)
327: 			return nil, fmt.Errorf("open boundary identity %s: %w", x.Path, e)
328: 		}
329: 		matches := identityEqual(x, h.identity)
330: 		if !matches {
331: 			for _, target := range p.Targets {
332: 				if strings.EqualFold(target.File.Path, x.Path) && identityCoreEqual(x, h.identity) && target.InstalledSD == h.identity.SecurityDescriptor && target.InstalledNativeSecuritySHA256 == h.identity.NativeSecuritySHA256 {
333: 					matches = true
334: 					break
335: 				}
336: 			}
337: 		}
338: 		if !matches {
339: 			windows.CloseHandle(h.handle)
340: 			closeHolds(holds)
341: 			return nil, fmt.Errorf("boundary identity drift: %s", x.Path)
342: 		}
343: 		holds = append(holds, h)
344: 	}
345: 	return holds, nil
346: }

348: func checkHolds(holds []nativeHold) error {
349: 	if len(holds) == 0 {
350: 		return errors.New("native boundary holds missing")
351: 	}
352: 	for _, h := range holds {
353: 		fresh, e := openNative(h.path, h.identity.Directory, !h.identity.Directory)
354: 		if e != nil {
355: 			return e
356: 		}
357: 		same := identityEqual(h.identity, fresh.identity)
358: 		if e = windows.CloseHandle(fresh.handle); e != nil {
359: 			return e
360: 		}
361: 		if !same {
362: 			return fmt.Errorf("native boundary identity changed: %s", h.path)
363: 		}
364: 	}
365: 	return nil
366: }

369: 	if _, e := windows.StringToSid(businessSID); e != nil {

378: 	if e != nil || v > 1 {

390: func restartSettingsForProcess(h windows.Handle) (string, uint32, bool, error) {
391: 	var size, flags uint32
392: 	r, _, _ := getApplicationRestartSettings.Call(uintptr(h), 0, uintptr(unsafe.Pointer(&size)), uintptr(unsafe.Pointer(&flags)))
393: 	if uint32(r) == 0x80070490 {
394: 		return "", 0, false, nil
395: 	} // HRESULT_FROM_WIN32(ERROR_NOT_FOUND)
396: 	if uint32(r) != 0 || size == 0 || size > 1025 {
397: 		return "", 0, false, fmt.Errorf("restart registration size/status invalid: %#x/%d", uint32(r), size)
398: 	}
399: 	buf := make([]uint16, size)
400: 	r, _, _ = getApplicationRestartSettings.Call(uintptr(h), uintptr(unsafe.Pointer(&buf[0])), uintptr(unsafe.Pointer(&size)), uintptr(unsafe.Pointer(&flags)))
401: 	if uint32(r) != 0 || size == 0 || size > uint32(len(buf)) || flags&^uint32(0xf) != 0 {
402: 		return "", 0, false, fmt.Errorf("restart registration read/status invalid: %#x/%d/%#x", uint32(r), size, flags)
403: 	}
404: 	return windows.UTF16ToString(buf[:size]), flags, true, nil

407: func bootClock() (uint64, time.Time, error) {
408: 	u, _, e := getTick.Call()
409: 	if e != windows.ERROR_SUCCESS && e != nil {
410: 		return 0, time.Time{}, e
411: 	}
412: 	var x struct {
413: 		BootTime          int64
414: 		CurrentTime       int64
415: 		TimeZoneBias      int64
416: 		CurrentTimeZoneID uint32
417: 		Reserved          uint32
418: 		BootTimeBias      uint64
419: 		SleepTimeBias     uint64
420: 	}
421: 	r, _, e := ntSystem.Call(3, uintptr(unsafe.Pointer(&x)), unsafe.Sizeof(x), 0)
422: 	if r != 0 {
423: 		return 0, time.Time{}, fmt.Errorf("NtQuerySystemInformation status %#x: %v", r, e)
424: 	}
425: 	if x.BootTime <= 0 {
426: 		return 0, time.Time{}, errors.New("native boot time unavailable")
427: 	}
428: 	return uint64(u), time.Unix(0, (x.BootTime-116444736000000000)*100).UTC(), nil
429: }

455: func readRecord(p EntryFencePlan) (boundaryRecord, error) {
456: 	var r boundaryRecord
457: 	dir, e := stateRoot(p)
458: 	if e != nil {
459: 		return r, e
460: 	}
461: 	// The operation directory and every record must be independently protected.
462: 	if e = checkManagedChain(p); e != nil {
463: 		return r, e
464: 	}
465: 	if e = checkAcceptanceFiles(p); e != nil {
466: 		return r, e
467: 	}
468: 	entries, e := os.ReadDir(dir)
469: 	if e != nil {
470: 		return r, e
471: 	}
472: 	if len(entries) < 2 || len(entries) > 24 {
473: 		return r, errors.New("boundary state inventory invalid")
474: 	}
475: 	for _, entry := range entries {
476: 		if entry.IsDir() || (entry.Name() != "plan.json" && entry.Name() != "pre.json" && entry.Name() != "post.json" && entry.Name() != "full-restart-complete.json" && entry.Name() != "source-accepted.json" && entry.Name() != "native-accepted.json" && entry.Name() != "backup-verified.json" && entry.Name() != "recovery-ready.json" && !strings.HasPrefix(entry.Name(), "event-")) {
477: 			return r, errors.New("unknown boundary state entry")
478: 		}
479: 	}
480: 	raw, _, e := readManagedFile(filepath.Join(dir, "plan.json"), p.ManagementSID, maxBoundaryPlanBytes)
481: 	if e != nil {
482: 		return r, errors.New("boundary plan file unavailable")
483: 	}
484: 	if e = json.Unmarshal(raw, &r.Plan); e != nil {
485: 		return r, e
486: 	}
487: 	if r.Plan.OperationID != p.OperationID || r.Plan.PlanSHA256 != p.PlanSHA256 {
488: 		return r, errors.New("boundary record plan mismatch")
489: 	}
490: 	pre, _, e := readManagedFile(filepath.Join(dir, "pre.json"), p.ManagementSID, 16<<10)
491: 	if e != nil {
492: 		return r, errors.New("boundary pre witness unavailable")
493: 	}
494: 	if e = json.Unmarshal(pre, &r.Pre); e != nil {
495: 		return r, e
496: 	}
497: 	post, _, e := readManagedFile(filepath.Join(dir, "post.json"), p.ManagementSID, 16<<10)
498: 	if e == nil {
499: 		if e = json.Unmarshal(post, &r.Post); e != nil {
500: 			return r, e
501: 		}
502: 	} else if !errors.Is(e, os.ErrNotExist) {
503: 		return r, e
504: 	}
505: 	for i := 1; i <= 16; i++ {
506: 		name := fmt.Sprintf("event-%02d.json", i)
507: 		raw, _, e := readManagedFile(filepath.Join(dir, name), p.ManagementSID, 16<<10)
508: 		if errors.Is(e, os.ErrNotExist) {
509: 			break
510: 		}
511: 		if e != nil {
512: 			return r, errors.New("boundary event missing or oversized")
513: 		}

552: func checkManagedDescriptor(sd *windows.SECURITY_DESCRIPTOR, sid string) error {
553: 	if sd == nil {
554: 		return errors.New("management descriptor missing")
555: 	}
556: 	owner, _, e := sd.Owner()
557: 	if e != nil || owner == nil || owner.String() != sid {
558: 		return errors.New("boundary management owner mismatch")
559: 	}
560: 	control, _, e := sd.Control()
561: 	if e != nil || control&windows.SE_DACL_PROTECTED == 0 {
562: 		return errors.New("boundary state DACL not protected")
563: 	}
564: 	acl, _, e := sd.DACL()
565: 	if e != nil || acl == nil || acl.AceCount < 1 || acl.AceCount > 2 {
566: 		return errors.New("boundary state DACL inventory invalid")
567: 	}
568: 	managerFull := false
569: 	for i := uint32(0); i < uint32(acl.AceCount); i++ {
570: 		var ace *windows.ACCESS_ALLOWED_ACE
571: 		if e = windows.GetAce(acl, i, &ace); e != nil {
572: 			return e
573: 		}
574: 		if ace.Header.AceType != windows.ACCESS_ALLOWED_ACE_TYPE || ace.Header.AceFlags&windows.INHERIT_ONLY_ACE != 0 {
575: 			return errors.New("boundary state ACE type forbidden")
576: 		}
577: 		trustee := (*windows.SID)(unsafe.Pointer(&ace.SidStart)).String()
578: 		if trustee != sid && trustee != "S-1-5-18" {
579: 			return errors.New("boundary state nonmanagement ACE forbidden")
580: 		}
581: 		if trustee == sid && uint32(ace.Mask)&0x001f01ff == 0x001f01ff {
582: 			managerFull = true

584: 	}
585: 	if !managerFull {
586: 		return errors.New("boundary state manager full access absent")
587: 	}
588: 	return nil

594: 	h, err := openNativeAccess(path, false, false, 0)

840: 	var shape businessTokenShape
841: 	var session, elevation, size uint32
842: 	if err := windows.ProcessIdToSessionId(pid, &session); err != nil {
843: 		return shape, err
844: 	}

846: 		return shape, errors.Join(err, errors.New("token elevation type unavailable"))

849: 	if err != nil || len(labelBytes) < int(unsafe.Sizeof(windows.Tokenmandatorylabel{})) {

866: 	}

873: // This is admission for the native business token only. A privilege present
874: // but disabled can be enabled later, so its Attributes cannot make it safe.
875: // The management token's separate owner-restoration path is unaffected.
876: func rejectFenceBypassPrivileges(raw []byte) error {
877: 	const countBytes = 4
878: 	if len(raw) < countBytes {
879: 		return errors.New("business token privilege inventory truncated")
880: 	}
881: 	count := binary.LittleEndian.Uint32(raw[:countBytes])
882: 	start := uint64(unsafe.Offsetof(windows.Tokenprivileges{}.Privileges))
883: 	stride := uint64(unsafe.Sizeof(windows.LUIDAndAttributes{}))
884: 	if start+uint64(count)*stride > uint64(len(raw)) {
885: 		return errors.New("business token privilege inventory length invalid")
886: 	}
887: 	privileges := (*windows.Tokenprivileges)(unsafe.Pointer(&raw[0])).AllPrivileges()
888: 	for _, name := range [...]string{"SeTakeOwnershipPrivilege", "SeRestorePrivilege", "SeBackupPrivilege"} {
889: 		namePtr, err := windows.UTF16PtrFromString(name)
890: 		if err != nil {
891: 			return err
892: 		}
893: 		var luid windows.LUID
894: 		if err := windows.LookupPrivilegeValue(nil, namePtr, &luid); err != nil {
895: 			return fmt.Errorf("business token privilege lookup %s: %w", name, err)
896: 		}
897: 		for _, privilege := range privileges {
898: 			if privilege.Luid == luid {
899: 				return fmt.Errorf("business token assigned fence-bypass privilege %s", name)
900: 			}
901: 		}
902: 	}
903: 	return nil
904: }

906: // Existing Explorer supplies a fresh business process token after restart.
907: // Its current PID, session and logon SID are evidence, never compared to the
908: // pre-restart values. Elevated-admin cross-account TOKEN_QUERY is probed.
909: func inspectExistingBusinessProcess(p EntryFencePlan, expected string, pre BootWitness) (businessTokenShape, error) {
910: 	var zero businessTokenShape
911: 	if pre.BusinessLogonSID == "" || !validDigest(pre.BusinessPrivilegesSHA256) || pre.BusinessIntegritySID == "" {
912: 		return zero, errors.New("pre business token shape unavailable")
913: 	}
914: 	systemRoot, rootErr := windows.GetSystemWindowsDirectory()
915: 	if rootErr != nil || !filepath.IsAbs(systemRoot) || filepath.Clean(systemRoot) != systemRoot {
916: 		return zero, errors.Join(rootErr, errors.New("native system root unavailable"))
917: 	}
918: 	explorerPath := filepath.Join(systemRoot, "explorer.exe")
919: 	_, currentBoot, err := bootClock()
920: 	if err != nil {
921: 		return zero, err
922: 	}
923: 	snapshot, err := windows.CreateToolhelp32Snapshot(windows.TH32CS_SNAPPROCESS, 0)
924: 	if err != nil {
925: 		return zero, err
926: 	}
927: 	defer windows.CloseHandle(snapshot)
928: 	var entry windows.ProcessEntry32
929: 	entry.Size = uint32(unsafe.Sizeof(entry))
930: 	if err = windows.Process32First(snapshot, &entry); err != nil {
931: 		return zero, err
932: 	}
933: 	var matched *businessTokenShape
934: 	for {
935: 		if strings.EqualFold(windows.UTF16ToString(entry.ExeFile[:]), "explorer.exe") {
936: 			proc, openErr := windows.OpenProcess(windows.PROCESS_QUERY_LIMITED_INFORMATION, false, entry.ProcessID)
937: 			if openErr != nil {
938: 				return zero, fmt.Errorf("Explorer candidate inaccessible: %w", openErr)
939: 			}
940: 			buf := make([]uint16, 32768)
941: 			n := uint32(len(buf))
942: 			imageErr := windows.QueryFullProcessImageName(proc, 0, &buf[0], &n)
943: 			if imageErr != nil {
944: 				windows.CloseHandle(proc)
945: 				return zero, imageErr
946: 			}
947: 			if !strings.EqualFold(windows.UTF16ToString(buf[:n]), explorerPath) {
948: 				windows.CloseHandle(proc)
949: 				goto nextProcess
950: 			}
951: 			var created, exited, kernel, userTime windows.Filetime
952: 			if err = windows.GetProcessTimes(proc, &created, &exited, &kernel, &userTime); err != nil || time.Unix(0, created.Nanoseconds()).Before(currentBoot.Add(-2*time.Second)) {
953: 				windows.CloseHandle(proc)
954: 				return zero, errors.Join(err, errors.New("Explorer creation predates current boot"))
955: 			}
956: 			var token windows.Token
957: 			tokenErr := windows.OpenProcessToken(proc, windows.TOKEN_QUERY, &token)
958: 			if tokenErr != nil {
959: 				windows.CloseHandle(proc)
960: 				return zero, tokenErr
961: 			}
962: 			user, userErr := token.GetTokenUser()
963: 			if userErr != nil {
964: 				token.Close()
965: 				windows.CloseHandle(proc)
966: 				return zero, userErr
967: 			}
968: 			if user.User.Sid.String() != p.BusinessSID {
969: 				token.Close()
970: 				windows.CloseHandle(proc)
971: 				goto nextProcess
972: 			}
973: 			shape, shapeErr := tokenSecurityShape(token, entry.ProcessID, p.BusinessSID)
974: 			closeErr := token.Close()
975: 			if shapeErr != nil || closeErr != nil {
976: 				windows.CloseHandle(proc)
977: 				return zero, errors.Join(shapeErr, closeErr)
978: 			}
979: 			imageSHA, imageErr := osProcessImageIdentity(explorerPath)
980: 			closeErr = windows.CloseHandle(proc)
981: 			if imageErr != nil || closeErr != nil {
982: 				return zero, errors.Join(imageErr, closeErr)
983: 			}
984: 			shape.ProcessID = entry.ProcessID
985: 			shape.ProcessCreated = time.Unix(0, created.Nanoseconds()).UTC()
986: 			shape.ProcessImageSHA = imageSHA
987: 			if err := sameBusinessPermissionSemantics(pre, shape, expected); err != nil {
988: 				return zero, err
989: 			}
990: 			if matched != nil {
991: 				if matched.SessionID != shape.SessionID || matched.LogonSID != shape.LogonSID || matched.ProcessImageSHA != shape.ProcessImageSHA {
992: 					return zero, errors.New("multiple current business Explorer sessions ambiguous")
993: 				}
994: 				if shape.ProcessCreated.Before(matched.ProcessCreated) {
995: 					matched = &shape
996: 				}
997: 			} else {
998: 				matched = &shape
999: 			}
1000: 		}
1001: 	nextProcess:
1002: 		err = windows.Process32Next(snapshot, &entry)
1003: 		if errors.Is(err, windows.ERROR_NO_MORE_FILES) {
1004: 			break
1005: 		}
1006: 		if err != nil {
1007: 			return zero, err
1008: 		}
1009: 	}
1010: 	if matched == nil {
1011: 		return zero, errors.New("current business Explorer token unavailable")
1012: 	}
1013: 	return *matched, nil
1014: }

1901: func setTargetDescriptor(expected FileIdentity, desiredSD string, desiredNativeRaw []byte) (outErr error) {
1902: 	var sd *windows.SECURITY_DESCRIPTOR
1903: 	var desiredSemantic string
1904: 	var e error
1905: 	if len(desiredNativeRaw) > 0 {
1906: 		desiredSemantic, e = nativeSecuritySemanticSHA256(desiredNativeRaw)
1907: 		if e != nil {
1908: 			return e
1909: 		}
1910: 		sd = (*windows.SECURITY_DESCRIPTOR)(unsafe.Pointer(&desiredNativeRaw[0]))
1911: 		if !sd.IsValid() || sd.String() != desiredSD {
1912: 			return errors.New("desired native security preimage differs from plan SDDL")
1913: 		}
1914: 	} else {
1915: 		sd, e = windows.SecurityDescriptorFromString(desiredSD)
1916: 		if e != nil {
1917: 			return e
1918: 		}
1919: 		_, desiredSemantic, e = nativeSecurityFromSDDL(desiredSD)
1920: 		if e != nil {
1921: 			return e
1922: 		}
1923: 	}
1924: 	owner, _, e := sd.Owner()
1925: 	if e != nil || owner == nil {
1926: 		return errors.New("target owner descriptor missing")
1927: 	}
1928: 	dacl, _, e := sd.DACL()
1929: 	if e != nil || dacl == nil {
1930: 		return errors.Join(e, errors.New("target DACL descriptor missing"))
1931: 	}
1932: 	prior, e := windows.SecurityDescriptorFromString(expected.SecurityDescriptor)
1933: 	if e != nil {
1934: 		return e
1935: 	}
1936: 	priorOwner, _, e := prior.Owner()
1937: 	if e != nil || priorOwner == nil {
1938: 		return errors.New("expected target owner descriptor missing")
1939: 	}
1940: 	priorControl, _, e := prior.Control()
1941: 	if e != nil {
1942: 		return e
1943: 	}
1944: 	control, _, e := sd.Control()
1945: 	if e != nil {
1946: 		return e
1947: 	}
1948: 	if (priorControl&windows.SE_DACL_AUTO_INHERITED != 0) != (control&windows.SE_DACL_AUTO_INHERITED != 0) {
1949: 		return errors.New("target DACL auto-inheritance class drift")
1950: 	}
1951: 	access := uint32(windows.WRITE_DAC)
1952: 	flags := windows.SECURITY_INFORMATION(windows.DACL_SECURITY_INFORMATION)
1953: 	if !priorOwner.Equals(owner) {
1954: 		access |= windows.WRITE_OWNER
1955: 		flags |= windows.OWNER_SECURITY_INFORMATION
1956: 	}
1957: 	hold, e := openNativeAccess(expected.Path, false, true, access)
1958: 	if e != nil {
1959: 		return fmt.Errorf("target mutation open READ_CONTROL|READ_DATA|WRITE_DAC plus owner-change access %#x: %w", access, e)
1960: 	}
1961: 	defer func() { outErr = errors.Join(outErr, windows.CloseHandle(hold.handle)) }()
1962: 	if !identityEqual(hold.identity, expected) {
1963: 		return errors.New("target same-handle native identity or descriptor changed before mutation")
1964: 	}
1965: 	if control&windows.SE_DACL_AUTO_INHERITED != 0 {
1966: 		// SetSecurityInfo retains the auto-inherited class. NtSetSecurityObject
1967: 		// clears it on this filesystem, so the frozen descriptor selects one
1968: 		// path before mutation; there is no failure-time fallback.
1969: 		setFlags := flags | windows.UNPROTECTED_DACL_SECURITY_INFORMATION
1970: 		if control&windows.SE_DACL_PROTECTED != 0 {
1971: 			setFlags = flags | windows.PROTECTED_DACL_SECURITY_INFORMATION
1972: 		}
1973: 		if e = windows.SetSecurityInfo(hold.handle, windows.SE_FILE_OBJECT, setFlags, owner, nil, dacl, nil); e != nil {
1974: 			return fmt.Errorf("target mutation SetSecurityInfo owner/DACL: %w", e)
1975: 		}
1976: 	} else {
1977: 		if e = ntSetSecurityObject.Find(); e != nil {
1978: 			return fmt.Errorf("NtSetSecurityObject unavailable: %w", e)
1979: 		}
1980: 		r, _, _ := ntSetSecurityObject.Call(uintptr(hold.handle), uintptr(flags), uintptr(unsafe.Pointer(sd)))
1981: 		runtime.KeepAlive(sd)
1982: 		if status := windows.NTStatus(r); status != 0 {
1983: 			return fmt.Errorf("target mutation NtSetSecurityObject status %#x: %w", uint32(status), status.Errno())
1984: 		}
1985: 	}
1986: 	_, readbackSDDL, readbackSemantic, e := nativeSecurityDescriptor(hold.handle, hold.identity.SACLReadStatus == "READABLE")
1987: 	if e != nil {
1988: 		return fmt.Errorf("target mutation GetSecurityInfo readback: %w", e)
1989: 	}
1990: 	if readbackSDDL != desiredSD || readbackSemantic != desiredSemantic {
1991: 		return fmt.Errorf("target descriptor same-handle readback mismatch: expected SDDL %q semantic %s observed SDDL %q semantic %s", desiredSD, desiredSemantic, readbackSDDL, readbackSemantic)
1992: 	}
1993: 	v, id, e := nativeFileID(hold.handle)
1994: 	if e != nil {
