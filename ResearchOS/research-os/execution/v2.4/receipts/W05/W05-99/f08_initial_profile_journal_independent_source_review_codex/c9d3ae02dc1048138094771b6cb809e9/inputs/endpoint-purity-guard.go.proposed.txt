const f08ProfileEndpointGlobalGap=f08Error("F08_PROFILE_ENDPOINT_REQUIRES_UNCAPTURED_GLOBAL_SEMANTICS")
func f08PureEndpointPrecondition(endpoint string) error {
	rest:=endpoint
	if i:=strings.Index(rest,"://");i>=0 {rest=rest[i+3:]} else if strings.HasPrefix(rest,"//"){rest=rest[2:]}else{return f08Binding}
	if i:=strings.IndexAny(rest,"/?#");i>=0{rest=rest[:i]}

	// conservative: an OPEN result still refuses, never creates authority.

	return nil
}
