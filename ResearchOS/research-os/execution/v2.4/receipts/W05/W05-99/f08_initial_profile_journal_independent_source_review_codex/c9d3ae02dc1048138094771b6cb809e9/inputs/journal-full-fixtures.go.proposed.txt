func TestF08FullJournalSingleFactMutations(t *testing.T){
	cases:=[]struct{name string;change func(*f08JournalFixture)}{
		{"missing_admitted_pin",func(f *f08JournalFixture){for i,x:=range f.rows{p:=x.([]any);if p[0]=="pins"&&p[1].([]any)[4]=="ADMITTED"{f.rows=append(f.rows[:i],f.rows[i+1:]...);return}}}},
