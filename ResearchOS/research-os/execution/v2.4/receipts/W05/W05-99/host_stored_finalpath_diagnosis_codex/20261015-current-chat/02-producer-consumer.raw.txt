15: // fileIdentity pins a directory by volume serial plus file index and reports the resolved
16: // final path, which is how a junction or symlink substitution is detected later.
17: func fileIdentity(path string) (volume string, fileID uint64, final string, err error) {
18: 	wide, err := windows.UTF16PtrFromString(path)
19: 	if err != nil {
20: 		return "", 0, "", err
21: 	}
22: 	h, err := windows.CreateFile(wide, 0, windows.FILE_SHARE_READ|windows.FILE_SHARE_WRITE|windows.FILE_SHARE_DELETE,
23: 		nil, windows.OPEN_EXISTING, windows.FILE_FLAG_BACKUP_SEMANTICS, 0)
24: 	if err != nil {
25: 		return "", 0, "", fmt.Errorf("open %s: %w", path, err)
26: 	}
27: 	defer windows.CloseHandle(h)
28: 	var info windows.ByHandleFileInformation
29: 	if err := windows.GetFileInformationByHandle(h, &info); err != nil {
30: 		return "", 0, "", err
31: 	}
32: 	vol := fmt.Sprintf("%08x", info.VolumeSerialNumber)
33: 	id := uint64(info.FileIndexHigh)<<32 | uint64(info.FileIndexLow)
34: 	buf := make([]uint16, 4096)
35: 	n, err := windows.GetFinalPathNameByHandle(h, &buf[0], uint32(len(buf)), 0)
36: 	if err != nil || n == 0 {
37: 		return vol, id, "", fmt.Errorf("final path unavailable for %s: %v", path, err)
38: 	}
39: 	final = windows.UTF16PtrToString(&buf[0])
40: 	if l := int(n); len(final) > l {
41: 		final = final[:l]
42: 	}
43: 	final = strings.TrimPrefix(final, `\\?\`)
44: 	return vol, id, final, nil
45: }

47: // chainHasReparsePoint walks every prefix of the path, so a junction anywhere above the target
48: // is caught even when the final component is an ordinary file.
49: func chainHasReparsePoint(path string) (bool, error) {
50: 	abs, err := filepath.Abs(path)
51: 	if err != nil {
52: 		return false, err
53: 	}
54: 	parts := strings.Split(abs, string(filepath.Separator))
55: 	var cur string
56: 	for _, p := range parts {
57: 		if p == "" {
58: 			cur = abs[:1] + string(filepath.Separator)
59: 			continue
60: 		}
61: 		if cur == "" {
62: 			cur = p + string(filepath.Separator)
63: 			continue
64: 		}
65: 		cur = filepath.Join(cur, p)
66: 		reparse, err := isReparsePoint(cur)
67: 		if err != nil {
68: 			// Not existing yet is normal for a write target; only existing components can be
69: 			// reparse points.
70: 			continue
71: 		}
72: 		if reparse {
73: 			return true, nil
74: 		}
75: 	}
76: 	return false, nil
77: }

79: func isReparsePoint(path string) (bool, error) {
80: 	wide, err := windows.UTF16PtrFromString(path)
81: 	if err != nil {
82: 		return false, err
83: 	}
84: 	h, err := windows.CreateFile(wide, 0, windows.FILE_SHARE_READ|windows.FILE_SHARE_WRITE|windows.FILE_SHARE_DELETE,
85: 		nil, windows.OPEN_EXISTING, windows.FILE_FLAG_BACKUP_SEMANTICS|windows.FILE_FLAG_OPEN_REPARSE_POINT, 0)
86: 	if err != nil {
87: 		return false, err

94: 	return info.FileAttributes&fileAttributeReparsePoint != 0, nil

88: // Register pins the chosen directory. A path that is itself a reparse point, or whose parent
89: // chain contains one, is refused: the binding must describe the real directory the user picked.
90: func (r *RootRegistry) Register(ref, displayPath, access string, now time.Time) (*RootBinding, error) {
91: 	if ref == "" || displayPath == "" {
92: 		return nil, fmt.Errorf("%w: root ref and path are required", ErrBadState)
93: 	}
94: 	if access != AccessReadOnly && access != AccessReadWrite {
95: 		return nil, fmt.Errorf("%w: unknown workspace access", ErrRootDenied)
96: 	}
97: 	abs, err := filepath.Abs(displayPath)
98: 	if err != nil {
99: 		return nil, err
100: 	}
101: 	real, err := filepath.EvalSymlinks(abs)
102: 	if err != nil {
103: 		return nil, err
104: 	}
105: 	if cleanReal := filepath.Clean(real); cleanReal != filepath.Clean(abs) {
106: 		return nil, fmt.Errorf("%w: chosen path resolves through a link", ErrPathReparsePoint)
107: 	}
108: 	if reparse, err := chainHasReparsePoint(clean(real)); err != nil {
109: 		return nil, err
110: 	} else if reparse {
111: 		return nil, ErrPathReparsePoint
112: 	}
113: 	vol, id, final, err := fileIdentity(clean(real))
114: 	if err != nil {
115: 		return nil, err
116: 	}
117: 	r.mu.Lock()
118: 	defer r.mu.Unlock()
119: 	if existing, ok := r.roots[ref]; ok && !existing.Revoked {
120: 		return nil, fmt.Errorf("%w: root ref already registered", ErrBadState)
121: 	}
122: 	b := &RootBinding{Ref: ref, DisplayPath: displayPath, RealPath: clean(real), Access: access,
123: 		VolumeID: vol, FileID: id, FinalPath: final, RegisteredAt: now.UTC().Format(time.RFC3339Nano)}
124: 	r.roots[ref] = b
125: 	if err := r.save(); err != nil {
126: 		delete(r.roots, ref)
127: 		return nil, err
128: 	}
129: 	return b, nil
130: }

132: // Resolve validates one published relative path against a bound root and returns the absolute
133: // path to use. Every check is re-done per access: registration time correctness does not grant
134: // later immunity, because links and permissions can change underneath.
135: func (r *RootRegistry) Resolve(ref, rel, wantAccess string) (string, error) {
136: 	r.mu.RLock()
137: 	b, ok := r.roots[ref]
138: 	r.mu.RUnlock()
139: 	if !ok || b == nil {
140: 		return "", ErrRootUnknown
141: 	}
142: 	if b.Revoked {
143: 		return "", ErrRootRevoked
144: 	}
145: 	if wantAccess == AccessReadWrite && b.Access == AccessReadOnly {
146: 		return "", ErrRootDenied
147: 	}
148: 	if err := CheckRelativePath(rel); err != nil {
149: 		return "", err
150: 	}
151: 	if err := r.verifyStillPinned(b); err != nil {
152: 		return "", err
153: 	}
154: 	vol, id, _, err := fileIdentity(b.RealPath)
155: 	if err != nil {
156: 		return "", ErrRootRevoked
157: 	}
158: 	if vol != b.VolumeID || id != b.FileID {
159: 		// Same path, different object: the authorised root was replaced under us.
160: 		return "", ErrPathReparsePoint
161: 	}
162: 	abs := filepath.Join(b.RealPath, filepath.FromSlash(NormalizeRelative(rel)))
163: 	inside, err := filepath.Rel(b.RealPath, abs)
164: 	if err != nil || inside == ".." || strings.HasPrefix(inside, ".."+string(filepath.Separator)) {
165: 		return "", ErrPathEscape
166: 	}
167: 	if reparse, err := chainHasReparsePoint(abs); err != nil {
168: 		return "", err
169: 	} else if reparse {
170: 		return "", ErrPathReparsePoint
171: 	}
172: 	return abs, nil
173: }

15: // current OS principal originally authorized. It rejects, rather than skips,

17: func (h *Host) SnapshotWorkspace(ctx context.Context, binding domain.WorkspaceBinding, destination string, limits domain.BackupWorkspaceLimits) (domain.BackupWorkspaceSnapshot, error) {
18: 	var out domain.BackupWorkspaceSnapshot
19: 	if _, err := h.reverify(); err != nil {
20: 		return out, err
21: 	}
22: 	if h.boundPrincipal() == "" || h.boundPrincipal() != binding.PrincipalID || binding.Status != domain.WorkspaceActive {
23: 		return out, fmt.Errorf("BACKUP_WORKSPACE_OWNER_DENIED")
24: 	}
25: 	if limits.MaxFiles < 0 || limits.MaxDirs < 0 || limits.MaxBytes < 0 || limits.MaxFileBytes < 1 || limits.MaxFileBytes > domain.BackupWorkspaceMaxFileBytes {
26: 		return out, fmt.Errorf("BACKUP_WORKSPACE_LIMIT_INVALID")
27: 	}
28: 	var pinned *RootBinding
29: 	for _, root := range h.roots.List() {
30: 		if root.Revoked || !strings.EqualFold(filepath.Clean(root.RealPath), filepath.Clean(binding.RealPath)) {

33: 		if binding.Access == AccessReadWrite && root.Access != AccessReadWrite {
34: 			return out, fmt.Errorf("BACKUP_WORKSPACE_ACCESS_MISMATCH")
35: 		}
36: 		if pinned != nil {
37: 			return out, fmt.Errorf("BACKUP_WORKSPACE_AMBIGUOUS_ROOT")
38: 		}
39: 		pinned = root
40: 	}
41: 	if pinned == nil {
42: 		return out, fmt.Errorf("BACKUP_WORKSPACE_ROOT_NOT_PINNED")
43: 	}
44: 	// A restored workspace lives under its new data root by design. Allow only
45: 	// that exact, currently pinned ref and path; every other data-root overlap
46: 	// remains recursive/unsafe, as does any backup output inside the workspace.
47: 	dataRoot := filepath.Dir(h.dir)
48: 	if backupPathsOverlap(binding.RealPath, destination) ||
49: 		(backupPathsOverlap(binding.RealPath, dataRoot) && !backupRestoredWorkspacePath(dataRoot, binding.AuthorizedRef, pinned)) {
50: 		return out, fmt.Errorf("BACKUP_WORKSPACE_RECURSION: workspace intersects data root or backup output")

57: 		return out, fmt.Errorf("BACKUP_WORKSPACE_ROOT_REPARSE")

60: 	if err != nil {
61: 		return out, err
62: 	}

65: 	out.Files = []domain.BackupWorkspaceFile{}

69: 			return walkErr
70: 		}
71: 		if err := ctx.Err(); err != nil {
72: 			return err
73: 		}

75: 			return nil
76: 		}

78: 		if err != nil {
79: 			return err
80: 		}

133: 		}

137: 	})
138: 	if err != nil {
139: 		return domain.BackupWorkspaceSnapshot{}, err
140: 	}
141: 	return out, nil
142: }

156: func backupPathsOverlap(left, right string) bool {
157: 	a, ea := filepath.Abs(left)
158: 	b, eb := filepath.Abs(right)
159: 	if ea != nil || eb != nil {
160: 		return true
161: 	}
162: 	a, b = filepath.Clean(a), filepath.Clean(b)
163: 	if strings.EqualFold(a, b) {
164: 		return true
165: 	}
166: 	for _, pair := range [][2]string{{a, b}, {b, a}} {
167: 		rel, err := filepath.Rel(pair[0], pair[1])
168: 		if err == nil && rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator)) && !filepath.IsAbs(rel) {
169: 			return true
170: 		}
171: 	}
172: 	return false
173: }

175: // PinRestoredWorkspace creates a new Host root record after the data root has
176: // reached its final path. It does not inherit the old host's private roots file.
177: func (h *Host) PinRestoredWorkspace(ctx context.Context, request domain.WorkspaceRestoreRequest) (domain.WorkspaceRestorePin, error) {
178: 	var pin domain.WorkspaceRestorePin
179: 	if _, err := h.reverify(); err != nil {
180: 		return pin, err
181: 	}
182: 	if h.boundPrincipal() == "" || request.PrincipalID != h.boundPrincipal() {
183: 		return pin, fmt.Errorf("RESTORE_WORKSPACE_OWNER_DENIED")
184: 	}
185: 	dataRoot := filepath.Dir(h.dir)
186: 	expected := domain.BackupWorkspaceDestination(dataRoot, request.AuthorizedRef)
187: 	if request.AuthorizedRef == "" || request.RealPath != expected || (request.Access != AccessReadOnly && request.Access != AccessReadWrite) {
188: 		return pin, fmt.Errorf("RESTORE_WORKSPACE_PATH_DENIED")
189: 	}
190: 	if reparse, err := chainHasReparsePoint(request.RealPath); err != nil || reparse {
191: 		return pin, fmt.Errorf("RESTORE_WORKSPACE_REPARSE")
192: 	}
193: 	root, err := h.roots.Register(request.AuthorizedRef, request.RealPath, request.Access, h.nowUTC())
194: 	if err != nil {
195: 		return pin, err
196: 	}
197: 	if _, err := h.roots.Resolve(root.Ref, "__researchos_pin_probe__", AccessReadOnly); err != nil {
198: 		// Resolve checks the pin; it need not open the absent probe path.
199: 		_ = h.roots.Revoke(root.Ref)
200: 		return pin, err
201: 	}
202: 	if err := ctx.Err(); err != nil {
203: 		_ = h.roots.Revoke(root.Ref)
204: 		return pin, err
205: 	}
206: 	return domain.WorkspaceRestorePin{AuthorizedRef: request.AuthorizedRef,
207: 		RealPath: root.RealPath, HostPrincipalRef: h.identity.Ref}, nil
208: }
