// Export-only test observations. HTTP budget is not product authorization.
type cli43HTTPCounts struct {
 Attempt int `json:"attempt"`; Admitted int `json:"admitted"`; Dispatched int `json:"dispatched"`
 Completed int `json:"completed"`; Rejected int `json:"rejected"`; Outstanding int `json:"outstanding"`
 Received int `json:"received"`; Responses int `json:"responses"`; BodyClosed int `json:"body_closed"`
}
type cli43HTTPBook struct {
 mu sync.Mutex; t *testing.T; dir,origin string; sequence int; stopped bool
 upstream,proxy cli43HTTPCounts; ingress int; stop []func()map[string]any; pending map[string][]int
 records map[int]map[string]any; firstOverflow any
}
var cli43HTTPRegistryMu sync.Mutex
var cli43HTTPRegistry=map[string]*cli43HTTPBook{}
type cli43ProxyKey struct{}
func cli43Book(dir string)*cli43HTTPBook{cli43HTTPRegistryMu.Lock();defer cli43HTTPRegistryMu.Unlock();return cli43HTTPRegistry[dir]}
func cli43Error(e error)any{if e==nil{return nil};return e.Error()}
func (b *cli43HTTPBook) issue(suffix string,record any){if e:=cli43Save(b.dir,suffix,record);e!=nil{b.t.Errorf("HTTP evidence: %v",e)}}
func cli43RequestKey(r *http.Request,raw []byte)string{return r.Method+" "+r.URL.RequestURI()+" "+fmt.Sprintf("%x",sha256.Sum256(raw))}
func (b *cli43HTTPBook) reserve(layer string,r *http.Request,raw []byte)(int,bool){
 b.mu.Lock();b.sequence++;seq:=b.sequence;c:=&b.upstream;if layer=="proxy"{c=&b.proxy;c.Received++};c.Attempt++
 admitted:=!b.stopped&&c.Admitted<32
 if admitted{c.Admitted++;c.Outstanding++}else{c.Rejected++;b.stopped=true}
 rec:=map[string]any{"sequence":seq,"layer":layer,"request_method":r.Method,"request_uri":r.URL.RequestURI(),"request":cli43Bytes(raw),"request_authorization":"OMITTED_EPHEMERAL_FIXTURE_CREDENTIAL","admitted":admitted,"dispatched":false,"response_observed":false,"completed":false,"request_key":cli43RequestKey(r,raw),"proxy_sequence":r.Context().Value(cli43ProxyKey{})}
 if layer=="proxy"{rec["already_received_network_attempt"]=true;rec["client_outbound_before_ingress"]=nil}
 b.records[seq]=rec
 b.issue(fmt.Sprintf("http-attempt-%04d.json",seq),rec) // Raw bytes persist before business decode/dispatch.
 first:=false
 if !admitted&&b.firstOverflow==nil{first=true;b.firstOverflow=map[string]any{"sequence":seq,"layer":layer,"reason":"ADMISSION_OR_DISPATCH_LIMIT_OR_STOPPED","already_received_network_attempt":layer=="proxy"};b.issue("http-first-overflow.json",b.firstOverflow)}
 stops:=append([]func()map[string]any{},b.stop...);b.mu.Unlock()
 if !admitted{b.t.Errorf("HTTP %s admission denied at attempt %d; no upstream dispatch",layer,seq);if first{for i,stop:=range stops{b.issue(fmt.Sprintf("http-stop-%02d.json",i),stop())}}}
 return seq,admitted
}
func (b *cli43HTTPBook) finish(seq int,fields map[string]any){
 b.mu.Lock();defer b.mu.Unlock();rec:=b.records[seq]
 for k,v:=range fields{rec[k]=v};if rec["completed"]==true{b.t.Error("duplicate HTTP completion");return}
 rec["completed"]=true;c:=&b.upstream;if rec["layer"]=="proxy"{c=&b.proxy};c.Completed++;if rec["admitted"]==true{c.Outstanding--}
 b.issue(fmt.Sprintf("http-completed-%04d.json",seq),rec)
}
func cli43PairProxy(req,received *http.Request)*http.Request{return req.WithContext(context.WithValue(req.Context(),cli43ProxyKey{},received.Context().Value(cli43ProxyKey{})))}
type cli43BudgetTransport struct{base http.RoundTripper;book *cli43HTTPBook}
func (p *cli43BudgetTransport) RoundTrip(r *http.Request)(*http.Response,error){
 b:=p.book;var raw []byte;var readErr,closeErr error
 if r.Body!=nil{
  if r.GetBody==nil{readErr=fmt.Errorf("fixture transport cannot preserve non-replayable request")}else{
   body,e:=r.GetBody();if e!=nil{readErr=e}else{raw,readErr=io.ReadAll(body);closeErr=body.Close()}
  }
 }
 seq,ok:=b.reserve("upstream",r,raw)
 if !ok{b.finish(seq,map[string]any{"pre_dispatch_rejected":true});return nil,fmt.Errorf("fixture upstream dispatch budget rejected")}
 if readErr!=nil||closeErr!=nil||r.URL.Scheme+"://"+r.URL.Host!=b.origin{
  e:=fmt.Errorf("fixture request read/close/origin: read=%v close=%v origin=%s",readErr,closeErr,r.URL.Host);b.finish(seq,map[string]any{"pre_dispatch_error":e.Error()});return nil,e
 }
 key:=cli43RequestKey(r,raw);b.mu.Lock();b.upstream.Dispatched++;b.records[seq]["dispatched"]=true;b.pending[key]=append(b.pending[key],seq);b.issue(fmt.Sprintf("http-dispatch-%04d.json",seq),b.records[seq]);b.mu.Unlock()
 resp,e:=p.base.RoundTrip(r)
 if e!=nil{b.finish(seq,map[string]any{"transport_error":e.Error(),"actual_upstream_response":nil});return resp,e}
 b.mu.Lock();b.upstream.Responses++;b.records[seq]["response_observed"]=true;b.mu.Unlock()
 resp.Body=&cli43ObservedBody{ReadCloser:resp.Body,book:b,seq:seq,status:resp.StatusCode,headers:resp.Header.Clone(),contentLength:resp.ContentLength}
 return resp,nil
}
type cli43ObservedBody struct{io.ReadCloser;book *cli43HTTPBook;seq,status int;headers http.Header;contentLength int64;raw bytes.Buffer;reads []map[string]any;eof,closed bool}
func (b *cli43ObservedBody) Read(p []byte)(int,error){n,e:=b.ReadCloser.Read(p);if n>0{b.raw.Write(p[:n])};b.reads=append(b.reads,map[string]any{"n":n,"error":cli43Error(e)});if e==io.EOF{b.eof=true};return n,e}
func (b *cli43ObservedBody) Close()error{
 e:=b.ReadCloser.Close();if b.closed{return e};b.closed=true;b.book.mu.Lock();b.book.upstream.BodyClosed++;b.book.mu.Unlock()
 b.book.finish(b.seq,map[string]any{"upstream_received_boundary":"http.RoundTripper response and consumed Body","status":b.status,"response_headers":b.headers,"response":cli43Bytes(b.raw.Bytes()),"response_content_length":b.contentLength,"read_calls":b.reads,"read_EOF_observed":b.eof,"response_close_error":cli43Error(e),"whole_response_transmitted_to_CLI_claim":false})
 return e
}
type cli43HTTPWriter struct{http.ResponseWriter;status int;headers http.Header;writes []map[string]any;body bytes.Buffer;headerCalls []map[string]any}
func (w *cli43HTTPWriter) WriteHeader(code int){
 h:=w.Header().Clone();w.headerCalls=append(w.headerCalls,map[string]any{"code":code,"handler_headers":h});if w.status==0&&code>=200{w.status=code;w.headers=h};w.ResponseWriter.WriteHeader(code)
}
func (w *cli43HTTPWriter) Write(p []byte)(int,error){
 if w.status==0{w.status=200;w.headers=w.Header().Clone()} // Do not force an extra underlying WriteHeader.
 n,e:=w.ResponseWriter.Write(p);accepted:=[]byte{};if n>=0&&n<=len(p){accepted=append(accepted,p[:n]...);w.body.Write(accepted)}
 w.writes=append(w.writes,map[string]any{"requested":cli43Bytes(p),"actual_n":n,"error":cli43Error(e),"accepted":cli43Bytes(accepted)});return n,e
}
func (w *cli43HTTPWriter) facts()map[string]any{return map[string]any{"boundary":"ResponseWriter handler commitment/accepted writes; NOT socket or CLI-client observation","status":w.status,"handler_committed_headers":w.headers,"write_header_calls":w.headerCalls,"write_calls":w.writes,"accepted_body":cli43Bytes(w.body.Bytes()),"automatic_transport_headers":nil,"actual_client_received_headers":nil,"actual_client_received_body":nil,"unobserved_socket_delivery":true}}
func cli43RecordingHandler(t *testing.T,dir string,next http.Handler)http.Handler{
 return http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
  b:=cli43Book(dir);raw,e:=io.ReadAll(r.Body);ce:=r.Body.Close();r.Body=io.NopCloser(bytes.NewReader(raw));key:=cli43RequestKey(r,raw)
  b.mu.Lock();b.ingress++;received:=b.ingress;seqs:=append([]int{},b.pending[key]...);delete(b.pending,key);b.mu.Unlock()
  observed:=&cli43HTTPWriter{ResponseWriter:w};var seq any
  if len(seqs)==1{seq=seqs[0]}else{t.Errorf("unpaired upstream ingress count=%d",len(seqs))}
  b.issue(fmt.Sprintf("upstream-received-%04d.json",received),map[string]any{"received_sequence":received,"dispatch_sequence":seq,"request":cli43Bytes(raw),"read_error":cli43Error(e),"close_error":cli43Error(ce)})
  if e!=nil||ce!=nil||len(seqs)!=1{http.Error(observed,"fixture observation error",503)}else{next.ServeHTTP(observed,r)}
  facts:=observed.facts();facts["received_sequence"]=received;facts["dispatch_sequence"]=seq;facts["request_key"]=key
  b.issue(fmt.Sprintf("upstream-handler-%04d.json",received),facts)
 })
}
func cli43ProxyHandler(t *testing.T,dir string,next http.Handler)http.Handler{
 return http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
  b:=cli43Book(dir);raw,e:=io.ReadAll(r.Body);ce:=r.Body.Close();r.Body=io.NopCloser(bytes.NewReader(raw));seq,ok:=b.reserve("proxy",r,raw)
  observed:=&cli43HTTPWriter{ResponseWriter:w}
  if !ok{http.Error(observed,"fixture admission budget",503)}else if e!=nil||ce!=nil{t.Errorf("proxy body capture: %v/%v",e,ce);http.Error(observed,"fixture request read",503)}else{
   next.ServeHTTP(observed,r.WithContext(context.WithValue(r.Context(),cli43ProxyKey{},seq)))
  }
  fields:=observed.facts();fields["request_read_error"]=cli43Error(e);fields["request_close_error"]=cli43Error(ce);fields["CLI_facing_handler_boundary"]=true
  b.finish(seq,fields)
 })
}
func cli43InstallHTTP(t *testing.T,dir string,server *httptest.Server){
 b:=&cli43HTTPBook{t:t,dir:dir,origin:server.URL,pending:map[string][]int{},records:map[int]map[string]any{}}
 b.stop=append(b.stop,func()map[string]any{e:=server.Listener.Close();server.CloseClientConnections();return map[string]any{"owned":"upstream","listener_close_error":cli43Error(e),"close_client_connections_returned":true}})
 cli43HTTPRegistryMu.Lock();if cli43HTTPRegistry[dir]!=nil{cli43HTTPRegistryMu.Unlock();t.Fatal("duplicate fixture registry")};cli43HTTPRegistry[dir]=b;cli43HTTPRegistryMu.Unlock()
 old:=http.DefaultTransport;http.DefaultTransport=&cli43BudgetTransport{base:old,book:b}
 base:=server.Client().Transport;if base==nil{base=old};server.Client().Transport=&cli43BudgetTransport{base:base,book:b}
 t.Cleanup(func(){http.DefaultTransport=old}) // No parallel tests; one selected leaf per fresh process.
}
func cli43RegisterProxy(dir string,proxy *httptest.Server){b:=cli43Book(dir);b.mu.Lock();defer b.mu.Unlock();b.stop=append(b.stop,func()map[string]any{e:=proxy.Listener.Close();proxy.CloseClientConnections();return map[string]any{"owned":"proxy","listener_close_error":cli43Error(e),"close_client_connections_returned":true}})}
func cli43HTTPFinish(t *testing.T,dir string)error{
 b:=cli43Book(dir);if b==nil{return fmt.Errorf("missing HTTP book")};b.mu.Lock();defer b.mu.Unlock()
 summary:=map[string]any{"upstream":b.upstream,"proxy":b.proxy,"upstream_received":b.ingress,"records":b.records,"first_overflow":b.firstOverflow,"stopped":b.stopped,"actual_received_HTTP_legs":b.ingress+b.proxy.Received,"qualifying_max_received_legs_per_case":64,"unobserved_client_outbound_attempts":nil,"setup_included_in_upstream":true,"pending":b.pending,"cap_boundary":"upstream outbound dispatch <=32; proxy business admission <=32; all observed ingress includes denied attempts"}
 if e:=cli43Save(dir,"http-summary.json",summary);e!=nil{return e}
 if b.stopped||b.upstream.Outstanding!=0||b.proxy.Outstanding!=0||b.upstream.Dispatched>32||b.proxy.Admitted>32||b.ingress+b.proxy.Received>64{return fmt.Errorf("HTTP budget/cleanup qualification failed")}
 if b.upstream.Attempt!=b.upstream.Completed||b.proxy.Attempt!=b.proxy.Completed||b.upstream.Dispatched!=b.ingress||len(b.pending)!=0{return fmt.Errorf("HTTP observation accounting mismatch")}
 return nil
}
