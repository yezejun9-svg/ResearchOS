88: func (s *Service) CreateBackupWithWorkspaces(ctx context.Context, destination string, key []byte, source WorkspaceBackupSource) (BackupManifest, error) {
89: 	if source == nil {

95: func workspaceBackupBindings(db *sql.DB) ([]WorkspaceBinding, error) {
96: 	rows, err := db.Query(`SELECT body FROM domain_objects WHERE project='' AND kind=? ORDER BY id`, KindWorkspaceBinding)
97: 	if err != nil {
98: 		return nil, err
99: 	}
100: 	defer rows.Close()
101: 	bindings := []WorkspaceBinding{}
102: 	for rows.Next() {
103: 		var raw string
104: 		if err := rows.Scan(&raw); err != nil {
105: 			return nil, err
106: 		}
107: 		binding, err := decodeBinding(raw)
108: 		if err != nil {
109: 			return nil, err
110: 		}
111: 		if binding.Status == WorkspaceActive {
112: 			bindings = append(bindings, *binding)
113: 		} else if binding.Status != WorkspaceRevoked {
114: 			return nil, fmt.Errorf("BACKUP_WORKSPACE_STATUS_INVALID")
115: 		}
116: 	}
117: 	return bindings, rows.Err()
118: }

128: func backupWorkspaceOverlaps(left, right string) bool {
129: 	a, ea := filepath.Abs(left)
130: 	b, eb := filepath.Abs(right)
131: 	if ea != nil || eb != nil {
132: 		return true
133: 	}
134: 	if strings.EqualFold(filepath.Clean(a), filepath.Clean(b)) {
135: 		return true
136: 	}
137: 	for _, pair := range [][2]string{{a, b}, {b, a}} {
138: 		rel, err := filepath.Rel(pair[0], pair[1])
139: 		if err == nil && rel != ".." && !strings.HasPrefix(rel, ".."+string(filepath.Separator)) && !filepath.IsAbs(rel) {
140: 			return true
141: 		}
142: 	}
143: 	return false
144: }
