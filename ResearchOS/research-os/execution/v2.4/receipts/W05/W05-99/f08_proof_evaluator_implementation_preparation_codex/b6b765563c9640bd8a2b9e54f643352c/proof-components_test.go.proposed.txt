import("database/sql/driver";"encoding/hex";"strings";"testing")

func f08ProofTestHex(t *testing.T,v any)string{t.Helper();b,e:=f08Encode(v);if e!=nil{t.Fatal(e)};return strings.ToUpper(hex.EncodeToString(b))}
func f08ProofTestDispatch()[]any{return []any{"p","run",f08Number(7),"dispatch","7B7D","manifest","STARTED","worker","dispatch","7B7D",f08Number(0),"","prepared","updated","reason"}}
func f08ProofTestSource(category string,row []any)*f08Sources{return &f08Sources{rows:map[string][][]any{category:{row}}}}
func f08ProofTestCall(t *testing.T,v *f08ProofValidator,mode int64,action string,row,old []any,sources []any)(driver.Value,error){t.Helper();var oldValue any;if old!=nil{oldValue=old};b,e:=f08Encode(map[string]any{"action":action,"row":row,"old":oldValue,"sources":sources});if e!=nil{t.Fatal(e)};return v.evaluate(mode,b)}

func TestF08ProofSourceMultisetAndKeyTypes(t *testing.T){
	relation:=[]any{"p","Task","task",f08Number(1),"/dependency_task_ids","p","Task","prior",f08Number(2)}
	t.Run("duplicate_relation_retained",func(t *testing.T){s,e:=f08DecodeSources([]any{[]any{"domain_refs",relation},[]any{"domain_refs",relation}});if e!=nil || len(s.rows["domain_refs"])!=2{t.Fatalf("actual duplicate multiset lost: %v",e)}})
	t.Run("duplicate_real_key_refused",func(t *testing.T){r:=[]any{"p","Task","t",f08Number(1),"7B7D",nil};if _,e:=f08DecodeSources([]any{[]any{"domain_versions",r},[]any{"domain_versions",r}});e==nil{t.Fatal("duplicate PK accepted")}})
	t.Run("integer_text_refused",func(t *testing.T){r:=append([]any(nil),relation...);r[3]="1";if _,e:=f08DecodeSources([]any{[]any{"domain_refs",r}});e==nil{t.Fatal("TEXT supplied as INTEGER")}})
	t.Run("nullable_sql_key_kept",func(t *testing.T){r:=[]any{nil,"p","run","7B7D"};s,e:=f08DecodeSources([]any{[]any{"routes",r},[]any{"routes",r}});if e!=nil || len(s.rows["routes"])!=2{t.Fatal("SQLite nullable key semantics lost",e)}})
	t.Run("unknown_category_refused",func(t *testing.T){if _,e:=f08DecodeSources([]any{[]any{"invented",relation}});e==nil{t.Fatal("unknown category")}})
}
func TestF08ProofDispatchUnknownMonotone(t *testing.T){
	v:=&f08ProofValidator{}
	for _,state:=range []string{"PREPARED","STARTED","UNKNOWN"}{t.Run(state+"_to_UNKNOWN",func(t *testing.T){old:=f08ProofTestDispatch();old[6]=state;row:=append([]any(nil),old...);row[6]="UNKNOWN";row[13]="later";row[14]="cleanup unavailable";if e:=v.dispatchUpdate(f08ProofTestSource("dispatches",old),old,row);e!=nil{t.Fatal(e)}})}
	t.Run("identity_erasure_refused",func(t *testing.T){old:=f08ProofTestDispatch();old[9]=f08ProofTestHex(t,map[string]any{"pid":f08Number(2)});row:=append([]any(nil),old...);row[6]="UNKNOWN";row[9]="7B7D";if e:=v.dispatchUpdate(f08ProofTestSource("dispatches",old),old,row);e==nil{t.Fatal("erased identity")}})
	t.Run("observed_to_unknown_refused",func(t *testing.T){old:=f08ProofTestDispatch();old[6]="OBSERVED";row:=append([]any(nil),old...);row[6]="UNKNOWN";if e:=v.dispatchUpdate(f08ProofTestSource("dispatches",old),old,row);e==nil{t.Fatal("OBSERVED rewind")}})
	t.Run("source_old_mismatch_refused",func(t *testing.T){old:=f08ProofTestDispatch();row:=append([]any(nil),old...);row[6]="UNKNOWN";actual:=append([]any(nil),old...);actual[3]="other";if e:=v.dispatchUpdate(f08ProofTestSource("dispatches",actual),old,row);e==nil{t.Fatal("caller OLD substituted")}})
}

func TestF08ProofPendingExactOriginalCommand(t *testing.T){
	q:=make([]any,18);q[1],q[2],q[3],q[14]="p","other-queue","s","SENDING"
	c1:=make([]any,12);c1[1],c1[2],c1[3],c1[8]="p","this","s","REQUESTED"
	c2:=append([]any(nil),c1...);c2[2]="other-control"
	s:=&f08Sources{rows:map[string][][]any{"queue":{q},"controls":{c1,c2}}}
	t.Run("other_pending_kept_sorted",func(t *testing.T){if !f08Equal(f08Pending(s,"p","s","this"),[]any{"other-control","other-queue"}){t.Fatal("other pending removed")}})
	t.Run("queue_same_command_not_hidden",func(t *testing.T){q[2]="this";if !f08Equal(f08Pending(s,"p","s","this"),[]any{"other-control","this"}){t.Fatal("pending queue hidden by control removal")}})
}

func TestF08ProofHistoricalEffectPrefix(t *testing.T){
	one:=map[string]any{"effect_key":"key","sequence":f08Number(1),"epoch":f08Number(7),"body_digest":"sha256:"+strings.Repeat("a",64)}
	two:=map[string]any{"effect_key":"key","sequence":f08Number(2),"epoch":f08Number(8),"body_digest":"sha256:"+strings.Repeat("b",64)}
	t.Run("legal_append_not_replacement",func(t *testing.T){if e:=f08EffectPrefix([]any{one},[]any{one,two});e!=nil{t.Fatal(e)}})
	t.Run("original_deleted",func(t *testing.T){if e:=f08EffectPrefix([]any{one},[]any{two});e==nil{t.Fatal("lost original")}})
	t.Run("epoch_relabelled",func(t *testing.T){changed:=map[string]any{};for k,v:=range one{changed[k]=v};changed["epoch"]=f08Number(8);if e:=f08EffectPrefix([]any{one},[]any{changed,two});e==nil{t.Fatal("old epoch reassigned")}})
	t.Run("duplicate_key_sequence",func(t *testing.T){if e:=f08EffectPrefix([]any{one},[]any{one,one});e==nil{t.Fatal("duplicate sequence")}})
}
