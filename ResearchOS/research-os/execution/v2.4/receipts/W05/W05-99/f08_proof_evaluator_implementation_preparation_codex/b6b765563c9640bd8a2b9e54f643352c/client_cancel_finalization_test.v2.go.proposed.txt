func f08TestRequireOne(t *testing.T, fn f08Scalar, args ...driver.Value) {
	t.Helper()
	v, err := fn(nil, args)
	if err != nil || v != int64(1) { t.Fatalf("expected int64(1), value type=%T error=%v", v, err) }
}

func f08TestRequireError(t *testing.T, fn f08Scalar, expected f08Error, args ...driver.Value) {
	t.Helper()
	v, err := fn(nil, args)
	if err == nil || v != nil { t.Fatalf("expected fail-closed nil/error, type=%T err=%v", v, err) }
	if expected != "" && err != expected { t.Fatalf("wrong fixed error: %v", err) }
}

func f08TestBinding(t *testing.T, f f08TestFixture, old, current []byte, snapshot map[string]any) []byte {
	t.Helper()
	p := map[string]any{"admitted_root_hash":"sha256:"+strings.Repeat("a",64), "original_response_digest":f08RawHash(current), "final_root_hash":nil, "final_response_digest":nil, "final_snapshot_binding":nil}
	if snapshot != nil {
		pendingHash, err := f08ValueHash(snapshot["pending_command_ids"])
		if err != nil { t.Fatal(err) }
		p["final_root_hash"] = "sha256:"+strings.Repeat("b",64)
		p["final_response_digest"] = f08RawHash(current)
		p["final_snapshot_binding"] = map[string]any{"session_id":"session-one", "project_id":"project-one", "revision":snapshot["revision"], "event_sequence":snapshot["event_sequence"], "pending_command_ids_digest":pendingHash}
	}
	if old != nil { p["original_response_digest"] = f08RawHash(old) }
	return f08TestJSON(t, map[string]any{"identity":f.identityObject, "response_proof":p})
}

	root:=map[string]any{"format":f08Format,"phase":"ADMITTED","principal":"principal","project":"p","command_id":"c","method":"session.control","operation":"v2:session.control","key":"c","request_hash":"request","client_id":"client","session_binding":nil,"target_binding":nil,"private_header_digest":"header","member_task_ids":[]any{"task"},"member_count":f08Number(1)}
	rh,e:=f08Hash(root);if e!=nil{t.Fatal(e)}
	task:=map[string]any{"entity_type":"Task","id":"task","revision":f08Number(1),"content_hash":nil}
	member:=map[string]any{"format":f08Format,"phase":"ADMITTED","admitted_root_ref":f08RootReference("ADMITTED",rh),"task_ref":task,"goal_ref":nil,"run_ref":nil,"runtime_binding":nil,"dispatch_stage":"NO_RUN","dispatch_origin_ref":nil,"queue_admission":nil,"reference_closure":[]any{}}
	mh,e:=f08Hash(member);if e!=nil{t.Fatal(e)};gh,e:=f08Hash([]any{map[string]any{"task_id":"task","member_hash":mh}});if e!=nil{t.Fatal(e)}
	parent:=[]any{"principal","p","c","request","v2:session.control","7B7D",rh,gh,f08ProofTestHex(t,map[string]any{"root":root,"members":[]any{member},"pins":[]any{}})}
	child:=[]any{"principal","p","c","request","task",f08Number(0),rh,mh,f08ProofTestHex(t,member)}
	v:=&f08ProofValidator{};s:=f08ProofTestSource("admitted",parent)
	t.Run("exact_parent_member",func(t *testing.T){if e:=v.materialization(s,"admitted_member.insert",child);e!=nil{t.Fatal(e)}})
	t.Run("wrong_index",func(t *testing.T){r:=append([]any(nil),child...);r[5]=f08Number(1);if e:=v.materialization(s,"admitted_member.insert",r);e==nil{t.Fatal("index alias")}})
	t.Run("wrong_original_task",func(t *testing.T){r:=append([]any(nil),child...);r[4]="other";if e:=v.materialization(s,"admitted_member.insert",r);e==nil{t.Fatal("task alias")}})
	t.Run("missing_parent",func(t *testing.T){if e:=v.materialization(&f08Sources{rows:map[string][][]any{}},"admitted_member.insert",child);e==nil{t.Fatal("orphan child")}})
}
