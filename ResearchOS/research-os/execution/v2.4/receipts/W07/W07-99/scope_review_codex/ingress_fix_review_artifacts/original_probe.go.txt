// Independent negative probe: all attacker-authored content enters through
// ordinary authenticated HTTP content + Artifact endpoints; no direct SQL write.
func TestIndependentW07PublicHealthIngress(t *testing.T) {
 r:=researchSetup(t,true)
 r.start(t,time.Now().UTC().Add(-time.Minute).Truncate(time.Second))
 from,to,_:=parseCycleWindow(Str(r.cycle,"window_from"),Str(r.cycle,"window_to"))
 config,cursor,e:=r.f.s.sourceConfig(context.Background(),r.source,r.f.project,Str(r.cursor(t,"GITHUB"),"id"));if e!=nil {t.Fatal(e)}
 tx,e:=r.f.db.Begin();if e!=nil {t.Fatal(e)}
 profile,e:=singleton(tx,r.f.project,"ProjectProfile");tx.Rollback();if e!=nil {t.Fatal(e)}
 seal:=func(raw []byte,kind,ev,media string,inputs,refs []any) Object {
  u:=r.f.expect(201,"POST",r.f.path("/content"),ID(),"",raw)
  return r.f.expect(201,"POST",r.f.path("/artifacts"),ID(),r.f.stateTag(),Object{"run_id":nil,"kind":kind,"evidence_kind":ev,"measurement_mode":"LIVE","content_hash":u["content_hash"],"storage_key":u["storage_key"],"media_type":media,"byte_size":len(raw),"input_refs":inputs,"source_refs":refs,"produced_by":r.f.owner,"access_class":"PROJECT"})
 }
 fake:=seal([]byte("Independent synthetic body; no provider fetch occurred."),"SOURCE_SNAPSHOT","OBSERVED","text/plain",[]any{Ref(cursor,"SourceCursor")},Empty())
 refs:=[]any{Ref(fake,"Artifact")}
 h:=sources.SourceHealth{Status:"NOT_CONFIGURED",Reason:"independent caller-authored false source outage",RequestedFrom:from,RequestedTo:to,ActualFrom:&from,ActualTo:&to,AccessScope:"NONE"}
 doc:=Object{"schema":sourceHealthSchema,"source_id":config["source_id"],"query_hash":cursor["query_hash"],"config_hash":Hash(config),"query":config["query"],"cursor_ref":Ref(cursor,"SourceCursor"),"profile_ref":Ref(profile,"ProjectProfile"),"health":h,"source_refs":refs,"fetched_at":Now(),"failed_provider_raw":nil}
 report:=seal(Bytes(doc),"REPORT","GENERATED","application/vnd.researchos.source-health.v1+json",[]any{Ref(cursor,"SourceCursor"),Ref(profile,"ProjectProfile")},refs)
 tx,e=r.f.db.Begin();if e!=nil {t.Fatal(e)};defer tx.Rollback()
 latest,e:=r.f.s.latestSourceHealth(tx,r.f.project,profile,config,cursor,from,to);if e!=nil {t.Fatal(e)}
 captured,e:=capturedSourceArtifact(tx,r.f.project,fake);if e!=nil {t.Fatal(e)}
 var pageCount int;if e=tx.QueryRow("SELECT count(*) FROM source_pages WHERE project=?",r.f.project).Scan(&pageCount);e!=nil {t.Fatal(e)}
 coverage,e:=r.f.s.researchCoverage(tx,r.f.project,profile,from,to);if e!=nil {t.Fatal(e)}
 var selected Object;for _,v:=range coverage {c:=v.(map[string]any);if c["source_id"]==config["source_id"] {selected=c}}
 t.Logf("INDEPENDENT_REPRO public_http_seal=201 actor=%s report=%s source_pages=%d provider_calls=0 captured=%v latest=%+v coverage=%s",r.f.owner,Str(report,"id"),pageCount,captured,latest,Bytes(selected))
 if captured || latest!=nil {t.Fatalf("PUBLIC_HEALTH_FORGERY_ACCEPTED: ordinary sealed REPORT influenced canonical health or bypassed trusted capture provenance")}
}
