+def cim_birth_microsecond(row: dict) -> int:
+    """Validate CIM's microsecond timestamp; do not claim 100ns precision."""
+    need(type(row) is dict and type(row.get('CreationTime100nsFromCIM')) is int and
+         type(row.get('CreationDateCimUtc')) is str,
+         'CIM creation timestamp absent')
+    parsed = stamp(row['CreationDateCimUtc']).astimezone(datetime.timezone.utc)
+    epoch = datetime.datetime(1601, 1, 1, tzinfo=datetime.timezone.utc)
+    elapsed = parsed - epoch
+    calculated = ((elapsed.days * 86400 + elapsed.seconds) * 10_000_000 +
+                  elapsed.microseconds * 10)
+    derived = row['CreationTime100nsFromCIM']
+    need(derived == calculated and derived > 0 and derived % 10 == 0,
+         'CIM creation ISO and converted microsecond mismatch')
+    return derived // 10
