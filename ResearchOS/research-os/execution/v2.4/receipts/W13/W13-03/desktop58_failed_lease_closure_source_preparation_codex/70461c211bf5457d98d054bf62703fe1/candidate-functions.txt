def all_refs(value):
    if isinstance(value,dict):
        if set(value)=={"path","sha256","bytes"}:
            yield value
        else:
            for item in value.values(): yield from all_refs(item)
    elif isinstance(value,list):
        for item in value: yield from all_refs(item)
