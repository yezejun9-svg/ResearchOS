package evaluation
import("testing";"fmt";"encoding/json";"os";"path/filepath")
func reviewCheck(t *testing.T,id string,want,got any){t.Helper();fmt.Println("INDEPENDENT_W10 "+string(Bytes(Object{"id":id,"expected":want,"actual":got})));if string(Bytes(want))!=string(Bytes(got)){t.Errorf("%s want %v got %v",id,want,got)}}
func TestIndependentRawRegression(t *testing.T){for _,mode:=range []string{"normal","duplicate","unknown","rawrows","rawhash","config","hard"}{t.Run(mode,func(t *testing.T){
 original,e:=os.ReadFile(filepath.Join(os.Getenv("W10_EARLY_OUT"),"evaluation-"+mode+".json"));if e!=nil{t.Fatal(e)}
 var old struct{Manifest Manifest `json:"manifest"`;Observed Manifest `json:"observed_manifest"`;Raw string `json:"raw_export"`;Hash string `json:"raw_export_hash"`;Normalized Object `json:"normalized"`};if e=json.Unmarshal(original,&old);e!=nil{t.Fatal(e)}
 exit:=0;raw:=[]byte(old.Raw);r,e:=Normalize(old.Manifest,old.Observed,raw,old.Hash,&exit,1);if e!=nil{t.Fatal(e)}
 want:="COMPLETE";if mode=="duplicate"||mode=="unknown"||mode=="rawhash"||mode=="config"{want="INVALID"};if mode=="rawrows"{want="INCOMPLETE"}
 reviewCheck(t,mode+"_integrity_from_original_raw",want,r["integrity_status"])
 changed:=old.Normalized;changed["integrity_status"]="COMPLETE";changed["counts"].(map[string]any)["duplicate_count"]=float64(0)
 validation:=Validate(old.Manifest,changed,raw,old.Hash)
 reviewCheck(t,mode+"_original_forged_report_rejected",mode!="normal"&&mode!="hard",validation!=nil)
 if mode=="rawrows"{reviewCheck(t,"rawrows_actual_error_preserved","ERROR",r["results"].([]Result)[0].Verdict);reviewCheck(t,"rawrows_missing_stays_missing","MISSING",r["results"].([]Result)[1].Verdict)}
 if mode=="normal"{changed["results"].([]any)[0].(map[string]any)["reason"]="forged reason";reviewCheck(t,"changed_result_rejected_against_original_raw",true,Validate(old.Manifest,changed,raw,old.Hash)!=nil)
 tampered:=append([]byte{},raw...);tampered=append(tampered,' ');reviewCheck(t,"changed_raw_bytes_cannot_reuse_frozen_hash",true,Validate(old.Manifest,r,tampered,old.Hash)!=nil)}
 if mode=="hard"{reviewCheck(t,"hard_fail_remains_despite_high_score","FAIL",r["results"].([]Result)[0].Verdict)}
 os.WriteFile(filepath.Join(os.Getenv("W10_REVIEW_OUT"),"evaluation-"+mode+".json"),Bytes(Object{"early_fixture_sha256":Digest(original),"raw_export":old.Raw,"frozen_raw_hash":old.Hash,"actual_raw_hash":Digest(raw),"normalized":r,"early_laundered_validation_error":fmt.Sprint(validation)}),0600)
})}}
