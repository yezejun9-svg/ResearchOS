func sourceSnapshot(t *testing.T,root string)map[string]any{t.Helper();out:=map[string]any{};e:=filepath.WalkDir(root,func(p string,d os.DirEntry,e error)error{if e!=nil{return e};sd,e:=windows.GetNamedSecurityInfo(p,windows.SE_FILE_OBJECT,windows.DACL_SECURITY_INFORMATION|windows.LABEL_SECURITY_INFORMATION);if e!=nil{return e};rel,_:=filepath.Rel(root,p);v:=map[string]any{"security_descriptor":sd.String(),"directory":d.IsDir()};if !d.IsDir(){raw,e:=os.ReadFile(p);if e!=nil{return e};v["sha256"]=fmt.Sprintf("%x",sha256.Sum256(raw))};out[rel]=v;return nil});if e!=nil{t.Fatal(e)};return out}
