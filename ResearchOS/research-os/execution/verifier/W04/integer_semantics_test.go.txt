
func TestVerifierLargeIntegerIdempotencyProbe(t *testing.T) {
 f:=setup(t);f.createProject();req:=f.requirement()
 contract:=Object{"contract_hash":Digest([]byte("fixture contract")),"contract_revision":1,"checks":[]any{Object{"check_id":"fixture-check","kind":"DETERMINISTIC","required":true,"expected":"actual evidence persists","command_key":"test.domain","rubric_ref":nil,"qualification_id":nil,"not_applicable_reason":nil}},"require_independent_verifier":true,"version_policy":"EXACT","frozen_at":Now()}
 goal:=f.expect(201,"POST",f.path("/goals"),ID(),f.stateTag(),Object{"title":"fixture goal","original_intent":"persist actual records","scope":"PROJECT","requirement_refs":[]any{Ref(req,"Requirement")},"problem_ids":[]any{"origin-problem"},"deliverables":[]any{"fixture record"},"acceptance_contract":contract})
 body:=Object{"goal_id":goal["id"],"title":"numeric probe","task_type":"fixture","stage":"verification","dependency_task_ids":Empty(),"requirement_refs":[]any{Ref(req,"Requirement")},"input_refs":Empty(),"acceptance_contract":contract,"origin_problem_id":"origin-problem","candidate_id":"candidate-one","stop_policy":Object{"wall_time_seconds":json.Number("9007199254740992"),"max_attempts":3,"max_parallel_runs":1,"stop_conditions":[]any{"no effective progress"}}}
 key,tag:=ID(),testTag(goal);firstHash:=Hash(body);status,first,_:=f.request("POST",f.path("/tasks"),key,tag,body);if status!=201{t.Fatalf("first rejected %d %s",status,Bytes(first))}
 body["stop_policy"].(map[string]any)["wall_time_seconds"]=json.Number("9007199254740993");secondHash:=Hash(body);status,second,h:=f.request("POST",f.path("/tasks"),key,tag,body)
 var raw string;if e:=f.db.QueryRow("SELECT body FROM domain_objects WHERE project=? AND kind='Task' AND id=?",f.project,first["id"]).Scan(&raw);e!=nil{t.Fatal(e)};stored,e:=Decode([]byte(raw));if e!=nil{t.Fatal(e)};storedN,e:=Num(stored["stop_policy"].(map[string]any),"wall_time_seconds");if e!=nil{t.Fatal(e)};requestedN,e:=Num(body["stop_policy"].(map[string]any),"wall_time_seconds");if e!=nil{t.Fatal(e)}
 t.Logf("status=%d replay=%s same_task=%v equal_hash=%v stored=%d second_requested=%d",status,h.Get("Idempotency-Replayed"),first["id"]==second["id"],firstHash==secondHash,storedN,requestedN)
 if status==201&&h.Get("Idempotency-Replayed")=="true"&&storedN!=requestedN{t.Fatal("REPRODUCED: semantically distinct integer input accepted as same idempotent request")}
}
