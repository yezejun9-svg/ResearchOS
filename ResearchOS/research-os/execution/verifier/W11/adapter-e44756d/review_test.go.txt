func rvAdapterCheck(t *testing.T, id string, want, got any) {
	t.Helper()
	status := "PASS"
	if string(Encode(want)) != string(Encode(got)) {
		status = "FAIL"
	}
	fmt.Println("RV11_ADAPTER_ASSERTION " + string(Encode(map[string]any{"id": id, "status": status, "expected": want, "actual": got})))
	if status != "PASS" {
		t.Errorf("%s mismatch", id)
	}
}

func rvChoiceEnvelope(choice string, confidence float64, probabilities map[string]float64) Envelope {
	request := Request{Model: "independent-revision", State: map[string]any{"synthetic": "independent adapter boundary"}, Questions: map[string]Question{"action": {Type: "choice", Instructions: "choose highest probability", Criteria: map[string]string{"KEEP": "valid", "DROP": "invalid", "WAIT": "insufficient"}}}}
	response := map[string]any{"model": "independent-revision", "answers": map[string]any{"action": map[string]any{"type": "choice", "choice": choice, "confidence": confidence, "probabilities": probabilities}}, "usage": map[string]any{"input_tokens": 3, "output_tokens": 2}}
	raw := Encode(response)
	return Envelope{Version: 1, Binding: Binding{ProjectID: "independent", InputHash: Digest([]byte("input")), RubricHash: Digest([]byte("rubric")), Scope: Scope{TaskType: "review", Language: "zh-en", Provider: "typesafe", ModelRevision: request.Model, RubricRevision: "7"}}, Request: request, RequestHash: Digest(Encode(request)), Response: raw, ResponseHash: Digest(raw), HTTPStatus: 200, MeasurementMode: "FIXTURE"}
}

func TestRV11ConfidenceDelta(t *testing.T) {
	for _, tc := range []struct {
		name, choice string
		confidence   float64
		prob         map[string]float64
		abstain      bool
		reason       string
	}{
		{"official-example", "KEEP", .81, map[string]float64{"KEEP": .88, "DROP": .12, "WAIT": 0}, false, ""},
		{"actual-live-shape", "KEEP", .89, map[string]float64{"KEEP": .94, "DROP": .06, "WAIT": 0}, false, ""},
		{"confidence-above-probability", "KEEP", .91, map[string]float64{"KEEP": .70, "DROP": .20, "WAIT": .10}, false, ""},
		{"threshold-exact", "KEEP", .8, map[string]float64{"KEEP": .94, "DROP": .06, "WAIT": 0}, false, ""},
		{"threshold-below", "KEEP", math.Nextafter(.8, 0), map[string]float64{"KEEP": .94, "DROP": .06, "WAIT": 0}, true, "action:LOW_CONFIDENCE"},
		{"probability-high-confidence-low", "KEEP", .74, map[string]float64{"KEEP": .99, "DROP": .01, "WAIT": 0}, true, "action:LOW_CONFIDENCE"},
		{"unknown-choice", "ERASE", .9, map[string]float64{"KEEP": .94, "DROP": .06, "WAIT": 0}, true, "UNKNOWN_ACTION"},
		{"nonmaximum", "DROP", .99, map[string]float64{"KEEP": .94, "DROP": .06, "WAIT": 0}, true, "CHOICE_NOT_MAXIMUM"},
		{"nonmaximum-third", "WAIT", .9, map[string]float64{"KEEP": .3, "DROP": .6, "WAIT": .1}, true, "CHOICE_NOT_MAXIMUM"},
		{"nonmaximum-beyond-tolerance", "KEEP", .9, map[string]float64{"KEEP": .49998, "DROP": .50002, "WAIT": 0}, true, "CHOICE_NOT_MAXIMUM"},
		{"tie-permitted", "KEEP", .9, map[string]float64{"KEEP": .5, "DROP": .5, "WAIT": 0}, false, ""},
		{"roundoff-tolerance", "KEEP", .9, map[string]float64{"KEEP": .499998, "DROP": .500002, "WAIT": 0}, false, ""},
		{"negative-confidence", "KEEP", -.01, map[string]float64{"KEEP": .94, "DROP": .06, "WAIT": 0}, true, "INVALID_PROBABILITY"},
		{"oversized-confidence", "KEEP", 1.01, map[string]float64{"KEEP": .94, "DROP": .06, "WAIT": 0}, true, "INVALID_PROBABILITY"},
		{"bad-distribution", "KEEP", .9, map[string]float64{"KEEP": .94, "DROP": .06, "WAIT": .1}, true, "INVALID_PROBABILITY_SUM"},
	} {
		t.Run(tc.name, func(t *testing.T) {
			r := Evaluate(rvChoiceEnvelope(tc.choice, tc.confidence, tc.prob), .8)
			reason := ""
			if len(r.Reasons) > 0 {
				reason = r.Reasons[0]
			}
			rvAdapterCheck(t, "CONFIDENCE-"+tc.name, []any{tc.abstain, tc.reason}, []any{r.Abstained, reason})
		})
	}
	env := rvChoiceEnvelope("KEEP", .81, map[string]float64{"KEEP": .88, "DROP": .12, "WAIT": 0})
	rvAdapterCheck(t, "FROZEN-THRESHOLD-NOT-LOWERED", true, Evaluate(env, .85).Abstained)
	rvAdapterCheck(t, "INVALID-THRESHOLD-REJECT", true, Evaluate(env, .49).Abstained)
}

type rvNoNetwork struct{ t *testing.T }

func (x rvNoNetwork) RoundTrip(r *http.Request) (*http.Response, error) {
	x.t.Fatalf("offline replay attempted network: %s", r.URL)
	return nil, fmt.Errorf("network forbidden")
}

func TestRV11OfflineActualResponses(t *testing.T) {
	// Fail even accidental HTTP access; Evaluate itself is a pure function.
	old := http.DefaultTransport
	http.DefaultTransport = rvNoNetwork{t}
	defer func() { http.DefaultTransport = old }()
	root := os.Getenv("RV11_ROOT")
	if root == "" {
		t.Fatal("RV11_ROOT required")
	}
	dataRaw, e := os.ReadFile(filepath.Join(root, "evaluation", "development", "jev", "dataset.json"))
	if e != nil {
		t.Fatal(e)
	}
	var data struct {
		Rubric struct {
			ConfidenceThreshold float64 `json:"confidence_threshold"`
		} `json:"rubric"`
		Cases []struct {
			ID       string `json:"id"`
			Scope    Scope  `json:"scope"`
			Expected string `json:"expected"`
		} `json:"cases"`
	}
	if e = json.Unmarshal(dataRaw, &data); e != nil {
		t.Fatal(e)
	}
	rvAdapterCheck(t, "REPLAY-THRESHOLD-FROZEN", .8, data.Rubric.ConfidenceThreshold)
	run := filepath.Join(root, "execution", "receipts", "W11", "development-live", "3dd0b57c-5a59-4b59-9201-951d6ecd3be1")
	count := 0
	for _, c := range data.Cases {
		raw, e := os.ReadFile(filepath.Join(run, c.ID+".raw.json"))
		if e != nil {
			t.Fatal(e)
		}
		var envelope Envelope
		if e = Strict(raw, &envelope); e != nil {
			t.Fatal(e)
		}
		r := Evaluate(envelope, data.Rubric.ConfidenceThreshold)
		fmt.Println("RV11_ADAPTER_REPLAY " + string(Encode(map[string]any{"id": c.ID, "scope": c.Scope, "expected": c.Expected, "selected": r.Selected, "abstained": r.Abstained, "reasons": r.Reasons, "usage": r.Usage, "response_hash": envelope.ResponseHash, "raw_file_hash": Digest(raw)})))
		rvAdapterCheck(t, "REPLAY-"+c.ID+"-PARSEABLE", true, r.Usage != nil)
		count++
	}
	rvAdapterCheck(t, "REPLAY-EXACT-60", 60, count)
}
