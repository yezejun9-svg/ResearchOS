package sources
import("bytes";"context";"fmt";"net/http";"net/http/httptest";"testing";"time")
func TestVerifierArxivPartialAndRedirect(t *testing.T){
 for _,redirect:=range []bool{false,true}{t.Run(fmt.Sprintf("redirect_%t",redirect),func(t *testing.T){
  foreignHits:=0; fragment:=append([]byte("%PDF-1.7\n"),bytes.Repeat([]byte("incomplete-page-stream "),100)...)
  foreign:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){foreignHits++;w.Header().Set("Content-Type","application/pdf");w.(http.Flusher).Flush();w.Write(fragment)}));defer foreign.Close()
  configured:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){if r.URL.Path=="/api/query"{fmt.Fprint(w,`<feed><entry><id>http://arxiv.org/abs/2304.03442v2</id><published>2023-04-07T00:00:00Z</published><updated>2023-07-01T00:00:00Z</updated><author><name>Author</name></author></entry></feed>`);return};if redirect{http.Redirect(w,r,foreign.URL+"/unexpected-origin",http.StatusFound);return};w.Header().Set("Content-Type","application/pdf");w.(http.Flusher).Flush();w.Write(fragment)}));defer configured.Close()
  a:=&Arxiv{Client:configured.Client(),APIURL:configured.URL+"/api/query",PDFBaseURL:configured.URL+"/pdf",MinInterval:time.Nanosecond}
  p,e:=a.Fetch(context.Background(),Query{SourceID:"paper",Text:"id:2304.03442v2",From:time.Date(2023,1,1,0,0,0,0,time.UTC),To:time.Date(2026,9,30,0,0,0,0,time.UTC),Limit:1})
  canonical:="";if len(p.Items)>0{canonical=p.Items[0].CanonicalURL}
  t.Logf("error=%v complete=%v failure=%q items=%d fragment_bytes=%d foreign_hits=%d reported_origin=%s",e,p.Complete,p.Failure,len(p.Items),len(fragment),foreignHits,canonical)
  if p.Complete&&len(p.Items)==1{t.Fatal("REPRODUCED: incomplete PDF accepted as complete; redirect case also fetched an unconfigured origin")}
 })}
}
